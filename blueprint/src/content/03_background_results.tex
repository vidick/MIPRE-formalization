other papers. Each entry gives a full statement, a pointer to the literature, and
comments on its role and on formalization. The project may initially assume these as
axioms; discharging them are independent sub-projects. Three of these results are
larger sub-projects and are developed in their own sections: the computability
results in Section~\ref{ch:computability} and parallel repetition in
Section~\ref{ch:parallel-repetition}; they are included in the summary table below
for completeness. Statements are given in the form closest to how they will be used.
The compression pipeline carries bipartite soundness in $\valstar$; synchronous and
tracial results appear as internal ingredients with their own transfer obligations.

\effortEasy{} for self-contained arguments of at most a few pages,

\effortHard{} for results whose formalization is a project in itself;

\subsection{The Gowers--Hatami theorem}
\label{sec:rr-gowers-hatami}

\qlib

\begin{definition}[Approximate representation]
\label{def:approx-rep}
\uses{def:hs-norm}
\lean{MIPRE.IsApproxRepresentation}
\leanok
\qlib{}
Let $G$ be a finite group and $\eps \in \R$. An \emph{$\eps$-approximate
representation} of $G$ is a function $f : G \to U(d)$ such that
\begin{equation*}
  \E_{x, y \sim G} \, \mathrm{Re}\, \langle f(x) f(y), f(xy) \rangle_{hs}
  \geq 1 - \eps\;,
\end{equation*}
where $\langle \cdot, \cdot \rangle_{hs}$ is the normalized Hilbert--Schmidt
inner product of Definition~\ref{def:hs-norm}. Since each value of $f$ is
unitary, $\norm{f(x) f(y) - f(xy)}_{hs}^2 = 2 - 2\, \mathrm{Re}\, \langle f(x)
f(y), f(xy) \rangle_{hs}$, so the condition is equivalent to the defect form
$\E_{x, y \sim G} \norm{f(x) f(y) - f(xy)}_{hs}^2 \leq 2 \eps$.
\end{definition}

\begin{theorem}[Gowers--Hatami]
\label{thm:gowers-hatami}
\uses{def:hs-norm,def:approx-rep}
\lean{MIPRE.gowers_hatami}
\leanok
Let $G$ be a finite group and $f : G \to U(d)$ an $\eps$-approximate
representation of $G$. Then there exist $d' \geq d$, an isometry
$V : \C^d \to \C^{d'}$, and a representation $\rho : G \to U(d')$ such that
\begin{equation*}
  \E_{x \sim G} \, \norm{f(x) - V^\dagger \rho(x) V}_{hs}^2 \leq 2 \eps\;.
\end{equation*}
\end{theorem}

\begin{proof}
\leanok
Let $L(G, \C^d) \cong \C^{|G| d}$ carry the inner product normalized by $1/|G|$,
let $V : \C^d \to L(G, \C^d)$ be the map $u \mapsto (x \mapsto f(x) u)$, and let
$\rho$ be the right regular representation, $(\rho(x) F)(y) = F(yx)$. Then $V$ is
an isometry, and the compression $V^\dagger \rho(x) V = \E_{y \sim G}
f(y)^\dagger f(yx)$ is an average of unitaries, hence has squared
Hilbert--Schmidt norm at most $1$. Moreover
$\E_{x} \mathrm{Re}\, \langle f(x), V^\dagger \rho(x) V \rangle_{hs} =
\E_{x, y} \mathrm{Re}\, \langle f(x) f(y), f(xy) \rangle_{hs} \geq 1 - \eps$ by
hypothesis, and the claim follows by expanding
$\norm{f(x) - V^\dagger \rho(x) V}_{hs}^2$. The construction gives
$d' = |G|\, d$.
\end{proof}

\paragraph{Source.} Gowers and Hatami~\cite{GH17}. A quantitative form suited to
applications in nonlocal games, and a version adapted to the tracial setting, is given
by de la Salle~\cite{dlS21} and in~\cite{Vid22}.

\paragraph{Comments.} This is the stability result underlying all rigidity
(self-testing) arguments in the proof: an approximate representation of the Pauli group
$\F_2^{2k}$ (with its cocycle) is close to an exact one, which forces the presence of
EPR pairs. The statement above is fully formalized (statement and proof) in
\texttt{MIPRE/Background/GowersHatami/Basic.lean}, following the regular-representation
argument sketched in the proof; the intermediate statement, with the exact
representation living on the concrete index set $G \times \{1, \dots, d\}$, is
\texttt{MIPRE.gowers\_hatami\_prod}. Two remarks on the statement. (i) It asserts
only $d' \geq d$, with witness $d' = |G| d$; it does not include the dimension bound
$d' \leq (1 + C\eps) d$ of~\cite{GH17}, whose proof goes through the Fourier-analytic
route via irreducible representations. Whether the sharper bound is needed downstream
(in the Pauli-group application feeding Theorem~\ref{thm:qld}) is to be revisited; if
so it will be added as a separate strengthening. (ii) In the Lean encoding the
isometry is stored as the $d \times d'$ matrix of its adjoint, so the isometry
condition reads $V V^\dagger = \Id$ and the compression is written
$V \rho(x) V^\dagger$. The normalized Hilbert--Schmidt norm is the tracial form
needed for the synchronous framework~\cite{dlS21,Vid22}.

\subsection{Almost-synchronous correlations}
\label{sec:rr-almost-sync}

\effortHard\ \qlib

\begin{theorem}[Almost-synchronous strategies are near synchronous strategies]
\label{thm:almost-sync}
\ledgernode{1.2.1.7.2, 1.2.1.7.2.1, 1.2.1.7.2.2}
\uses{def:sync-game,def:sync-value,def:q-value,def:inconsistency}
There are universal constants $C > 0$ and $0 < c \leq 1$ such that the following holds.
Let $\game$ be a synchronous game whose question distribution has the same marginal
$\nu$ on both coordinates and gives the diagonal pointwise proportional weight:
for some $\kappa \in (0, 1]$,
\begin{equation*}
  \mu(x,x) \;\geq\; \kappa\nu(x) \qquad \text{for all } x \in \mX\;.
\end{equation*}
Then, writing $\eps = 1 - \valstar(\game)$,
\begin{equation*}
  1 - \synval(\game) \;\leq\; C \bigl( \eps / \kappa \bigr)^{c}\;.
\end{equation*}
More precisely, the correlation of every tensor-product strategy of value $1 - \eps$
is approximated, on average over the game's questions, by a mixture of synchronous
correlations. The mixture has average game value at least $1 - C(\eps/\kappa)^c$,
so at least one component attains that bound; the bound is not asserted for every component.
\end{theorem}

\paragraph{Source.} Vidick~\cite{Vid22}, building on~\cite{KPS18,PSSTW16}; the
statement above is the corollary needed here, with constants independent of the game's
size. What the source controls is not the game's value directly but the
\emph{synchronicity defect} of a strategy --- its inconsistency
(Definition~\ref{def:inconsistency}) at equal questions, taken against the common
marginal $\nu$; the two-sided correlation replacement uses that common marginal.
The diagonal-weight hypothesis is what turns the game's rejection
probability $\eps$ into the bound $\eps/\kappa$ on that defect, the pairs $(x,x)$ carrying
at least a $\kappa$ fraction of the weight of $x$. Without it the two values are unrelated
(Remark~\ref{rem:almost-sync-hypothesis}).

\paragraph{Comments.} The proof uses the orthonormalization
lemma (Section~\ref{sec:rr-orthonormalization}) and a rounding argument. Used
contrapositively, the theorem turns a
bound $\synval(\game) \leq 1 - \delta$ into $\valstar(\game) \leq 1 - \Omega(\kappa
\delta^{1/c})$: the diagonal weight multiplies the gap recovered, so a small $\kappa$ is a
loss. The hypothesis is also not preserved by repetition. The pipeline instead carries
soundness in $\valstar$ throughout and uses direct bipartite repetition.

\begin{remark}[The diagonal-weight hypothesis cannot be dropped]
\label{rem:almost-sync-hypothesis}
\uses{thm:almost-sync,def:sync-game,def:sync-value,def:q-value}
Some lower bound on the diagonal weight is necessary, and its absence is not merely
lossy: the two values become unrelated. Let $\mX = \mA = \{0,1\}$, let $\mu$ give weight
$\frac12$ to each of $(0,1)$ and $(1,0)$ and none to the diagonal, and let $D$ accept
exactly the answer pair $(a,b) = (0,1)$ at those two question pairs and reject unequal
answers at the two diagonal pairs, as Definition~\ref{def:sync-game} requires --- which
costs nothing here, the diagonal carrying no weight. Alice answering $0$ and Bob answering
$1$, deterministically, is a tensor-product strategy of value $1$, so
$\valstar(\game) = 1$. A synchronous strategy, by contrast, is a single family
$\{M^x_a\}$ of projective measurements against a tracial state $\tau$; writing
$P = M^0_0$ and $Q = M^1_0$, its value is
\begin{equation*}
  \tfrac12 \tau\bigl( P (\Id - Q) \bigr) + \tfrac12 \tau\bigl( Q (\Id - P) \bigr)
  \;=\; \tfrac12 \bigl( \tau(P) + \tau(Q) \bigr) - \tau(PQ)
\end{equation*}
by traciality. Now $\tau(PQ) \geq \max\{0,\ \tau(P) + \tau(Q) - 1\}$: the first bound
because $\tau(PQ) = \tau(QPQ) \geq 0$, the second because
$\tau\bigl( (\Id - P)(\Id - Q) \bigr) = \tau\bigl( (\Id - Q)(\Id - P)(\Id - Q) \bigr)
\geq 0$. With $s = \tau(P) + \tau(Q)$ the value is therefore at most
$\min\{ s/2,\ 1 - s/2 \} \leq \frac12$, attained in dimension $1$ at $P = \Id$, $Q = 0$.
Hence $\synval(\game) = \frac12$ while $1 - \valstar(\game) = 0$.

This game is not symmetric, and for this instance symmetry alone repairs the conclusion:
symmetrizing $D$ so as to accept every unequal answer pair off the diagonal gives
$\synval(\game) = 1$ too, by $P = \ket{0}\bra{0}$, $Q = \ket{1}\bra{1}$ in dimension $2$.
The counterexample therefore refutes the theorem as quantified over all synchronous games,
rather than exhibiting a failure among the symmetric games the pipeline produces; what the
source's proof needs in either case is the diagonal weight itself, the synchronicity
defect being measured against the marginal and controlled only by the diagonal pairs.

A positive diagonal weight does not rescue the printed statement either: the hypothesis
has to be quantitative. For $p \in (0,1)$ let $\mu$ give weight $p/2$ to each of $(1,1)$
and $(2,2)$ and $(1-p)/2$ to each of $(1,2)$ and $(2,1)$, with $D$ accepting equal answers
on the diagonal and $(a,b) = (0,1)$ off it; every question then carries diagonal weight
exactly $p$. Answering $0$ and $1$ deterministically loses only the diagonal, so
$\valstar(\game) \geq 1 - p$, while a synchronous strategy wins the diagonal with certainty
($\sum_a \tau(M^x_a M^x_a) = 1$ for a projective family) and, by the computation above, at
most half of the remaining weight, so $\synval(\game) = \frac{1+p}{2}$. Thus
$1 - \synval(\game) = \frac{1-p}{2}$ stays above $\frac14$ for $p \leq \frac12$ while
$1 - \valstar(\game) \leq p$ tends to $0$: no constants independent of the game satisfy the
printed inequality, and the $\kappa$ of the corrected one cannot be absorbed into $C$.

Nor does the hypothesis survive repetition: if $\mu(x,x) = \kappa \sum_y \mu(x,y)$ for
every $x$, the $k$-fold product distribution satisfies the same inequality with $\kappa^k$
and no better. A bound on the synchronous value of a $k$-fold repetition therefore
does not supply a uniform quantum gap once $k$ grows with the instance size. The
pipeline's repetition theorem therefore uses direct bipartite repetition, with no
application of Theorem~\ref{thm:almost-sync} to either the input or repeated game.
The separate role of almost-synchronicity in the paper's tensor-code proof is explained
in Remark~\ref{rem:almost-sync-transitive}.

Even with the hypothesis, the transport cannot by itself reach the threshold
$\frac12$ that Definition~\ref{def:mipstar} demands. Take $\mX = \{1,2\}$, a singleton
answer alphabet, $\mu(1,1) = \mu(2,2) = \frac12$, and $D$ accepting at $(1,1)$ only: both
values are $\frac12$ and $\kappa = 1$, so the corrected inequality forces $C \geq
2^{c-1}$, whence $(2C)^{-1/c} \leq \frac12$. From $\synval(\game) \leq \frac12$ it then
concludes at best $\valstar(\game) \leq 1 - \kappa (2C)^{-1/c}$, and that bound is never
better than $\frac12$. The threshold must therefore come from
Theorem~\ref{thm:direct-repetition-q}, which is a $\valstar$ theorem, and carrying
soundness in $\valstar$ throughout avoids this transport loss.
\end{remark}

\subsection{The orthonormalization lemma}
\label{sec:rr-orthonormalization}

\effortMedium\ \qlib

\begin{theorem}[Orthonormalization]
\label{thm:orthonormalization}
\ledgernode{1.2.1.7.1, 1.2.1.7.1.1, 1.2.1.7.1.2, 1.2.1.7.1.3}
\uses{def:distance}
\lean{MIPRE.Orthonormalization.povm_orthogonalization_finDim}
\leanok
Let $\cal{M}$ be a von Neumann algebra on a finite-dimensional Hilbert space, let
$\varphi$ be a normal state on $\cal{M}$, and let $\{M_a\}_{a \in \mA}$ be a POVM in
$\cal{M}$ that is $\eps$-nearly projective:
\begin{equation*}
  \varphi\Bigl( \sum_{a \in \mA} M_a^2 \Bigr) > 1 - \eps\;.
\end{equation*}
Then there is a projective measurement $\{P_a\}_{a \in \mA}$ in $\cal{M}$ with
\begin{equation*}
  \varphi\Bigl( \sum_{a \in \mA} (M_a - P_a)^*(M_a - P_a) \Bigr) < 9\eps\;.
\end{equation*}
The same holds, on average, for a family $\{M^x_a\}$ indexed by $x \sim \mu$, by applying
it to $\varphi$ averaged over $x$.
\end{theorem}

\begin{remark}[What is proved, and in what generality]
\label{rem:orthonormalization-scope}
\uses{thm:orthonormalization}
\leanok
The statement above is the finite-dimensional case, which is the one the main theorem
needs and the one that is formalized. The vendored development proves three
unconditional cases, in increasing generality of the algebra and decreasing generality of
the state: every von Neumann algebra on a finite-dimensional space, for every normal
state (\texttt{MIPRE.Orthonormalization.povm\_orthogonalization\_finDim}, the
statement above, bridging \texttt{povm\_orthogonalization\_finDim\_vn}); $\cal{B}(H)$
for arbitrary $H$ (\texttt{povm\_orthogonalization\_fullAlgebra\_general}); and II$_1$
factors whose trace and state are both of trace-class form
(\texttt{povm\_orthogonalization\_II\textsubscript{1}Factor}), which is the case a
tracial, commuting-operator strategy lives in.

The general case --- every von Neumann algebra, every normal state --- is proved only as
an implication. Its hypothesis, \texttt{MvNStructureTheory}, bundles eight facts: seven
are classical Murray--von Neumann structure theory that Mathlib does not have (normal
functionals are of trace-class form; the type decomposition; the finite decomposition;
increasing nets of finite projections in the semifinite part; the center-valued trace;
halving in type III), and the eighth is the measurable-selection step of the source's own
Lemma~3.1 for the finite type~I part. No term of that type is constructed, so the
unconditional general statement remains open upstream, and with it the three companion
results (Theorems 1.1 and 1.4 and Corollary 1.5 of~\cite{dlS21}), which are formalized in
finite dimension and conditionally in general.
\texttt{MIPRE/Background/Orthonormalization/Axioms.lean} makes the split mechanical: it
fails to build if an unconditional result acquires an axiom, or if one of the four open
targets is closed upstream without this blueprint being updated.
\end{remark}

\paragraph{Source.} de la Salle~\cite{dlS21}, which gives a dimension-free bound with
explicit constants; earlier versions with worse parameters appear in~\cite{KPS18}
and in~\cite{JNVWY20}. The Lean proof is vendored from the
\texttt{orthogonalization} package of the \texttt{commuting-repetition} repository
(the same upstream as Theorem~\ref{thm:direct-repetition-co}) by
\texttt{scripts/vendor-repetition.py}; see that directory's \texttt{README.md} for what
the copy changes (import prefixes, nothing else).

\paragraph{Comments.} Used constantly: soundness analyses produce POVMs that are only
approximately projective, and this lemma replaces them by genuinely projective
measurements on the same space, which is what makes the ``projective strategies
throughout'' convention of Section~\ref{sec:departures} viable (avoiding Naimark
dilation). The finite-dimensional case suffices for the main theorem and is a clean,
self-contained formalization target in linear algebra.

The first of the two claims about the source made by the verification
campaign~\cite{Audit26} --- that this statement was over-hypothesized, the result needing
only a \emph{normal} state rather than a normal faithful tracial one, with constant $9$
and exponent $1$ --- is now settled in the direction the campaign predicted, and settled
by proof rather than by reading: the statement above carries neither faithfulness nor
traciality and is checked. That matters beyond bookkeeping, because the tracial
hypothesis excluded the bipartite vector-state applications that several of the
projectivization sites of Section~\ref{ch:proof-structure} need; those sites can now use
this lemma, where before they needed Naimark dilation or the projective-by-definition
convention of Section~\ref{sec:departures}. What is settled is the mathematics, not the
attribution: the campaign's second claim, that crediting the weaker predecessor
to~\cite{KPS18} and~\cite{JNVWY20} is wrong and the source credits the low individual
degree paper and Kempe--Vidick instead, is still unchecked here, \cite{dlS21} not being
vendored in this repository. The campaign also offers a consistency-to-projectivity step
giving $O(\delta)$ in place of $O(\delta^{1/4})$, which would sharpen the proof of
Theorem~\ref{thm:almost-sync}.

The formalized hypothesis and conclusion are both strict. For a POVM, completeness
identifies the defect with $1-\varphi(\sum_a M_a^2)$, but does not turn a non-strict
bound into a strict one. In finite dimension a separate compactness argument retains
the constant $9$: apply the theorem at every larger error, then pass to a convergent
subsequence of projective measurements and use continuity. This also covers zero
defect; Corollary~\ref{cor:ortho-from-consistency} spells out the limiting step needed
by its application. That corollary has no Lean proof mark here.

\subsection{Magic Square rigidity}
\label{sec:rr-magic-square}

\effortEasy\ \qlib

\begin{theorem}[Magic Square rigidity]
\label{thm:ms-rigidity}
\uses{def:q-value,def:distance}
Let $\strategy = (\ket{\psi}, A, B)$ be a strategy that succeeds in the Magic Square
game $\game^{\MS}$ with probability $1 - \eps$. Then there exist local isometries
$\phi_\alice : \mH_\alice \to (\C^2)^{\ot 2} \ot \mH_{\alice''}$, $\phi_\bob :
\mH_\bob \to (\C^2)^{\ot 2} \ot \mH_{\bob''}$ and a state $\ket{\aux} \in
\mH_{\alice''} \ot \mH_{\bob''}$ such that
\begin{equation*}
  \norm{ \phi_\alice \ot \phi_\bob \ket{\psi} - \ket{\EPR_2}^{\ot 2} \ot \ket{\aux} }
  \leq O(\sqrt{\eps})\;,
\end{equation*}
and, writing $\tilde A = \phi_\alice A \phi_\alice^\dagger$, the observables for the
questions $\Variable_1$ and $\Variable_5$ satisfy
$\tilde A^{\Variable_1}_b \approx_{\sqrt\eps} \sigma^X_b \ot \Id$ and
$\tilde A^{\Variable_5}_b \approx_{\sqrt\eps} \sigma^Z_b \ot \Id$ on
$\ket{\EPR_2}^{\ot 2} \ot \ket{\aux}$ (and symmetrically for $\tilde B$). In
particular the corresponding $\pm 1$-observables approximately anticommute:
$\tilde A^{\Variable_1} \tilde A^{\Variable_5} \approx_{\sqrt\eps}
- \tilde A^{\Variable_5} \tilde A^{\Variable_1}$.
\end{theorem}

\paragraph{Source.} Wu, Bancal, McKague and Scarani~\cite{WBMS16}; the game is the
Mermin--Peres magic square. The form above is~\cite[Section 8]{JNVWY20}.

\effortMedium

\begin{lemma}[Direct Magic Square anticommutation --- what the pipeline actually needs]
\label{lem:ms-direct-anticomm}
\ledgernode{1.2.2.4.1, 1.2.2.4.1.6, 1.2.2.4.1.7, 1.2.2.4.1.9}
\uses{def:q-value,def:magic-square}
Let $\strategy = (\ket{\psi}, A, B)$ be any finite-dimensional strategy, with arbitrary
POVMs, that fails the Magic Square game with probability $\eps$, and let $B_j$ denote the
$\pm 1$-observable (the POVM difference) of Bob's answer to variable $\Variable_j$. Then
\begin{equation*}
  \norm{ \{ B_1, B_5 \} \ket{\psi} }^2 \;\leq\; 186624\, \eps\;,
\end{equation*}
and symmetrically for Alice, where $\{X, Y\} = XY + YX$. The bound holds on the
\emph{original} observables: no isometry, no projectivity, no full Schmidt rank and no
extracted EPR pairs are required. It averages without loss --- for any family indexed by
$\omega$ with distribution $\nu$,
$\E_\nu \norm{\{B_{1,\omega}, B_{5,\omega}\} \ket{\psi_\omega}}^2
\leq 186624\, \E_\nu \eps_\omega$ --- and transfers to the point observables of the Pauli
game: if $P, Q$ have cross-party consistency errors $s_X, s_Z$ against $B_1, B_5$, then
$\norm{\{P,Q\}\ket{\psi}} \leq 2 s_X + 2 s_Z + \norm{\{B_1,B_5\}\ket{\psi}}$.
\end{lemma}

\begin{proof}
\uses{thm:orthonormalization}
Dilate each constraint POVM, after repairing invalid outcomes to a fixed parity-valid triple,
to a projective measurement on one ancilla; its outcome PVM gives commuting reflections
$C_{c,j}$ whose product over a constraint is exactly that constraint's sign. Bob's original
POVMs are untouched. Uniformity of the $36$ oriented question incidences bounds each
conditional loss by $36\eps$, and
$\norm{(C_{c,j} - B_j)\ket{\psi}}^2 \leq 2 - 2(1 - 2\ell_{c,j}) \leq 4 \ell_{c,j}$ using
$C_{c,j}^2 = \Id$ and $B_j^2 \leq \Id$, so $\gamma := \max_{c,j} \norm{(C_{c,j} -
B_j)\ket{\psi}}$ satisfies $\gamma^2 \leq 144 \eps$.

Write $a, \dots, i$ for Bob's variables $1, \dots, 9$ in row-major order. Substituting one
operator at a time costs $\gamma$ each, cross-party products appearing in reversed order, and
a constraint relation inserted before a right suffix of length $k$ costs the relation's own
$3\gamma$ plus $2k\gamma$. The six-step path
\begin{equation*}
  d\,a\,e\,b \;\to\; g\,e\,b \;\to\; g\,h \;\to\; i \;\to\; -f\,c
  \;\to\; -d\,e\,c \;\to\; -d\,e\,a\,b\;,
\end{equation*}
applying the column-one, column-two, row-three, column-three, row-two and row-one
constraints in turn, costs $(7 + 3 + 3 + 3 + 5 + 3)\gamma = 24\gamma$, and its two ends give
$\norm{ d (ae + ea) b \ket{\psi} } \leq 24 \gamma$. The sign appears exactly once, at the
column-three constraint, which is the one of product $-1$.

Put $R = ae + ea$ and $T = Rb$. Since $\norm{(\Id - B_j^2) W \ket{\psi}} \leq (2 + k)\gamma$
for a word $W$ of length $k$, the two length-three words in $T$ give
$\norm{(\Id - d^2) T \ket{\psi}} \leq 10\gamma$, and hence $\norm{T\ket{\psi}} \leq 34\gamma$.
Finally Alice's reflection for variable $2$ commutes with the Bob operator $R$, squares to
$\Id$, and approximates $b$ on $\ket{\psi}$ to within $\gamma$, which removes the trailing
factor: $\norm{R \ket{\psi}} \leq 34\gamma + 2\gamma = 36\gamma$. Squaring and substituting
$\gamma^2 \leq 144\eps$ gives $1296 \cdot 144 = 186624$. Exchanging the parties gives Alice's
bound. The averaged form is the pointwise bound averaged, with no Jensen step, and the
point-observable transfer is two triangle inequalities across the parties.
\end{proof}

\paragraph{Comments.} This lemma is why Theorem~\ref{thm:ms-rigidity} is no longer an
assumption of this blueprint (Remark~\ref{rem:admitted-nodes}). Full two-EPR self-testing is

better target than Theorem~\ref{thm:ms-rigidity}: the ingredients are a Naimark dilation, a

reflections with six relations.

\paragraph{Comments.} The direct bound is the elementary anticommutation input that
the Pauli test amplifies to many qubits. Its bipartite form already matches the

\label{sec:rr-lidt}

\begin{definition}[Canonical-line low individual degree test]
\label{def:lidt}
\uses{def:game}
\lean{MIPRE.LIDT.lidtGame}
\leanok
Let $\F$ be a finite field with $q$ elements and let $m \geq 1$ and $d$ be integers.
The \emph{$(m,q,d)$-low individual degree test} is the two-player game in which the
verifier picks one of the following three sub-tests with probability $\tfrac13$ each,
having sampled a uniform point $u \sim \F^m$.
\begin{enumerate}
\item \textbf{Axis-parallel lines test.} Sample a uniform direction $i \sim [m]$ and a
  uniform role. One player receives the line $\ell = \{u + t e_i\}$ and answers a
  univariate polynomial $f$ of degree at most $d$; the other receives $u$ and
  answers $a \in \F$. Accept if $f(u) = a$.
\item \textbf{Self-consistency test.} Both players receive $u$ and answer elements of
  $\F$. Accept if the answers are equal.
\item \textbf{Diagonal lines test.} Sample a uniform $j \sim [m]$, a uniform direction
  $v \in \F^m$ whose coordinates after the $j$-th vanish, and a uniform role. One
  player receives the line $\ell = \{u + t v\}$ and answers a univariate polynomial
  of degree at most $md$; the other receives $u$ and answers $a \in \F$. Accept if
  $f(u) = a$.
\end{enumerate}
\end{definition}

\paragraph{Comments.} This is Figure~1 of~\cite{JNVWY21qld}. The direction $v$ of the
diagonal lines test may be zero, in which case the ``line'' is the single point $u$;
this degenerate case is part of the test as stated. In the formalization, line questions
are the lines themselves, presented canonically (the direction is rescaled so that its
first nonzero coordinate is $1$, and the base point is the point of the line whose
corresponding coordinate vanishes), so that the same line asked by two sub-tests is the
same question; a line answer is a polynomial in the parameter of that canonical
parametrization.

\begin{theorem}[Quantum soundness of the canonical-line low individual degree test]
\label{thm:lidt-soundness}
\uses{def:lidt,def:tensor-strategy,def:inconsistency}
\lean{MIPRE.LIDT.lowIndividualDegree_soundness}
\leanok
Let
\begin{equation*}
  \delta_{lidt}(\eps, m, d, q, k)
  = 10^5 \, k^2 m^4 \bigl( \eps^{1/40000} + (d/q)^{1/40000}
      + e^{-k/(2560000 m^2)} \bigr).
\end{equation*}
Let $\strategy = (\ket{\psi}, A, B)$ be a tensor-product strategy for the
$(m,q,d)$-low individual degree test that succeeds with probability at least $1 - \eps$,
and let $k$ be an integer with $k \geq 400md$ and $k > 0$. Then there exist projective
measurements $G^\alice = \{G^\alice_g\}$ on Alice's space and $G^\bob = \{G^\bob_g\}$ on
Bob's space, both with outcomes the polynomials $g$ in $m$ variables of individual degree
at most $d$, such that, writing $\delta = \delta_{lidt}(\eps, m, d, q, k)$ and letting
$u \sim \F^m$ be uniform:
\begin{enumerate}
\item Alice's point measurement $A^u$ and the evaluation $G^\bob_{[g(u) = \cdot]}$
  are $\delta$-consistent;
\item the evaluation $G^\alice_{[g(u) = \cdot]}$ and Bob's point measurement $B^u$
  are $\delta$-consistent;
\item $G^\alice$ and $G^\bob$ are $\delta$-consistent.
\end{enumerate}
\end{theorem}
\begin{proof}
\leanok
This is Theorem~1.3 of~\cite{JNVWY21qld} (\texttt{thm:main-formal} there), whose proof
is the MIPStarRE formalization vendored into this repository. The Lean proof translates
a strategy for our game into MIPStarRE's strategy container (the state as a density
matrix; the line measurements as coarse-grainings of the measurements at the canonical
line questions along the change of parametrization, which makes them covariant under
rebasing), shows that MIPStarRE's failure surrogate is at most the rejection probability
$1 - \mathrm{val}$ (the consistency defect of each branch is at most the rejection
probability of that branch, since acceptance forces the coarse-grained answers to
agree), applies their main theorem, and reads the resulting measurements back into our
vocabulary.
\end{proof}

\paragraph{Source.} Ji, Natarajan, Vidick, Wright and Yuen~\cite{JNVWY21qld},
Theorem~1.3 (the formal version for general, not necessarily symmetric, strategies).
The proof is a formalization by Sirui Lu and collaborators
(\url{https://github.com/LionSR/MIPStarRE}), vendored into this repository with the
authors' permission at \texttt{MIPRE/Background/LIDT/MIPStarRE/}; the statement above is
this project's own, and is connected to theirs by an explicit bridge.

\paragraph{Comments.} Two hypotheses correct the printed statement, as established by
that formalization: the sampling parameter satisfies $k \geq 400md$ where the paper
prints $k \geq md$, and $k$ must be positive, since the printed error vanishes at
$k = 0$. The parameter $k$ remains free: the error carries both a factor $k^2$ and a
term $e^{-k/(2560000m^2)}$. A chosen-$k$ corollary in the shape of $\delta_{qld}$ below

\subsubsection{The paper's seeded CL test and its tensor-code route}

The following is the paper's route to a different game interface. Its tensor-code
reduction and reviewed intermediate statements are not the proof of
Theorem~\ref{thm:lidt-soundness}, and have no Lean proof marks here. The seeded game
and its theorem are in the pinned paper snapshot~\cite{AuditSync26},
\texttt{ldt.tex}, Definition of the game and Theorem \texttt{lem:ld-soundness}.

\begin{definition}[Seeded CL low individual degree test]
\label{def:lidt-cl}
\uses{def:game}
For admissible $q$, positive integers $m,d,\mathrm{ldc}$ and $m\mid q$, the
\emph{seeded CL test} is the paper's typed low-degree game with parameters
$(q,m,d,\mathrm{ldc})$. Each ordered pair of types from
$\{\mathtt{Point},\mathtt{ALine},\mathtt{DLine}\}$ has weight $1/9$, with question
contents sampled from the corresponding CL distribution. Point questions ask for
an element of $\F_q^{\mathrm{ldc}}$. Axis-line questions retain a seed-labelled
description $(u_0,s)$ and ask for $\mathrm{ldc}$ degree-$d$ polynomials; diagonal-line
questions retain $(u_0,s,v)$ and ask for $\mathrm{ldc}$ degree-$md$ polynomials.
The seed selects the direction class, with leading coordinates of the diagonal
direction set to zero. The verifier checks the answer format, equality on identical
same-type questions, and evaluation agreement for either line type against a point;
the other type pairs impose no further check. In particular, different seed
descriptions of the same geometric line remain different questions.
\end{definition}

\begin{theorem}[Quantum soundness of the seeded CL test]
\label{thm:lidt-cl-soundness}
\ledgernode{1.2.1}
\uses{def:lidt-cl,lem:lidt-derandomize,lem:lidt-ldc,def:inconsistency}
There are universal constants $a\geq 1$ and $0<b<1$ such that every projective
strategy passing the seeded CL test with probability at least $1-\eps$, for
$\eps>0$, admits projective measurements $G^\alice,G^\bob$ on the respective
players' spaces with outcomes tuples $(g_1,\ldots,g_{\mathrm{ldc}})$ of polynomials
of individual degree at most $d$. Their evaluations are consistent with the other
player's point measurement on average over uniform $u\in\F_q^m$, and the two
full tuple measurements are consistent with each other, all with error at most
\[
  \delta_{\mathrm{CL}}(\eps,q,m,d,\mathrm{ldc})
  = a(dm\,\mathrm{ldc})^a\bigl(\eps^b+q^{-b}+2^{-bmd}\bigr).
\]
The full tuple consistency has no question average. This paper theorem remains a
formalization target, including for $\mathrm{ldc}=1$.
\end{theorem}

\begin{remark}[Planned CL-to-canonical adapter and parameter corollary]
\label{rem:lidt-cl-adapter}
\uses{def:lidt-cl,def:lidt,thm:lidt-soundness,thm:lidt-cl-soundness}
Reusing Theorem~\ref{thm:lidt-soundness} to prove the single-codeword case of
Theorem~\ref{thm:lidt-cl-soundness} requires an explicit strategy adapter. It must
control the question distributions and subtest weights, reverse coordinate order,
remove the seed labels with a proved choice or averaging argument, and transport
line answers under canonicalization and rebasing. It must also preserve the degree
bounds and projectivity, or account for their repair, and recover the two original
point-consistency conclusions and full-codeword consistency. A seeded strategy need
not assign the same measurement to different descriptions of the same line.

A separate quantitative obligation is to choose an integer $k\geq400md$, $k>0$,
and bound the canonical theorem's error, together with every adapter loss, by the
single-codeword $\delta_{\mathrm{CL}}$ above. The simultaneous extension and the
padded Pauli use must then propagate the changed dimension and any ancillas. Neither
the adapter nor this parameter corollary is supplied by the existing Lean theorem.
Formalizing the paper's tensor-code route below is an alternative way to establish
the seeded CL theorem; a proof by either route would not certify the intermediate
claims of the other.
\end{remark}

\begin{lemma}[Setup, and the two tests are not the same test]
\label{lem:lidt-reduction-setup}

\begin{lemma}[Self-improvement and pasting]

\begin{corollary}[Rigidity transfer to symmetric strategies, with the marginal condition]
\label{cor:sync-transfer}
\ledgernode{1.2.1.7.2.3}
\uses{thm:almost-sync,def:sync-game,def:distance}
Let $\game$ be a symmetric game with common question marginal $\nu$, let $\mY$ and $\mB$ be
finite non-empty auxiliary sets, let $p$ be a distribution on $\mX \times \mY$, and let
$R \geq 1$ satisfy
\begin{equation*}
  p_{\mX}(x) \;\leq\; R\, \nu(x) \qquad \text{for all } x \in \mX\;.
\end{equation*}
For each $(x,y)$ let the sets $g_{xy}(a) \subseteq \mB$, $a \in \mA$, be pairwise disjoint
(not necessarily exhaustive), and for a POVM $N^y$ write
$N^y_{g_{xy}(a)} = \sum_{b \in g_{xy}(a)} N^y_b$. Let $\kappa : [0,1] \to \R_{\geq 0}$ be
convex and non-decreasing, and suppose that every symmetric projective maximally
entangled (PME) strategy $T = (\ket{\Phi}, Q)$ for $\game$ (so $\ket{\Phi}$ is
maximally entangled), of value $\omega(T)$, admits complete auxiliary POVMs $N^y$ with
\begin{equation*}
  F_p(T, N) \;:=\; \E_{(x,y) \sim p} \sum_{a \in \mA}
    \bra{\Phi} Q^x_a \ot N^y_{g_{xy}(a)} \ket{\Phi} \;\geq\; \kappa\bigl( \omega(T) \bigr)\;.
\end{equation*}
Then there are universal constants $C_0, c_0 > 0$ --- supplied by the squared-distance
decomposition behind Theorem~\ref{thm:almost-sync} --- such that, writing
$e(\delta) = C_0 \delta^{c_0}$ and $s(\delta) = 2\sqrt{2\delta} + 8\sqrt{e(\delta)}$, every
symmetric projective strategy $\strategy = (\ket{\psi}, A)$ of value $\omega$ and
synchronicity defect
$\delta = 1 - \E_{x \sim \nu} \sum_a \bra{\psi} A^x_a \ot (A^x_a)^{\mathsf{T}} \ket{\psi}$
admits complete auxiliary POVMs $M^y$ on its own Bob space with
\begin{equation*}
  F_p(\strategy, M) \;\geq\; \max\Bigl\{ 0,\;
    \kappa\bigl( \max\{0, \omega - s(\delta)\} \bigr) - \sqrt{R\, e(\delta)} \Bigr\}\;.
\end{equation*}
Two parts of the hypothesis are load-bearing and easy to lose. \emph{Symmetric} means the
Schmidt form $\ket{\psi} = \sum_j \sqrt{r_j} \ket{j}\ket{j}$ with $r_j \geq 0$ and Bob's game
measurements the transposes of Alice's; swap symmetry alone is not a sufficient input. And the
conclusion delivers a POVM: it asserts no projectivity, so a consumer needing projective
operators must project separately, at its own cost.
\end{corollary}

\begin{remark}[The printed transfer corollary is false without the marginal condition]
\label{rem:sync-transfer-marginal}
\uses{cor:sync-transfer,thm:almost-sync}
The marginal condition $p_{\mX} \leq R\nu$ in Corollary~\ref{cor:sync-transfer} is not
book-keeping. Corollary 4.1 of~\cite{Vid22} is stated for an arbitrary auxiliary distribution
$p$, and in that unrestricted form it is false; the source applies an estimate averaged over
$\nu$ to $p_{\mX}$ without justification. The counterexample is two questions wide. Let
$\game$ ask only question $0$ and check equality, let $p$ be supported on question $1$ alone,
let $g(a) = \{a\}$, and let $\kappa = 1$. Every symmetric PME strategy meets the
auxiliary hypothesis, using the transpose of its question-$1$ measurement: on a
maximally entangled state, agreement is the normalized trace of
$\sum_a (Q^1_a)^2=\Id$. But the
symmetric projective strategy $\ket{00}$ measuring $\sigma^Z$ at question $0$ and $\sigma^X$
at question $1$ has game value $1$ and synchronicity defect $0$, while every Bob POVM achieves
auxiliary agreement exactly $\tfrac12$ --- so the conclusion would force
$\tfrac12 \geq \kappa(1) = 1$. Here $p_{\mX}$ is supported where $\nu$ vanishes, so no finite
$R$ exists and Corollary~\ref{cor:sync-transfer} does not apply.

What rescues the use made of it here is that the tensor code test's own sampling is
comfortable: $p$ is uniform on point questions while $\nu(u) = \tfrac49 n^{-m}$, giving
$R = \tfrac94$, so Lemma~\ref{lem:tensor-codes-bipartite} pays only the universal
synchronization losses. This is worth recording because a formalization must carry the
condition explicitly --- it is exactly the hypothesis a paraphrase drops --- and because it is
the one defect the 2026-09-14 adversarial round of the companion campaign turned up (a major
challenge against ledger node $1.2.1.7.2.3$, resolved by amending the node to the statement
given above rather than by repairing the source).
\end{remark}

\begin{lemma}[From the synchronous to the two-prover form]
\label{lem:tensor-codes-bipartite}
\ledgernode{1.2.1.7.8}
\uses{thm:tensor-codes,cor:sync-transfer,rem:sync-transfer-marginal,thm:orthonormalization}
With the same code hypotheses and $t=n-\Delta+1$, let a finite-dimensional two-prover
projective strategy pass the two-prover tensor code test with probability at least
$1-\eps'$, and let $k\geq12mt$. There are projective measurements
$G^\alice,G^\bob$, with outcomes in $C^{\ot m}$, on the players' respective original
spaces. With error
$\eta'=\poly(m,t,k)\cdot\poly(\eps',n^{-1},e^{-\Omega(k/m^2)})$, Alice's point
measurement is consistent with $G^\bob$ evaluated at a uniform point; the restriction
of $G^\alice$ to a uniform axis-parallel line is consistent with Bob's line
measurement; and $G^\alice,G^\bob$ are consistent as full codeword measurements.
The last conclusion has no question average. The reverse point consistency follows
from the first and third conclusions and the test's point self-consistency, with a
change of universal constants.
\end{lemma}

\begin{proof}
For symmetric PME strategies, the normalized trace realizes their correlations, so
Theorem~\ref{thm:tensor-codes} supplies the PME premise of
Corollary~\ref{cor:sync-transfer}. For a general strategy, symmetrize and apply the
corrected transfer with a concavified error function, at $R = \tfrac94$ as
Remark~\ref{rem:sync-transfer-marginal} records. Symmetrization is where care is needed: the
swap construction preserves the passing probability, but what the corollary consumes is the
Schmidt-basis form with Bob's operators the transposes of Alice's, which swap symmetry alone
does not provide --- the approximate Takagi bridge of Remark~\ref{rem:lidt-formalized} is what
supplies it. The auxiliary measurements are compressed back to the original spaces
and projectivized there, using Theorem~\ref{thm:orthonormalization}; projectivity is
not a conclusion of the transfer corollary itself. The consistency estimates then
give the point, line and full-codeword conclusions. These are steps of the paper's
reviewed proof, not consequences already checked by the canonical-line Lean theorem.
\end{proof}

\begin{lemma}[Several codewords: the simultaneity reduction]

$1.2.1.7.2.4$ records that the interface between the product-basis transposes
of~\cite{JNVWY20} and the Schmidt-diagonal symmetric strategies of~\cite{Vid22} is

\end{remark}

\begin{theorem}[Pauli basis test / quantum low individual degree test]
\label{thm:qld}
\ledgernode{1.2.2, 1.2.2.17}
\uses{def:q-value,def:distance,lem:ms-direct-anticomm,thm:lidt-cl-soundness}
There exists a function
\begin{equation*}
  \delta_{qld}(\eps, m, d, q) = a (md)^a \bigl( \eps^b + q^{-b} + 2^{-bmd} \bigr)
\end{equation*}
for universal constants $a \geq 1$ and $0 < b < 1$ such that the following holds. For
all admissible parameter tuples $\qldparams = (q, m, d)$ and all strategies
$\strategy = (\ket{\psi}, A, B)$ for the game $\game^{\pauli}_{\qldparams}$ that
succeed with probability at least $1 - \eps$, there exist local isometries
$\phi_\alice, \phi_\bob$ mapping into $\mH' \ot (\C^q)^{\ot M}$ with $M = 2^m$, and a
state $\ket{\aux}$, such that
$\norm{\phi_\alice \ot \phi_\bob \ket{\psi} - \ket{\aux} \ot \ket{\EPR_q}^{\ot M}}
\leq \delta_{qld}$, and for $W \in \{X, Z\}$ the measurements for the questions
$(\pauli, W)$ are $\delta_{qld}$-close to the Pauli basis measurements $\tau^W_u$, $u
\in \F_q^M$, acting on $\ket{\EPR_q}^{\ot M}$.
\end{theorem}

\paragraph{Source.} The pinned paper~\cite{AuditSync26}, \texttt{qld-appendix.tex}
and its five proof files, states this Pauli theorem and its reduction to the seeded
CL test. The reduction follows the architecture of~\cite[Section 7]{JNVWY20};
the canonical-line soundness theorem of~\cite{JNVWY21qld} is related background.
Lemma~\ref{lem:ms-direct-anticomm} supplies the base anticommutation directly.

\paragraph{Comments.} This is a substantial remaining formalization target and the
workhorse of question reduction: introspection certifies the players' sampling of
their own questions against Pauli basis measurements. The reduction uses direct
Magic Square anticommutation, linearity and orthonormalization, together with the
seeded CL theorem. The available canonical-line theorem can replace the paper's
classical route only after the obligations in Remark~\ref{rem:lidt-cl-adapter} are met.

The theorem's bipartite statement matches the pipeline's $\valstar$ soundness
interface, so its uses require no additional synchronous-value transport. The verification
campaign~\cite{Audit26} further reports three corrections in the tensor-code material the
proof passes through --- $\kappa$-convexity requires $c_1 \leq 1$, the synchronicity
defect obeys $\delta_{\mathrm{sync}} \leq 4\eps$ rather than $3\eps$, and one corner block
pairs with a transpose --- none checked here, none affecting the statement above, all
relevant to whoever writes its proof.


\subsubsection{The shape of the soundness proof}

\begin{remark}[How the soundness proof of Theorem~\ref{thm:qld} is organized]
\label{rem:qld-architecture}
\uses{thm:qld,thm:lidt-cl-soundness,rem:lidt-cl-adapter,thm:orthonormalization,lem:ms-direct-anticomm}
The soundness half of Theorem~\ref{thm:qld} is developed in the five-file appendix of
the pinned paper~\cite{AuditSync26}. Its
architecture is worth stating before its pieces, because the pieces are unreadable without it.
There are five stages.
\begin{enumerate}
\item \textbf{Preliminaries}: three utility lemmas about closeness, one of which
  (Corollary~\ref{cor:ortho-from-consistency}) is used on almost every page.
\item \textbf{Win implications}: unpack ``succeeds with probability $1 - \eps$'' into one
  consistency or commutation statement per subtest (Lemma~\ref{lem:qld-win}).
\item \textbf{Expansion}: adjoin EPR ancillas and replace the strategy's measurements by ones
  on the enlarged space that approximately commute across the two bases
  (Lemmas~\ref{lem:qld-expanded-points} and~\ref{lem:qld-expanded-lines}).
\item \textbf{Combining}: build a \emph{single} measurement returning both the $X$- and the
  $Z$-polynomial, by padding the ambient space and applying seeded CL soundness to
  the padded strategy (Lemma~\ref{lem:qld-simultaneous}).
\item \textbf{Separation and the swap isometry}: extract exact Pauli operators from that
  single measurement and rotate them onto honest EPR pairs
  (Lemmas~\ref{lem:qld-exact-paulis} and~\ref{lem:qld-swap}).
\end{enumerate}
Stage 4 consumes Theorem~\ref{thm:lidt-cl-soundness} at $(q,4m,d,1)$, after
constructing a strategy for the seeded CL game and projectivizing its line POVMs.
Theorem~\ref{thm:lidt-soundness} provides reusable checked analysis, but does not yet
justify this application: the game adapter and error corollary in
Remark~\ref{rem:lidt-cl-adapter} remain open. Stage 5 recovers the Pauli group
relations exactly, which permits the standard swap isometry; the expanded
measurements in stage 3 satisfy only approximate commutation.
\end{remark}

\begin{lemma}[Averaging preserves closeness]
\label{lem:qld-averaging}
\ledgernode{1.2.2.1}
\uses{def:distance}
Let $A^x \approx_\delta B^x$ on average over $x \sim \mu$, and let $\alpha_x$ be coefficients
with $\abs{\alpha_x} \leq 1$. Then $\E_x \alpha_x A^x \approx_\delta \E_x \alpha_x B^x$.
\end{lemma}

\begin{proof}
Triangle inequality and Jensen. Note the hypothesis is $\abs{\alpha_x} \leq 1$, not
$\abs{\alpha_x} = 1$: the lemma is used with genuinely contractive coefficients.
\end{proof}

\effortEasy

\begin{lemma}[From POVM elements to generalized observables]
\label{lem:qld-povm-to-obs}
\ledgernode{1.2.2.2}
\uses{def:distance,lem:qld-averaging}
Let $\{A^x_a\}$ and $\{B^x_a\}$ be POVMs with a common outcome set $\mA$, and let $\alpha_a$
be coefficients of unit modulus. If $A^x_a \approx_\delta B^x_a$ then the generalized
observables $A^x = \sum_a \alpha_a A^x_a$ and $B^x = \sum_a \alpha_a B^x_a$ satisfy
$A^x \approx_{\abs{\mA} \delta} B^x$.
\end{lemma}

\effortMedium

\begin{corollary}[Orthonormalization from consistency]
\label{cor:ortho-from-consistency}
\ledgernode{1.2.2.3, 1.2.2.3.1}
\uses{thm:orthonormalization,def:inconsistency,def:distance}
There is $\eta(\delta) = O(\delta)$ such that the following holds. Let $\{Q_a\}$ be a POVM on
$\mH_\alice$ and $\{R_a\}$ a POVM on $\mH_\bob$ that are $\delta$-consistent on
$\ket{\psi}$. Then there is a \emph{projective} measurement $\{P_a\}$ on $\mH_\alice$ with
$P_a \ot \Id \approx_{\eta(\delta)} Q_a \ot \Id$ on $\ket{\psi}$.
\end{corollary}

\begin{proof}
\uses{thm:orthonormalization}
This is not Theorem~\ref{thm:orthonormalization} restated: that theorem requires strict
near-projectivity in a normal state, while what the appendix has is
bipartite consistency. The bridge is a Cauchy--Schwarz step. Write
$\gamma = \sum_{a \neq b} \bra{\psi} Q_a \ot R_b \ket{\psi}$, so that
$\sum_a \bra{\psi} Q_a \ot R_a \ket{\psi} = 1 - \gamma$. Since
$\sum_a R_a^2 \leq \sum_a R_a = \Id$, Cauchy--Schwarz over $a$ gives
$\sum_a \bra{\psi} Q_a^2 \ot \Id \ket{\psi} \geq (1 - \gamma)^2 \geq 1 - 2\gamma$, which is
non-strict near-projectivity at $2\gamma$. For every $\eps>2\gamma$, apply
Theorem~\ref{thm:orthonormalization} to the algebra
$\cal{L}(\mH_\alice)\ot\Id$ with the normal vector state of $\ket{\psi}$. It gives a
projective family of the form $P_a^{\eps}\ot\Id$ with squared-distance sum less than
$9\eps$. In finite dimension, projective measurements with the fixed finite outcome
set form a compact set. Letting $\eps$ decrease to $2\gamma$ along a sequence,
extract a convergent subsequence and use continuity to obtain
$\sum_a\norm{((Q_a-P_a)\ot\Id)\ket{\psi}}^2\leq18\gamma=O(\delta)$.
This argument includes $\gamma=0$.
\end{proof}

\begin{lemma}[Win implications]
\label{lem:qld-win}
\ledgernode{1.2.2.4, 1.2.2.4.2}
\uses{lem:ms-direct-anticomm,def:inconsistency,def:distance}

\end{proof}

\begin{lemma}[Expanded point measurements]
\label{lem:qld-expanded-points}
\ledgernode{1.2.2.5}
\uses{lem:qld-win,cor:ortho-from-consistency,def:distance}

\begin{theorem}[Exact linearity from approximate linearity]
\label{thm:linearity}
\ledgernode{1.2.2.7, 1.2.2.7.1, 1.2.2.7.2, 1.2.2.7.3}
\uses{def:distance,thm:gowers-hatami}
Let $\{O^u\}_{u \in \F_2^t}$ be observables satisfying approximate linearity on average,
$O^u O^{u'} \approx_\delta O^{u + u'}$. Then on an extended space there are \emph{exactly}
linear observables $\{L^u\}$ --- a representation of $\F_2^t$, with $L^0 = \Id$ --- such that
$L^u \approx_\delta O^u$ on average over $u$. The error is linear in $\delta$, with no loss in
the exponent.
\end{theorem}

\begin{proof}
Imported from Natarajan and Vidick~\cite{NV17}, whose proof is the Fourier argument: the
operators $\widehat{A}^u = \E_a (-1)^{a \cdot u} A(a)$ have squares forming a POVM by
Parseval, Naimark dilates that POVM to a projective family on an extended space, and the
exactly linear observables are recovered from it by inverting the transform. The source is
vendored and was audited line by line, including its state-dependent distance toolkit; the
only facts it needs beyond that are Parseval and Naimark.
\end{proof}

\end{lemma}

\subsection{Cook--Levin and succinct satisfiability}

\effortHard

\begin{remark}[What stage 1.2 leaves open, and where its weight is]
\label{rem:qld-admitted}
\uses{thm:qld,lem:ms-direct-anticomm,thm:lidt-cl-soundness,def:distance}
Stage $1.2$ of the ledger is by far its heaviest: after the refinement rounds of September
2026 it holds $149$ of the $316$ live nodes, and $169$ of the campaign's $309$ challenges
were raised against its original $62$, against $46$ for answer reduction and $19$ for
repetition. Three facts about it bear on how this section should be formalized.

First, the Magic Square input is no longer an assumption. It was: node $1.2.2.4.1$ was an
import of~\cite[Theorem 6.9]{CS17}, neither vendored nor reproved, carrying two adaptations
made in the paper --- a weakening of trace-norm $O(\eps)$ closeness to Euclidean
$O(\sqrt{\eps})$, and a local basis change between two anticommuting pairs. What made it
tractable was how narrowly it was consumed, through a single anticommutation consequence,
and that observation is what eventually discharged it: the consequence is now proved
directly from the game, as Lemma~\ref{lem:ms-direct-anticomm}. The completeness leg never
used it at all, resting instead on the in-file argument that anticommuting observables give
a perfect Magic Square strategy (Lemma~\ref{lem:mermin-peres} below is that strategy, and it
is already in Lean).

Second, the soundness proof of Theorem~\ref{thm:qld} is not in the main text at all: it is
deferred to a five-file appendix, which is where most of stage $1.2$'s challenge weight
sits, including one of the campaign's three \emph{critical} findings, at node $1.2.2.15$:
the construction of exact Pauli operators applied Schwartz--Zippel in the wrong direction
--- distinct polynomials of degree $md$ \emph{agree} on at most an $md/q$ fraction, so for a
non-multilinear $g$ the disagreement indicator has expectation near $1$ --- which made the
step equivalent to an unproven assertion about how much mass the simultaneous measurement
places on non-multilinear outcomes. It is resolved, by bounding that mass first. A
formalization should treat the appendix as the real content and the main-text statement as
its interface.

The other two critical findings are worth naming here, because neither reaches this
blueprint and the reason is the same in both cases. At node $1.4$, the anchored repetition
theorem was applied to a game that is not of the anchored product form at all: detyping
destroys the product structure, and restricting a repeated strategy to the event that every
coordinate is a real question costs $16^k$. At node $1.5$, the compression statement had
omitted the doubly exponential term in its entanglement lower bound, which the recursion
needs. Both are resolved upstream; both live in the entanglement-form apparatus that
Theorem~\ref{thm:direct-repetition-q} and the $\valstar$ chain replace
(Remark~\ref{rem:admitted-nodes}). All three criticals are resolved, so this is not a
soundness worry --- it is an indication of where the delicate machinery is, and two thirds
of it is machinery this route does not carry.

Third, the introspection statement that consumes all of this (node $1.2$,
Theorem~\ref{thm:introspection}) leaves a soundness gap of $1/\poly(n)$ --- \emph{not} a
constant, which is the stronger reading and the one the ledger is explicit about. That is
exactly why parallel repetition has to follow it: repetition is what turns a gap shrinking
with $n$ back into the constant $\frac12$ the recursion needs.

\begin{definition}[Succinctly described 3SAT and decoupled 5SAT]
\label{def:succinct-3sat}
\lean{MIPRE.SAT.Circuit.formula3, MIPRE.SAT.Circuit.formula5, MIPRE.SAT.Circuit.DescribesDecider}
\leanok
A Boolean circuit $C$ with three inputs of length $m$ and three single-bit inputs
\emph{succinctly describes} the 3SAT formula on the $2^m$ variables $x_0, \ldots,
x_{2^m - 1}$ whose clauses are the $x_{i_1}^{o_1} \vee x_{i_2}^{o_2} \vee x_{i_3}^{o_3}$
with $C(i_1, i_2, i_3, o_1, o_2, o_3) = 1$, integers presented to the circuit in binary
(\cite[Definition of succinct descriptions of 3SAT formulas, Section 10]{JNVWY20}). A
\emph{decoupled 5SAT} formula has five blocks of variables and clauses with one literal from
each block, evaluated on five separate assignments; a circuit with two inputs of length
$\ell$, three of length $r$ and five single-bit inputs succinctly describes one in the same
way. Such a circuit \emph{describes the decider $\decider$ on $(n, x, y)$ at time $T$} if,
for all $a, b \in \{0,1\}^{2^\ell}$, some three assignments complete $(a, b)$ to a
satisfying tuple exactly when $a$ and $b$ are the two-bit tape encodings ($0 \mapsto 00$,
$1 \mapsto 01$, blank $\mapsto 10$) of strings of length at most $T$ on which $\decider$
accepts $(n, x, y, \cdot, \cdot)$ within $T$ steps. Circuits have fan-in and fan-out at most
two and a terminal output gate, and their size is the number of gates.
\end{definition}

\begin{theorem}[Explicit succinct Cook--Levin]
\label{thm:succinct-sat}
\ledgernode{1.3.1}
\uses{def:succinct-3sat,def:decider,lem:correct-tableau}
\lean{MIPRE.SAT.SuccinctCookLevin}
\leanok
There is a polynomial-time algorithm that, given a decider $\decider$, an index $n$, a
time bound $T$ (in binary), a question length $Q$, a bound $\sigma$ and strings $x, y$, with
$\max\{Q, 2\lceil \log n \rceil\} \leq T$, $\abs{\decider} \leq \sigma$ and $\abs{x},
\abs{y} \leq Q$, outputs a circuit $C$ on $3m + 3$ inputs which succinctly describes
(Definition~\ref{def:succinct-3sat}) a 3SAT formula $\varphi$ on $2^m$ variables such that
\begin{enumerate}
\item for all $a, b \in \{0,1\}^{2T}$, the formula $\varphi$ has a satisfying assignment
  whose first $4T$ variables are $(a, b)$ if and only if $a$ and $b$ are the two-bit tape
  encodings of strings $a', b'$ of length at most $T$ such that $\decider$ accepts $(n, x,
  y, a', b')$ within $T$ steps;
\item $m = O(\log T + \log \sigma)$ and $2^m \geq 4T$;
\item $C$ has at most $s(n, T, Q, \sigma) = \poly(\log n, \log T, Q, \sigma)$ gates;
\item the algorithm runs in time $\poly(\log n, \log T, Q, \sigma)$;
\item $m$ and $s$ are computable from the parameters in time polynomial in their bit
  lengths.
\end{enumerate}
\end{theorem}

\paragraph{Lean statement.} The theorem is the structure
\texttt{MIPRE.SAT.SuccinctCookLevin}: the algorithm a polynomial-time function of the
ambient model (\texttt{PolyTimeFun}) from $((\decider, n, T, Q, \sigma), x, y)$ to circuits
as data, ``within $T$ steps'' read as cost at most $T$ in the ambient model, the reading of
every time bound of the pipeline, and the five items as fields. An earlier statement asked
for a succinct description in the sense of Definition~\ref{def:succinct}, which is a
program answering bit queries and not the circuit the proof produces and the PCP consumes,
and it carried none of the five items, all of which Lemma~\ref{lem:decoupled-5sat} and the
PCP parameters depend on. The statement is validated by its consumer
(\texttt{MIPRE.SAT.DecoupledDescriber}, Lemma~\ref{lem:decoupled-5sat}) before its proof;
the route to the proof is \texttt{planning/succinct-cook-levin.md}.

\paragraph{Formalization status.} The mathematics of the describer is formalized
(\texttt{MIPRE/TM/CookLevin/}): the check circuit of the interpreter machine, from the truth
table of its local check predicate (\texttt{chk\_isCheckCircuit}); the describer circuit
\texttt{descCirc}, the flattening of a formula on its $3m + 3$ inputs, with its input count
and well-formedness; what it describes, \texttt{mem\_formula3\_iff} --- a clause is accepted
exactly when its three indices decode to the literals of a clause of the tableau of
Lemma~\ref{lem:correct-tableau} on the interpreter machine (together with two clauses
bounding the answers by $T$); and item 1, \texttt{extendsAnswers\_iff}.

The describer is also formalized as a polynomial-time function of the ambient model
(\texttt{descCircP}, item 4), on four conventions that keep the program layer close to the
mathematics: a formula's encoding is the encoding of its post-order list, so each builder is
an \texttt{append}, a \texttt{map} or a \texttt{foldl} on lists; a program is read as a
\emph{reader} of the shared input, so it transcribes its definition line for line rather
than point-free; a field record is a record of readers whose evaluation is a structure
literal of their own values, which makes every family formula match its program
definitionally; and widths and offsets are unary while a field constant is a resize of a bit
string, so no arithmetic on numbers is needed. Item 3 then needs no separate argument: a
program's output is no larger than its running time and a circuit's gate count is at most
the size of its encoding, so the gate bound \emph{is} the time bound of the program that
writes the circuit (\texttt{descCirc\_size\_le}). What is left is $e$ as a program of the
parameters, the substitution of the validity hypotheses into that bound (item 5), and the
assembly of the structure.

\paragraph{Source.} Cook~\cite{Coo71} and Levin for the tableau reduction; Papadimitriou
and Yannakakis~\cite{PY86} for succinct representations and their complexity;
the explicit form needed here is~\cite[Section 10]{JNVWY20}.

\paragraph{Comments.} This is the classical core of answer reduction: the players prove
satisfiability of $\varphi$ (arithmetized over a low-degree extension) instead of
sending $(a,b)$. The statement is long but entirely combinatorial. Cook--Levin has been
formalized in other proof assistants; a Lean formalization with the explicit-uniformity
guarantees above is a substantial but well-specified sub-project.

A caution on the construction, from the verification campaign~\cite{Audit26} and not
checked here: in the Tseitin encoding of~\cite{NW19}, which the explicit form
of~\cite[Section 10]{JNVWY20} follows, the formula is reported to omit the output-wire
conjunct, so that it is satisfiable for every input and the central equivalence of the
construction fails as written. The statement above is end-to-end and so does not inherit
the defect textually, but a formalization of it must supply what the repair needs: the
output-wire conjunct, the out-degree-zero hypothesis that yields individual degree $6$,
and the exact variable count on which the later PCP parameters depend --- none of which
the statement currently fixes.

\begin{lemma}[Correctness of the Cook--Levin tableau]
\label{lem:correct-tableau}
\uses{def:multiinput-tm}
\lean{MIPRE.TM.CookLevin.tableau_sat_iff}
\leanok
Let $M$ be a multi-input Turing machine (Definition~\ref{def:multiinput-tm}) with $i$ input
tapes and $w$ work tapes over a finite alphabet, $\mathsf{acc}$, $s_0$, $s_1$ symbols and
$S$ a step bound, and fix the strings on some of the input tapes. The \emph{tableau formula}
of $S$ steps of $M$ is the 3SAT formula on the variables ``cell $p$ of tape $d$ at time $t$
holds $v$'' (one for each of $2S + 7$ cells and each value: a symbol, the blank or the
boundary marker), ``the head of tape $d$ is on cell $p$ at time $t$'', ``the state at time
$t$ is $q$'', the emission flags of every step, and, for every step and every choice of a
five-cell window on each tape, the gate variables of a circuit deciding local consistency
of the window; its clauses are the start row, the boundary rows, the free-tape row (a
string over $\{s_0, s_1\}$ followed by blanks), the emission bookkeeping, the acceptance
row and the circuit-to-3SAT clauses of the local check at every window. The formula is
satisfiable if and only if $M$, started on the fixed strings and on some strings over
$\{s_0, s_1\}$ on the free input tapes, halts within $S$ steps having output exactly
$\mathsf{acc}$.
\end{lemma}

\begin{proof}
\leanok
Completeness: the encoding of the run satisfies every clause, the window clauses because
the windows of a run pass the local check (\texttt{locallyConsistent\_localCfgOf}) and the
circuit computes it, which is its specification (\texttt{IsCheckCircuit}). Soundness: the
time-$0$ rows of a satisfying assignment determine the free strings; by induction on time
the rows encode the run, because a locally consistent window whose time-$t$ part is that
of a configuration has the time-$(t + 1)$ part of its successor (\texttt{post\_of\_pre});
the acceptance row and the emission bookkeeping then give the output. Every head stays on
an interior cell: a head moves at most one cell per step and a tape has $2S + 7$ cells.
\end{proof}

\paragraph{Source.} The tableau of~\cite[Section 10]{JNVWY20} and the standard argument of
Cook~\cite{Coo71}; the companion paper's \texttt{answer\_reduction.tex} (its
\texttt{lem:correct-tableau}).

\paragraph{Comments.} The lemma is Cook--Levin for one fixed machine; the theorem applies
it to a fixed interpreter machine for the decider's program, and the succinct description
is a circuit for the clauses of this formula --- pieces S2 and S3 of
\texttt{planning/succinct-cook-levin.md}. The interpreter machine (S2) is formalized:
\texttt{MIPRE.TM.Interp.U}, with \texttt{accepts\_of\_acceptsWithin} and
\texttt{acceptsWithin\_of\_accepts} (\texttt{MIPRE/TM/Interp/Run.lean}) --- on the inputs
$(\decider, n, T, x, y, a, b)$ with $\abs{a}, \abs{b} \leq T$, it halts having output
exactly $1$, within an explicit polynomial number of steps, if and only if $\decider$
accepts $(n, x, y, a, b)$ within cost $T$. The local check circuit is a parameter of the
formula, with its specification a hypothesis of the lemma, so that the lemma is proved once
and the circuit supplied with the describer. The windows are five cells wide: a head at one
of the middle three cells then sees, in the window, both the cell it leaves and the cell it
reaches, and the clamped moves of an input head at the ends of the input are decided from
the window alone.

\begin{lemma}[The value is approximable from below]
\label{lem:value-lower-approx}
\lean{MIPRE.ValueApprox.rePred_lt_quantumValue, MIPRE.ValueApprox.RawStrategy,
  MIPRE.ValueApprox.RawStrategy.interp, MIPRE.ValueApprox.Check,
  MIPRE.ValueApprox.primrecPred_check, MIPRE.ValueApprox.check_iff_interp,
  MIPRE.ValueApprox.exists_check_iff, MIPRE.ValueApprox.REPred.of_primrecRel_exists,
  MIPRE.rePred_lt_quantumValue_comp, MIPRE.exists_semidecider_lt_quantumValue,
  MIPRE.Cost.exists_semidecider}
\leanok
\ledgernode{1.1.7.2, 1.1.7.2.1, 1.1.7.2.2, 1.1.7.2.5, 1.1.7.2.6, 1.8.1}
\uses{def:game-description,def:q-value,def:re,lem:rational-strategies-suffice}
Write $\game_g$ for the game described by a game description $g$
(Definition~\ref{def:game-description}). The set
\[
  \{ (g, p, q) : g \text{ a game description},\ p, q \in \N,\ \valstar(\game_g) > p/q \}
\]
is recursively enumerable.
\end{lemma}

\begin{proof}
\leanok
By Lemma~\ref{lem:rational-strategies-suffice}, $\valstar(\game_g) > p/q$ if and only if
some valid exact strategy with entries in $\mathbb{Q}(i)$ has value greater than $p/q$.
A \emph{raw candidate} is a common denominator $k$, two dimensions, the Gaussian-integer
numerators of the two players' operators as lists of list matrices, and the numerators of
the state; it denotes the exact strategy with entries divided by $k$, and every valid exact
strategy with $\mathbb{Q}(i)$ entries is denoted by one (clear a common denominator of its
finitely many entries). Raw candidates are first-order data, hence enumerable. For a raw
candidate the check ``$k > 0$, each family of operators is Hermitian, idempotent
($M^2 = kM$) and sums to $k\Id$, the state is nonzero, and $p \cdot W \cdot k^2 \cdot
\langle v, v\rangle < q \cdot \sum_{x,y} w(x,y) \sum_{a,b} D(x,y,a,b)\, \mathrm{re}\langle v,
(A^x_a \otimes B^y_b) v\rangle$'' is a finite conjunction of polynomial identities and one
comparison in Gaussian-integer arithmetic, so it is primitive recursive, and it holds if
and only if the denoted strategy is valid with value greater than $p/q$ (for $q = 0$, where
$p/q = 0$, the comparison is replaced by positivity of the numerator). Dovetailing over the
candidates and halting on the first that passes the check therefore halts exactly when
$\valstar(\game_g) > p/q$; the boundary case $\valstar(\game_g) = p/q$ is correctly
non-halting.
\end{proof}

\paragraph{Source.} Folklore; see~\cite[Section 12]{JNVWY20}, and statement (S) in the
proof of \texttt{cor:mip-re} of \texttt{recursive.tex} for a complete argument. The
statement was previously given for $\synval$ ``and likewise for $\valstar$''; it is now the
$\valstar$ statement, which is the one the pipeline consumes
(Remark~\ref{rem:compression-abstract}) and the one the paper proves. The $\synval$
analogue needs its own enumeration and density argument, since rounding the entries of a
synchronous strategy preserves neither projectivity nor synchronicity; it is not needed and
not planned. Thresholds are nonnegative rationals written as fractions $p/q$ of naturals, as
in the paper's (S), where $t \in [0,1]$; a negative threshold is trivial since $\valstar
\geq 0$. In Lean the game description is \texttt{HaltingGameValue.GameData}
(Definition~\ref{def:game-description}), read as a game of the foundations by
\texttt{GameData.game}, and recursive enumerability is Mathlib's \texttt{REPred}, the
domain of a partial computable function; integers travel as pairs of naturals, because
Mathlib's primitive recursion has no arithmetic on $\mathbb{Z}$. The form the
compressibility criterion consumes (its hypothesis that the complement of $B$ is the halting
set of a program, Lemma~\ref{lem:compressible-criterion}) is
\texttt{MIPRE.exists\_semidecider\_lt\_quantumValue}: for a computable family of game
descriptions and a threshold $p/q$, a well-scoped program of the ambient model halting
exactly on the strings whose game has value greater than $p/q$. It comes from the generic
passage \texttt{MIPRE.Cost.exists\_semidecider} from an \texttt{REPred} on bit strings to a
program (Definition~\ref{def:re}): a \texttt{ToPartrec} code for the predicate, translated
into the model by \texttt{Prog.ofCode}, run after an ambient loop has shifted every bit up
by one, since a \texttt{ToPartrec} code cannot see trailing zeros of its input and the bit
$0$ encodes as \texttt{nil}.

\paragraph{Comments.} The Lean route to the $\valstar$ half departs from the paper's proof
of (S). The paper enumerates \emph{approximate} candidates --- tuples with entries in
$\frac1k \mathbb{Z}[i]$ satisfying relaxed POVM constraints, tested by positive
semidefiniteness (Lemma~\ref{lem:norm-two-psd}) --- and repairs each into an exact POVM by
conjugating with $T_x^{-1/2}$, with the explicit slack $\Delta(d,k) = 250(\abs{\cal A} +
\abs{\cal B})^2 d/k$ (its Claims ``Stability'' and ``Density''; ledger nodes

transform of a skew-Hermitian matrix, which is what gets rounded

Lemma~\ref{lem:rational-strategies-suffice} is sharp. Lemma~\ref{lem:norm-two-psd} is
proved but no longer on the path (the ledger's node \texttt{1.1.7.2.1}, exact arithmetic

\begin{lemma}[Gaussian-rational projective measurements are dense]
\label{lem:rational-pvm-dense}
\lean{MIPRE.ValueApprox.IsPVM, MIPRE.ValueApprox.IsPVM.exists_unitary_pattern,
  MIPRE.ValueApprox.cayley, MIPRE.ValueApprox.cayley_mem_unitaryGroup,
  MIPRE.ValueApprox.exists_isSkewHermitian_cayley_eq,
  MIPRE.ValueApprox.exists_norm_eq_one_isUnit_one_add_smul,
  MIPRE.ValueApprox.norm_cayley_sub_cayley_le, MIPRE.ValueApprox.GaussianRat,
  MIPRE.ValueApprox.exists_unitary_entriesIn_norm_smul_sub_le,
  MIPRE.ValueApprox.IsPVM.exists_entriesIn_norm_sub_le}
\leanok
Let $\{M_a\}_{a \in \cal A}$ be a projective measurement on $\C^d$ and $\eta > 0$. There is
a projective measurement $\{M'_a\}_{a \in \cal A}$ on $\C^d$ with all entries in
$\mathbb{Q}(i)$ such that $\norm{M_a - M'_a} \leq \eta$ for every $a$.
\end{lemma}

\begin{proof}
\leanok
Diagonalize a Hermitian combination $\sum_a w_a M_a$ with distinct real weights $w_a$:
$M_a = U \Pi_a U^*$ for a unitary $U$ and the coordinate projections $\Pi_a$ of a pattern
$r : [d] \to \cal A$. For a unit scalar $z$, $(zU) \Pi_a (zU)^* = U \Pi_a U^*$, and
$\det(\Id + zU)$ is a polynomial in $z$ with constant term $1$, so some unit $z$ makes
$\Id + zU$ invertible; then $zU = (\Id - S)(\Id + S)^{-1}$ is the Cayley transform of the
skew-Hermitian matrix $S = 2(\Id + zU)^{-1} - \Id$. Round $S$ entrywise to $R$ with entries
in $\mathbb{Q}(i)$ and re-symmetrize, $S' = (R - R^*)/2$: $S'$ is skew-Hermitian, has
entries in $\mathbb{Q}(i)$, and $\abs{S_{ij} - S'_{ij}}$ is at most the rounding
precision. Its Cayley transform $U' = (\Id - S')(\Id + S')^{-1}$ is unitary with entries
in $\mathbb{Q}(i)$, and $\norm{zU - U'} \leq 2 \norm{S - S'}$, since $zU - U' =
2(\Id + S)^{-1}(S' - S)(\Id + S')^{-1}$ and $\norm{(\Id + S)^{-1}} \leq 1$ for skew-Hermitian
$S$ (the cross term of $\norm{x + Sx}^2$ vanishes). Finally $M'_a = U' \Pi_a U'^*$
satisfies $\norm{M_a - M'_a} \leq 2 \norm{zU - U'}$.
\end{proof}

\begin{lemma}[Exact rational strategies suffice]
\label{lem:rational-strategies-suffice}
\uses{def:tensor-strategy,def:q-value,lem:rational-pvm-dense,lem:perturbation-split}
\lean{MIPRE.ValueApprox.ExactStrategy, MIPRE.ValueApprox.ExactStrategy.IsValid,
  MIPRE.ValueApprox.ExactStrategy.EntriesIn, MIPRE.ValueApprox.ExactStrategy.value,
  MIPRE.ValueApprox.ExactStrategy.toStrategy,
  MIPRE.ValueApprox.ExactStrategy.value_le_quantumValue,
  MIPRE.ValueApprox.abs_bornValue_sub_le, MIPRE.ValueApprox.exists_exactStrategy_value_gt,
  MIPRE.ValueApprox.lt_quantumValue_iff}
\leanok
\ledgernode{1.1.7.2.3, 1.1.7.2.4}
An \emph{exact strategy} for a game $\game$ is the data of dimensions $d_\alice, d_\bob$,
matrices $A^x_a \in \C^{d_\alice \times d_\alice}$ and $B^y_b \in \C^{d_\bob \times
d_\bob}$, and a vector $u \in \C^{d_\alice} \ot \C^{d_\bob}$; it is \emph{valid} if each
$\{A^x_a\}_a$ and each $\{B^y_b\}_b$ is a projective measurement and $u \neq 0$, and its
value is
\[
  \omega \;=\; \frac{1}{\norm{u}^2} \sum_{x,y} \mu(x,y) \sum_{a,b} D(x,y,a,b)\,
  \bra{u} A^x_a \ot B^y_b \ket{u},
\]
a rational function of the entries. Suppose the answer alphabets of $\game$ are nonempty.
Then for every real $t$, $\valstar(\game) > t$ if and only if some valid exact strategy
with all entries in $\mathbb{Q}(i)$ has value $\omega > t$.
\end{lemma}

\begin{proof}
\leanok
A valid exact strategy with the state normalized to $u/\norm{u}$ is a tensor-product
strategy of value $\omega$, so $\omega \leq \valstar(\game)$. Conversely, if
$\valstar(\game) > t$ then, $\valstar$ being a supremum over a nonempty set, some
tensor-product strategy $(\psi, \{A^x_a\}, \{B^y_b\})$ has value $v > t$, and it suffices
to find, for every $\varepsilon > 0$, a valid exact strategy with $\mathbb{Q}(i)$ entries of
value $> v - \varepsilon$. Replace each $\{A^x_a\}_a$ and $\{B^y_b\}_b$ by a $\mathbb{Q}(i)$
projective measurement within $\eta$ of it (Lemma~\ref{lem:rational-pvm-dense}), and
$\psi$ by an entrywise rounding $u \neq 0$ with $\norm{u - \psi} \leq \eta/4$, so that
$\norm{u/\norm{u} - \psi} \leq \eta/2$. The value is Lipschitz in the data: by
Lemma~\ref{lem:perturbation-split}, a Born term $\bra{\psi} A \ot B \ket{\psi}$ changes by
at most $\norm{A - A'}\norm{B} + \norm{A'}\norm{B - B'} + 2\norm{A'}\norm{B'}\norm{\psi -
\psi'}$ between unit vectors $\psi, \psi'$, using $\norm{A \ot B} \leq \norm{A}\norm{B}$;
projections have norm at most one, so each term moves by at most $4\eta$ and the value, a
$\mu$-average of sums of at most $\abs{\cal A}\abs{\cal B}$ such terms, by at most
$4\abs{\cal A}\abs{\cal B}\eta$. Take $\eta < \varepsilon / (4\abs{\cal A}\abs{\cal B})$.
\end{proof}

\paragraph{Comments.} The lemma is the mathematical content of
Lemma~\ref{lem:value-lower-approx}, isolated from the computability: only the right-hand
side involves the enumeration. It accounts for the ledger's stability and density nodes
(\texttt{1.1.7.2.3}, \texttt{1.1.7.2.4}) by a different proof, as explained under
Lemma~\ref{lem:value-lower-approx}: stability is vacuous for exact candidates, and density
is Lemma~\ref{lem:rational-pvm-dense} plus the Lipschitz bound, whose constant
$4\abs{\cal A}\abs{\cal B}$ is crude and immaterial. The nonemptiness of the answer
alphabets is what makes a strategy exist at all: with it $\valstar$ is a supremum over a
nonempty set, without it the supremum is $0$ by convention and the equivalence fails for
$t < 0$. Game descriptions (Definition~\ref{def:game-description}) always satisfy it. In
Lean the data type \texttt{ExactStrategy} carries no proofs, so that the computability layer
can enumerate it and test validity.

\begin{lemma}[Decidable candidate set: a norm constraint is two psd tests]
\label{lem:norm-two-psd}
\lean{MIPRE.ValueApprox.posSemidef_realSmul_one_add_and_sub_iff}
\leanok
\ledgernode{1.1.7.2.1}
Let $M$ be a Hermitian matrix and $c$ a real number. Then $c\Id + M \succeq 0$ and
$c\Id - M \succeq 0$ both hold if and only if $-c\norm{x}^2 \leq \bra{x} M \ket{x}
\leq c \norm{x}^2$ for every vector $x$ --- which for Hermitian $M$ says
$\norm{M} \leq c$.
\end{lemma}

\begin{proof}
\leanok
Both directions are the defining property of positive semidefiniteness applied to the
quadratic form, read once forwards and once backwards; the scalar step is
$\bra{x}(c\Id \pm M)\ket{x} = c\norm{x}^2 \pm \bra{x} M \ket{x}$.
\end{proof}

\begin{lemma}[The perturbation split]
\label{lem:perturbation-split}
\uses{def:q-value}
\lean{MIPRE.ValueApprox.dotProduct_mulVec_perturb,
MIPRE.ValueApprox.kronecker_sub_kronecker,
MIPRE.ValueApprox.dotProduct_kronecker_perturb}
\leanok
\ledgernode{1.1.7.2.3}
\ledgernode{1.1.7.2.4}
For matrices $M, N$ and vectors $\psi, \phi$,
\[
\bra{\psi} M \ket{\psi} - \bra{\phi} N \ket{\phi}
  = \bra{\psi} (M - N) \ket{\psi} + \bra{\psi - \phi} N \ket{\psi}
    + \bra{\phi} N \ket{\psi - \phi},
\]
and $A \ot B - A' \ot B' = (A - A') \ot B + A' \ot (B - B')$; combining the two splits a
bipartite Born expectation in which both players' operators and the state have moved into
four terms, each moving exactly one of $A$, $B$ and $\psi$.
\end{lemma}

\begin{proof}
\leanok
Expand and cancel. Both identities are exact: no norm and no estimate occurs in either,
and the analytic bounds are obtained afterwards by bounding the terms separately.
\end{proof}

\paragraph{Comments.} This is the easy inclusion $\MIPstar \subseteq \RE$ and one half
of Theorem~\ref{thm:mipstar-eq-re}. The pipeline consumes only the $\valstar$ half, as the
semidecision procedure of Remark~\ref{rem:compression-abstract}. The commuting-operator analogue (approximating
$\valco$ from \emph{above} via the NPA hierarchy~\cite{NPA08}) is \emph{not} needed for
the main theorem, but becomes relevant for the downstream consequences in
Section~\ref{ch:downstream}.
