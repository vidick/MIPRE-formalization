other papers. Each entry gives a full statement, a pointer to the literature, and
comments on its role and on formalization. The project may initially assume these as
axioms; discharging them are independent sub-projects. Three of these results are
larger sub-projects and are developed in their own sections: the computability

The compression pipeline carries bipartite soundness in $\valstar$; synchronous and

\subsection{The Gowers--Hatami theorem}
\label{sec:rr-gowers-hatami}

\qlib

\begin{definition}[Approximate representation]
\label{def:approx-rep}
\uses{def:hs-norm}
\lean{MIPRE.IsApproxRepresentation}
\leanok
\qlib{}
Let $G$ be a finite group and $\eps \in \R$. An \emph{$\eps$-approximate
representation} of $G$ is a function $f : G \to U(d)$ such that
\begin{equation*}
  \E_{x, y \sim G} \, \mathrm{Re}\, \langle f(x) f(y), f(xy) \rangle_{hs}
  \geq 1 - \eps\;,
\end{equation*}
where $\langle \cdot, \cdot \rangle_{hs}$ is the normalized Hilbert--Schmidt
inner product of Definition~\ref{def:hs-norm}. Since each value of $f$ is
unitary, $\norm{f(x) f(y) - f(xy)}_{hs}^2 = 2 - 2\, \mathrm{Re}\, \langle f(x)
f(y), f(xy) \rangle_{hs}$, so the condition is equivalent to the defect form
$\E_{x, y \sim G} \norm{f(x) f(y) - f(xy)}_{hs}^2 \leq 2 \eps$.
\end{definition}

\begin{theorem}[Gowers--Hatami]
\label{thm:gowers-hatami}
\uses{def:hs-norm,def:approx-rep}
\lean{MIPRE.gowers_hatami}
\leanok
Let $G$ be a finite group and $f : G \to U(d)$ an $\eps$-approximate
representation of $G$. Then there exist $d' \geq d$, an isometry
$V : \C^d \to \C^{d'}$, and a representation $\rho : G \to U(d')$ such that
\begin{equation*}
  \E_{x \sim G} \, \norm{f(x) - V^\dagger \rho(x) V}_{hs}^2 \leq 2 \eps\;.
\end{equation*}
\end{theorem}

\begin{proof}
\leanok
Let $L(G, \C^d) \cong \C^{|G| d}$ carry the inner product normalized by $1/|G|$,
let $V : \C^d \to L(G, \C^d)$ be the map $u \mapsto (x \mapsto f(x) u)$, and let
$\rho$ be the right regular representation, $(\rho(x) F)(y) = F(yx)$. Then $V$ is
an isometry, and the compression $V^\dagger \rho(x) V = \E_{y \sim G}
f(y)^\dagger f(yx)$ is an average of unitaries, hence has squared
Hilbert--Schmidt norm at most $1$. Moreover
$\E_{x} \mathrm{Re}\, \langle f(x), V^\dagger \rho(x) V \rangle_{hs} =
\E_{x, y} \mathrm{Re}\, \langle f(x) f(y), f(xy) \rangle_{hs} \geq 1 - \eps$ by
hypothesis, and the claim follows by expanding
$\norm{f(x) - V^\dagger \rho(x) V}_{hs}^2$. The construction gives
$d' = |G|\, d$.
\end{proof}

\paragraph{Source.} Gowers and Hatami~\cite{GH17}. A quantitative form suited to
applications in nonlocal games, and a version adapted to the tracial setting, is given
by de la Salle~\cite{dlS21} and in~\cite{Vid22}.

\paragraph{Comments.} This is the stability result underlying all rigidity
(self-testing) arguments in the proof: an approximate representation of the Pauli group
$\F_2^{2k}$ (with its cocycle) is close to an exact one, which forces the presence of
EPR pairs. The statement above is fully formalized (statement and proof) in
\texttt{MIPRE/Background/GowersHatami/Basic.lean}, following the regular-representation
argument sketched in the proof; the intermediate statement, with the exact
representation living on the concrete index set $G \times \{1, \dots, d\}$, is
\texttt{MIPRE.gowers\_hatami\_prod}. Two remarks on the statement. (i) It asserts
only $d' \geq d$, with witness $d' = |G| d$; it does not include the dimension bound
$d' \leq (1 + C\eps) d$ of~\cite{GH17}, whose proof goes through the Fourier-analytic
route via irreducible representations. Whether the sharper bound is needed downstream
(in the Pauli-group application feeding Theorem~\ref{thm:qld}) is to be revisited; if
so it will be added as a separate strengthening. (ii) In the Lean encoding the
isometry is stored as the $d \times d'$ matrix of its adjoint, so the isometry
condition reads $V V^\dagger = \Id$ and the compression is written
$V \rho(x) V^\dagger$. The normalized Hilbert--Schmidt norm is the tracial form
needed for the synchronous framework~\cite{dlS21,Vid22}.

\subsection{Almost-synchronous correlations}
\label{sec:rr-almost-sync}

\effortHard\ \qlib

\begin{theorem}[Almost-synchronous strategies are near synchronous strategies]
\label{thm:almost-sync}
\ledgernode{1.2.1.7.2, 1.2.1.7.2.1, 1.2.1.7.2.2}
\uses{def:sync-game,def:sync-value,def:q-value,def:inconsistency}
There are universal constants $C > 0$ and $0 < c \leq 1$ such that the following holds.
Let $\game$ be a synchronous game whose question distribution has the same marginal
$\nu$ on both coordinates and gives the diagonal pointwise proportional weight:
for some $\kappa \in (0, 1]$,
\begin{equation*}
  \mu(x,x) \;\geq\; \kappa\nu(x) \qquad \text{for all } x \in \mX\;.
\end{equation*}
Then, writing $\eps = 1 - \valstar(\game)$,
\begin{equation*}
  1 - \synval(\game) \;\leq\; C \bigl( \eps / \kappa \bigr)^{c}\;.
\end{equation*}
More precisely, the correlation of every tensor-product strategy of value $1 - \eps$
is approximated, on average over the game's questions, by a mixture of synchronous
correlations. The mixture has average game value at least $1 - C(\eps/\kappa)^c$,
so at least one component attains that bound; the bound is not asserted for every component.
\end{theorem}

\paragraph{Source.} Vidick~\cite{Vid22}, building on~\cite{KPS18,PSSTW16}; the
statement above is the corollary needed here, with constants independent of the game's
size. What the source controls is not the game's value directly but the
\emph{synchronicity defect} of a strategy --- its inconsistency
(Definition~\ref{def:inconsistency}) at equal questions, taken against the common
marginal $\nu$; the two-sided correlation replacement uses that common marginal.
The diagonal-weight hypothesis is what turns the game's rejection
probability $\eps$ into the bound $\eps/\kappa$ on that defect, the pairs $(x,x)$ carrying
at least a $\kappa$ fraction of the weight of $x$. Without it the two values are unrelated
(Remark~\ref{rem:almost-sync-hypothesis}).

\paragraph{Comments.} The proof uses the orthonormalization
lemma (Section~\ref{sec:rr-orthonormalization}) and a rounding argument. Used
contrapositively, the theorem turns a
bound $\synval(\game) \leq 1 - \delta$ into $\valstar(\game) \leq 1 - \Omega(\kappa
\delta^{1/c})$: the diagonal weight multiplies the gap recovered, so a small $\kappa$ is a
loss. The hypothesis is also not preserved by repetition. The pipeline instead carries
soundness in $\valstar$ throughout and uses direct bipartite repetition.

\begin{remark}[The diagonal-weight hypothesis cannot be dropped]
\label{rem:almost-sync-hypothesis}
\uses{thm:almost-sync,def:sync-game,def:sync-value,def:q-value}
Some lower bound on the diagonal weight is necessary, and its absence is not merely
lossy: the two values become unrelated. Let $\mX = \mA = \{0,1\}$, let $\mu$ give weight
$\frac12$ to each of $(0,1)$ and $(1,0)$ and none to the diagonal, and let $D$ accept
exactly the answer pair $(a,b) = (0,1)$ at those two question pairs and reject unequal
answers at the two diagonal pairs, as Definition~\ref{def:sync-game} requires --- which
costs nothing here, the diagonal carrying no weight. Alice answering $0$ and Bob answering
$1$, deterministically, is a tensor-product strategy of value $1$, so
$\valstar(\game) = 1$. A synchronous strategy, by contrast, is a single family
$\{M^x_a\}$ of projective measurements against a tracial state $\tau$; writing
$P = M^0_0$ and $Q = M^1_0$, its value is
\begin{equation*}
  \tfrac12 \tau\bigl( P (\Id - Q) \bigr) + \tfrac12 \tau\bigl( Q (\Id - P) \bigr)
  \;=\; \tfrac12 \bigl( \tau(P) + \tau(Q) \bigr) - \tau(PQ)
\end{equation*}
by traciality. Now $\tau(PQ) \geq \max\{0,\ \tau(P) + \tau(Q) - 1\}$: the first bound
because $\tau(PQ) = \tau(QPQ) \geq 0$, the second because
$\tau\bigl( (\Id - P)(\Id - Q) \bigr) = \tau\bigl( (\Id - Q)(\Id - P)(\Id - Q) \bigr)
\geq 0$. With $s = \tau(P) + \tau(Q)$ the value is therefore at most
$\min\{ s/2,\ 1 - s/2 \} \leq \frac12$, attained in dimension $1$ at $P = \Id$, $Q = 0$.
Hence $\synval(\game) = \frac12$ while $1 - \valstar(\game) = 0$.

This game is not symmetric, and for this instance symmetry alone repairs the conclusion:
symmetrizing $D$ so as to accept every unequal answer pair off the diagonal gives
$\synval(\game) = 1$ too, by $P = \ket{0}\bra{0}$, $Q = \ket{1}\bra{1}$ in dimension $2$.
The counterexample therefore refutes the theorem as quantified over all synchronous games,
rather than exhibiting a failure among the symmetric games the pipeline produces; what the
source's proof needs in either case is the diagonal weight itself, the synchronicity
defect being measured against the marginal and controlled only by the diagonal pairs.

A positive diagonal weight does not rescue the printed statement either: the hypothesis
has to be quantitative. For $p \in (0,1)$ let $\mu$ give weight $p/2$ to each of $(1,1)$
and $(2,2)$ and $(1-p)/2$ to each of $(1,2)$ and $(2,1)$, with $D$ accepting equal answers
on the diagonal and $(a,b) = (0,1)$ off it; every question then carries diagonal weight
exactly $p$. Answering $0$ and $1$ deterministically loses only the diagonal, so
$\valstar(\game) \geq 1 - p$, while a synchronous strategy wins the diagonal with certainty
($\sum_a \tau(M^x_a M^x_a) = 1$ for a projective family) and, by the computation above, at
most half of the remaining weight, so $\synval(\game) = \frac{1+p}{2}$. Thus
$1 - \synval(\game) = \frac{1-p}{2}$ stays above $\frac14$ for $p \leq \frac12$ while
$1 - \valstar(\game) \leq p$ tends to $0$: no constants independent of the game satisfy the
printed inequality, and the $\kappa$ of the corrected one cannot be absorbed into $C$.

Nor does the hypothesis survive repetition: if $\mu(x,x) = \kappa \sum_y \mu(x,y)$ for
every $x$, the $k$-fold product distribution satisfies the same inequality with $\kappa^k$
and no better. A bound on the synchronous value of a $k$-fold repetition therefore
does not supply a uniform quantum gap once $k$ grows with the instance size. The
pipeline's repetition theorem therefore uses direct bipartite repetition, with no
application of Theorem~\ref{thm:almost-sync} to either the input or repeated game.
The separate role of almost-synchronicity in the paper's tensor-code proof is explained
in Remark~\ref{rem:almost-sync-transitive}.

Even with the hypothesis, the transport cannot by itself reach the threshold
$\frac12$ that Definition~\ref{def:mipstar} demands. Take $\mX = \{1,2\}$, a singleton
answer alphabet, $\mu(1,1) = \mu(2,2) = \frac12$, and $D$ accepting at $(1,1)$ only: both
values are $\frac12$ and $\kappa = 1$, so the corrected inequality forces $C \geq
2^{c-1}$, whence $(2C)^{-1/c} \leq \frac12$. From $\synval(\game) \leq \frac12$ it then
concludes at best $\valstar(\game) \leq 1 - \kappa (2C)^{-1/c}$, and that bound is never
better than $\frac12$. The threshold must therefore come from
Theorem~\ref{thm:direct-repetition-q}, which is a $\valstar$ theorem, and carrying
soundness in $\valstar$ throughout avoids this transport loss.
\end{remark}

\subsection{The orthonormalization lemma}
\label{sec:rr-orthonormalization}

\effortMedium\ \qlib

\begin{theorem}[Orthonormalization]
\label{thm:orthonormalization}
\ledgernode{1.2.1.7.1, 1.2.1.7.1.1, 1.2.1.7.1.2, 1.2.1.7.1.3}
\uses{def:distance}
\lean{MIPRE.Orthonormalization.povm_orthogonalization_finDim}
\leanok
Let $\cal{M}$ be a von Neumann algebra on a finite-dimensional Hilbert space, let
$\varphi$ be a normal state on $\cal{M}$, and let $\{M_a\}_{a \in \mA}$ be a POVM in
$\cal{M}$ that is $\eps$-nearly projective:
\begin{equation*}
  \varphi\Bigl( \sum_{a \in \mA} M_a^2 \Bigr) > 1 - \eps\;.
\end{equation*}
Then there is a projective measurement $\{P_a\}_{a \in \mA}$ in $\cal{M}$ with
\begin{equation*}
  \varphi\Bigl( \sum_{a \in \mA} (M_a - P_a)^*(M_a - P_a) \Bigr) < 9\eps\;.
\end{equation*}
The same holds, on average, for a family $\{M^x_a\}$ indexed by $x \sim \mu$, by applying
it to $\varphi$ averaged over $x$.
\end{theorem}

\begin{remark}[What is proved, and in what generality]
\label{rem:orthonormalization-scope}
\uses{thm:orthonormalization}
\leanok
The statement above is the finite-dimensional case, which is the one the main theorem
needs and the one that is formalized. The vendored development proves three
unconditional cases, in increasing generality of the algebra and decreasing generality of
the state: every von Neumann algebra on a finite-dimensional space, for every normal
state (\texttt{MIPRE.Orthonormalization.povm\_orthogonalization\_finDim}, the
statement above, bridging \texttt{povm\_orthogonalization\_finDim\_vn}); $\cal{B}(H)$
for arbitrary $H$ (\texttt{povm\_orthogonalization\_fullAlgebra\_general}); and II$_1$
factors whose trace and state are both of trace-class form
(\texttt{povm\_orthogonalization\_II\textsubscript{1}Factor}), which is the case a
tracial, commuting-operator strategy lives in.

The general case --- every von Neumann algebra, every normal state --- is proved only as
an implication. Its hypothesis, \texttt{MvNStructureTheory}, bundles eight facts: seven
are classical Murray--von Neumann structure theory that Mathlib does not have (normal
functionals are of trace-class form; the type decomposition; the finite decomposition;
increasing nets of finite projections in the semifinite part; the center-valued trace;
halving in type III), and the eighth is the measurable-selection step of the source's own
Lemma~3.1 for the finite type~I part. No term of that type is constructed, so the
unconditional general statement remains open upstream, and with it the three companion
results (Theorems 1.1 and 1.4 and Corollary 1.5 of~\cite{dlS21}), which are formalized in
finite dimension and conditionally in general.
\texttt{MIPRE/Background/Orthonormalization/Axioms.lean} makes the split mechanical: it
fails to build if an unconditional result acquires an axiom, or if one of the four open
targets is closed upstream without this blueprint being updated.
\end{remark}

\paragraph{Source.} de la Salle~\cite{dlS21}, which gives a dimension-free bound with
explicit constants; earlier versions with worse parameters appear in~\cite{KPS18}
and in~\cite{JNVWY20}. The Lean proof is vendored from the
\texttt{orthogonalization} package of the \texttt{commuting-repetition} repository
(the same upstream as Theorem~\ref{thm:direct-repetition-co}) by
\texttt{scripts/vendor-repetition.py}; see that directory's \texttt{README.md} for what
the copy changes (import prefixes, nothing else).

\paragraph{Comments.} Used constantly: soundness analyses produce POVMs that are only
approximately projective, and this lemma replaces them by genuinely projective
measurements on the same space, which is what makes the ``projective strategies
throughout'' convention of Section~\ref{sec:departures} viable (avoiding Naimark
dilation). The finite-dimensional case suffices for the main theorem and is a clean,
self-contained formalization target in linear algebra.

The first of the two claims about the source made by the verification
campaign~\cite{Audit26} --- that this statement was over-hypothesized, the result needing
only a \emph{normal} state rather than a normal faithful tracial one, with constant $9$
and exponent $1$ --- is now settled in the direction the campaign predicted, and settled
by proof rather than by reading: the statement above carries neither faithfulness nor
traciality and is checked. That matters beyond bookkeeping, because the tracial
hypothesis excluded the bipartite vector-state applications that several of the
projectivization sites of Section~\ref{ch:proof-structure} need; those sites can now use
this lemma, where before they needed Naimark dilation or the projective-by-definition
convention of Section~\ref{sec:departures}. What is settled is the mathematics, not the
attribution: the campaign's second claim, that crediting the weaker predecessor
to~\cite{KPS18} and~\cite{JNVWY20} is wrong and the source credits the low individual
degree paper and Kempe--Vidick instead, is still unchecked here, \cite{dlS21} not being
vendored in this repository. The campaign also offers a consistency-to-projectivity step
giving $O(\delta)$ in place of $O(\delta^{1/4})$, which would sharpen the proof of
Theorem~\ref{thm:almost-sync}.

The formalized hypothesis and conclusion are both strict. For a POVM, completeness
identifies the defect with $1-\varphi(\sum_a M_a^2)$, but does not turn a non-strict
bound into a strict one. A compactness argument in finite dimension would retain the
constant $9$ at the non-strict bound --- apply the theorem at every larger error, then
pass to a convergent subsequence of projective measurements and use continuity --- and
would also cover zero defect. Corollary~\ref{cor:ortho-from-consistency}, the one
application in this project, does not need it: its Cauchy--Schwarz step produces slack
$\gamma^2$ over the bound it has to clear, so the strict hypothesis holds as stated. The
comments after that corollary record the repair.

\subsection{Magic Square rigidity}
\label{sec:rr-magic-square}

\effortEasy\ \qlib

\begin{theorem}[Magic Square rigidity]
\label{thm:ms-rigidity}
\uses{def:q-value,def:distance}
Let $\strategy = (\ket{\psi}, A, B)$ be a strategy that succeeds in the Magic Square
game $\game^{\MS}$ with probability $1 - \eps$. Then there exist local isometries
$\phi_\alice : \mH_\alice \to (\C^2)^{\ot 2} \ot \mH_{\alice''}$, $\phi_\bob :
\mH_\bob \to (\C^2)^{\ot 2} \ot \mH_{\bob''}$ and a state $\ket{\aux} \in
\mH_{\alice''} \ot \mH_{\bob''}$ such that
\begin{equation*}
  \norm{ \phi_\alice \ot \phi_\bob \ket{\psi} - \ket{\EPR_2}^{\ot 2} \ot \ket{\aux} }
  \leq O(\sqrt{\eps})\;,
\end{equation*}
and, writing $\tilde A = \phi_\alice A \phi_\alice^\dagger$, the observables for the
questions $\Variable_1$ and $\Variable_5$ satisfy
$\tilde A^{\Variable_1}_b \approx_{\sqrt\eps} \sigma^X_b \ot \Id$ and
$\tilde A^{\Variable_5}_b \approx_{\sqrt\eps} \sigma^Z_b \ot \Id$ on
$\ket{\EPR_2}^{\ot 2} \ot \ket{\aux}$ (and symmetrically for $\tilde B$). In
particular the corresponding $\pm 1$-observables approximately anticommute:
$\tilde A^{\Variable_1} \tilde A^{\Variable_5} \approx_{\sqrt\eps}
- \tilde A^{\Variable_5} \tilde A^{\Variable_1}$.
\end{theorem}

\paragraph{Source.} Wu, Bancal, McKague and Scarani~\cite{WBMS16}; the game is the
Mermin--Peres magic square. The form above is~\cite[Section 8]{JNVWY20}.

\effortMedium

\begin{lemma}[Direct Magic Square anticommutation --- what the pipeline actually needs]
\label{lem:ms-direct-anticomm}
\ledgernode{1.2.2.4.1, 1.2.2.4.1.6, 1.2.2.4.1.7, 1.2.2.4.1.9}
\uses{def:q-value,def:magic-square,def:lcs-game,def:state-distance}
\lean{MIPRE.QLD.MS.anti, MIPRE.QLD.MS.ms_direct_anticomm,
MIPRE.QLD.MS.ms_direct_anticomm', MIPRE.QLD.MS.ms_direct_anticomm_avg}
\leanok
Let $\ket{\psi}$ be a unit vector on $\C^{d_\alice} \ot \C^{d_\bob}$ and let $A, B$ be
arbitrary POVM families for the Magic Square game (Definition~\ref{def:lcs-game} at the
magic square's layout), failing it with probability at most $\eps \geq 0$. Write $B_j$ for the
$\pm 1$-observable of the player who receives variable $\Variable_j$ --- the difference of the
two \emph{bit} outcomes, every other answer contributing zero --- and $A_j$ for the other
player's. Then
\begin{equation*}
  \norm{ (\{ B_1, B_5 \} \ot \Id) \ket{\psi} }^2 \;\leq\; 186624\, \eps\;,
  \qquad
  \norm{ (\{ A_1, A_5 \} \ot \Id) \ket{\psi} }^2 \;\leq\; 186624\, \eps\;,
\end{equation*}
where $\{X, Y\} = XY + YX$. The bound holds on the \emph{original} observables: no isometry,
no projectivity, no full Schmidt rank and no extracted EPR pairs are required. It averages
without loss: for a family indexed by $\omega$ on the same pair of spaces, with weights
$\nu_\omega \geq 0$,
$\sum_\omega \nu_\omega \norm{(\{B_{1,\omega}, B_{5,\omega}\} \ot \Id)\ket{\psi_\omega}}^2
\leq 186624 \sum_\omega \nu_\omega \eps_\omega$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:orthonormalization}
Dilate each constraint POVM, after repairing invalid outcomes to a fixed parity-valid triple,
to a projective measurement on one ancilla; its outcome PVM gives commuting reflections
$C_{c,j}$ whose product over a constraint is exactly that constraint's sign. Bob's original
POVMs are untouched. Uniformity of the $36$ oriented question incidences bounds each
conditional loss by $36\eps$, and
$\norm{(C_{c,j} - B_j)\ket{\psi}}^2 \leq 2 - 2(1 - 2\ell_{c,j}) \leq 4 \ell_{c,j}$ using
$C_{c,j}^2 = \Id$ and $B_j^2 \leq \Id$, so $\gamma := \max_{c,j} \norm{(C_{c,j} -
B_j)\ket{\psi}}$ satisfies $\gamma^2 \leq 144 \eps$.

Write $a, \dots, i$ for Bob's variables $1, \dots, 9$ in row-major order. Substituting one
operator at a time costs $\gamma$ each, cross-party products appearing in reversed order, and
a constraint relation inserted before a right suffix of length $k$ costs the relation's own
$3\gamma$ plus $2k\gamma$. The six-step path
\begin{equation*}
  d\,a\,e\,b \;\to\; g\,e\,b \;\to\; g\,h \;\to\; i \;\to\; -f\,c
  \;\to\; -d\,e\,c \;\to\; -d\,e\,a\,b\;,
\end{equation*}
applying the column-one, column-two, row-three, column-three, row-two and row-one
constraints in turn, costs $(7 + 3 + 3 + 3 + 5 + 3)\gamma = 24\gamma$, and its two ends give
$\norm{ d (ae + ea) b \ket{\psi} } \leq 24 \gamma$. The sign appears exactly once, at the
column-three constraint, which is the one of product $-1$.

Put $R = ae + ea$ and $T = Rb$. Since $\norm{(\Id - B_j^2) W \ket{\psi}} \leq (2 + k)\gamma$
for a word $W$ of length $k$, the two length-three words in $T$ give
$\norm{(\Id - d^2) T \ket{\psi}} \leq 10\gamma$, and hence $\norm{T\ket{\psi}} \leq 34\gamma$.
Finally Alice's reflection for variable $2$ commutes with the Bob operator $R$, squares to
$\Id$, and approximates $b$ on $\ket{\psi}$ to within $\gamma$, which removes the trailing
factor: $\norm{R \ket{\psi}} \leq 34\gamma + 2\gamma = 36\gamma$. Squaring and substituting
$\gamma^2 \leq 144\eps$ gives $1296 \cdot 144 = 186624$. Exchanging the parties gives Alice's
bound. The averaged form is the pointwise bound averaged, with no Jensen step.
\end{proof}

\paragraph{Comments.} This lemma is why Theorem~\ref{thm:ms-rigidity} is no longer an
assumption of this blueprint (Remark~\ref{rem:admitted-nodes}). Full two-EPR self-testing is
strictly stronger and is \emph{not} proved by this route --- but the pipeline never needed it:
what the Pauli basis test consumes is one anticommutation bound on the original observables,
and the argument above delivers it from the game alone, with an explicit constant and without
any of the structural hypotheses rigidity would impose. For a formalization this is a much
better target than Theorem~\ref{thm:ms-rigidity}: the ingredients are a Naimark dilation, a
telescoping substitution bound, and one six-step word identity in a group generated by nine
reflections with six relations.

\paragraph{Comments.} The direct bound is the elementary anticommutation input that
the Pauli test amplifies to many qubits. Its bipartite form already matches the
soundness proof's use; a tracial specialization is optional and is not an additional
pipeline prerequisite.

One clause moved out of the statement. An earlier revision also asserted the \emph{transfer}
to the point observables of the Pauli basis test: if $P, Q$ have cross-party consistency errors
$s_X, s_Z$ against $B_1, B_5$ then
$\norm{\{P,Q\}\ket{\psi}} \leq 2 s_X + 2 s_Z + \norm{\{B_1,B_5\}\ket{\psi}}$. That is not a
statement about the Magic Square game at all --- $P$ and $Q$ are the \emph{Pauli} test's
observables --- so it belongs to Lemma~\ref{lem:qld-win}, which is where it is used, and it is
stated there. The argument is two substitutions across the parties: $P$ and $Q$ sit on one
factor and $B_1, B_5$ on the other, so $PQ\ket{\psi} \approx_{s_Z} P B_5 \ket{\psi}
= B_5 P \ket{\psi} \approx_{s_X} B_5 B_1 \ket{\psi}$, and symmetrically for $QP$.

The game this lemma is about is the one of Definition~\ref{def:lcs-game} at the magic square's
layout, and that definition is \emph{symmetric in the players}: an equation, a variable of it,
and a uniform orientation, so $36$ equiprobable oriented incidences. That is what makes the
statement's Alice half expressible at all --- a one-directional game would never send Alice a
variable, so $A^{\Variable_j}$ would not exist --- and it is where the constant comes from:
$\sum_{c,j} \ell_{c,j} \leq 36\eps$ gives $\gamma^2 \leq 144\eps$ and
$36^2 \cdot 144 = 186624$, the paper's constant. An earlier revision of the formalization had
the one-directional game, where the same argument gives $18\eps$, $\gamma^2 \leq 72\eps$ and
$36^2 \cdot 72 = 93312\,\eps$; either serves Lemma~\ref{lem:qld-win}, which needs only
$O(\eps)$, but the game and the constant must match.

\subsection{The classical low individual degree test}
\label{sec:rr-lidt}

\effortHard

\begin{definition}[Canonical-line low individual degree test]
\label{def:lidt}
\uses{def:game}
\lean{MIPRE.LIDT.lidtGame}
\leanok
Let $\F$ be a finite field with $q$ elements and let $m \geq 1$ and $d$ be integers.
The \emph{$(m,q,d)$-low individual degree test} is the two-player game in which the
verifier picks one of the following three sub-tests with probability $\tfrac13$ each,
having sampled a uniform point $u \sim \F^m$.
\begin{enumerate}
\item \textbf{Axis-parallel lines test.} Sample a uniform direction $i \sim [m]$ and a
  uniform role. One player receives the line $\ell = \{u + t e_i\}$ and answers a
  univariate polynomial $f$ of degree at most $d$; the other receives $u$ and
  answers $a \in \F$. Accept if $f(u) = a$.
\item \textbf{Self-consistency test.} Both players receive $u$ and answer elements of
  $\F$. Accept if the answers are equal.
\item \textbf{Diagonal lines test.} Sample a uniform $j \sim [m]$, a uniform direction
  $v \in \F^m$ whose coordinates after the $j$-th vanish, and a uniform role. One
  player receives the line $\ell = \{u + t v\}$ and answers a univariate polynomial
  of degree at most $md$; the other receives $u$ and answers $a \in \F$. Accept if
  $f(u) = a$.
\end{enumerate}
\end{definition}

\paragraph{Comments.} This is Figure~1 of~\cite{JNVWY21qld}. The direction $v$ of the
diagonal lines test may be zero, in which case the ``line'' is the single point $u$;
this degenerate case is part of the test as stated. In the formalization, line questions
are the lines themselves, presented canonically (the direction is rescaled so that its
first nonzero coordinate is $1$, and the base point is the point of the line whose
corresponding coordinate vanishes), so that the same line asked by two sub-tests is the
same question; a line answer is a polynomial in the parameter of that canonical
parametrization.

\begin{theorem}[Quantum soundness of the canonical-line low individual degree test]
\label{thm:lidt-soundness}
\uses{def:lidt,def:tensor-strategy,def:inconsistency}
\lean{MIPRE.LIDT.lowIndividualDegree_soundness}
\leanok
Let
\begin{equation*}
  \delta_{lidt}(\eps, m, d, q, k)
  = 10^5 \, k^2 m^4 \bigl( \eps^{1/40000} + (d/q)^{1/40000}
      + e^{-k/(2560000 m^2)} \bigr).
\end{equation*}
Let $\strategy = (\ket{\psi}, A, B)$ be a tensor-product strategy for the
$(m,q,d)$-low individual degree test that succeeds with probability at least $1 - \eps$,
and let $k$ be an integer with $k \geq 400md$ and $k > 0$. Then there exist projective
measurements $G^\alice = \{G^\alice_g\}$ on Alice's space and $G^\bob = \{G^\bob_g\}$ on
Bob's space, both with outcomes the polynomials $g$ in $m$ variables of individual degree
at most $d$, such that, writing $\delta = \delta_{lidt}(\eps, m, d, q, k)$ and letting
$u \sim \F^m$ be uniform:
\begin{enumerate}
\item Alice's point measurement $A^u$ and the evaluation $G^\bob_{[g(u) = \cdot]}$
  are $\delta$-consistent;
\item the evaluation $G^\alice_{[g(u) = \cdot]}$ and Bob's point measurement $B^u$
  are $\delta$-consistent;
\item $G^\alice$ and $G^\bob$ are $\delta$-consistent.
\end{enumerate}
\end{theorem}
\begin{proof}
\leanok
This is Theorem~1.3 of~\cite{JNVWY21qld} (\texttt{thm:main-formal} there), whose proof
is the MIPStarRE formalization vendored into this repository. The Lean proof translates
a strategy for our game into MIPStarRE's strategy container (the state as a density
matrix; the line measurements as coarse-grainings of the measurements at the canonical
line questions along the change of parametrization, which makes them covariant under
rebasing), shows that MIPStarRE's failure surrogate is at most the rejection probability
$1 - \mathrm{val}$ (the consistency defect of each branch is at most the rejection
probability of that branch, since acceptance forces the coarse-grained answers to
agree), applies their main theorem, and reads the resulting measurements back into our
vocabulary.
\end{proof}

\paragraph{Source.} Ji, Natarajan, Vidick, Wright and Yuen~\cite{JNVWY21qld},
Theorem~1.3 (the formal version for general, not necessarily symmetric, strategies).
The proof is a formalization by Sirui Lu and collaborators
(\url{https://github.com/LionSR/MIPStarRE}), vendored into this repository with the
authors' permission at \texttt{MIPRE/Background/LIDT/MIPStarRE/}; the statement above is
this project's own, and is connected to theirs by an explicit bridge.

\paragraph{Comments.} Two hypotheses correct the printed statement, as established by
that formalization: the sampling parameter satisfies $k \geq 400md$ where the paper
prints $k \geq md$, and $k$ must be positive, since the printed error vanishes at
$k = 0$. The parameter $k$ remains free: the error carries both a factor $k^2$ and a
term $e^{-k/(2560000m^2)}$. A chosen-$k$ corollary in the shape of $\delta_{qld}$ below
is a separate quantitative derivation; it is not part of the displayed Lean theorem.


\subsubsection{The paper's seeded CL test and its tensor-code route}

The following is the paper's route to a different game interface. Its tensor-code
reduction and reviewed intermediate statements are not the proof of
Theorem~\ref{thm:lidt-soundness}, and have no Lean proof marks here. The seeded game
and its theorem are in the pinned paper snapshot~\cite{AuditSync26},
\texttt{ldt.tex}, Definition of the game and Theorem \texttt{lem:ld-soundness}.

The single-codeword case of the seeded theorem \emph{is} proved here, by a different route:
an explicit adapter to the canonical-line test, Theorem~\ref{thm:lidt-cl-soundness-one} below.
It certifies none of the paper's intermediate statements, and they certify none of it.

\begin{definition}[Seeded CL low individual degree test]
\label{def:lidt-cl}
\uses{def:game,def:cl-canonical}
\lean{MIPRE.LIDT.CL.chi, MIPRE.LIDT.CL.zeroBelow, MIPRE.LIDT.CL.rep,
  MIPRE.LIDT.CL.lineParam, MIPRE.LIDT.CL.Ty, MIPRE.LIDT.CL.Question,
  MIPRE.LIDT.CL.Answer, MIPRE.LIDT.CL.Sample, MIPRE.LIDT.CL.Sample.question,
  MIPRE.LIDT.CL.lineVsPoint, MIPRE.LIDT.CL.Question.fmtOk, MIPRE.LIDT.CL.subtests,
  MIPRE.LIDT.CL.accepts, MIPRE.LIDT.CL.clGame,
  MIPRE.LIDT.CL.accepts_aline_values_eq_false,
  MIPRE.LIDT.CL.accepts_point_apolys_eq_false, MIPRE.LIDT.CL.accepts_aline_dline,
  MIPRE.LIDT.CL.aline_question_ne, MIPRE.LIDT.CL.card_chi_fiber,
  MIPRE.LIDT.CL.Answer.toValue, MIPRE.LIDT.CL.pointPOVMA, MIPRE.LIDT.CL.pointPOVMB,
  MIPRE.LIDT.LinePoly.eval_add}
\leanok
For admissible $q$, positive integers $m,d,\mathrm{ldc}$ and $m\mid q$, the
\emph{seeded CL test} is the paper's typed low-degree game with parameters
$(q,m,d,\mathrm{ldc})$. Each ordered pair of types from
$\{\mathtt{Point},\mathtt{ALine},\mathtt{DLine}\}$ has weight $1/9$, with question
contents sampled from the corresponding CL distribution. Point questions ask for
an element of $\F_q^{\mathrm{ldc}}$. Axis-line questions retain a seed-labelled
description $(u_0,s)$ and ask for $\mathrm{ldc}$ degree-$d$ polynomials; diagonal-line
questions retain $(u_0,s,v)$ and ask for $\mathrm{ldc}$ degree-$md$ polynomials.
The seed selects the direction class, with leading coordinates of the diagonal
direction set to zero. The verifier checks the answer format, equality on identical
same-type questions, and evaluation agreement for either line type against a point;
the other type pairs impose no further check. In particular, different seed
descriptions of the same geometric line remain different questions.
\end{definition}

\paragraph{Comments.} The formalization is \texttt{MIPRE.LIDT.CL.clGame}
(\texttt{MIPRE/Background/LIDT/CLGame.lean}), beside the canonical-line game
\texttt{MIPRE.LIDT.lidtGame} of Definition~\ref{def:lidt} whose soundness is proved; keeping
both in one vocabulary is what makes Remark~\ref{rem:lidt-cl-adapter}'s adapter expressible at
all. Three points.

The base point of a line question is the canonical representative $L^{\mathrm{Ln}}_w(u)$ of
Definition~\ref{def:cl-canonical} --- the canonical linear map with kernel
$\mathrm{span}\{w\}$ --- which is why that vocabulary had to come first. The seed function
$\chi$ is the paper's \texttt{eq:chi-func}, $0$-indexed: under the fixed bijection
$\F_q \cong \{0, \ldots, q-1\}$, $\chi(s)$ is the block of $q/m$ elements that $s$ falls into,
and $m \mid q$ is where it is used.

The seeding is not cosmetic, and \texttt{aline\_question\_ne} records it: two seeds in the same
block give the same base point and the same direction, hence the same geometric line, and yet
are \emph{different questions}. A strategy may answer them differently, which is exactly the
obligation Lemma~\ref{lem:lidt-reduction-setup} discharges by choosing a seed in each fibre.

The format check is not decoration, and the formalization now separates it from the subtests
(\texttt{Question.fmtOk} and \texttt{subtests}, with \texttt{accepts} their conjunction). Two
clauses of \texttt{fig:ld-decider} look alike and are opposite: an answer whose format does not
match its question's type is \emph{rejected} (``If the answers returned by the players do not
fit this format the decision procedure rejects''), whereas a well-formatted answer on one of
the two cross type pairs is \emph{accepted} (``in all cases where no action is indicated,
accept''). An earlier version of \texttt{clGame} collapsed both into one catch-all that
accepted, which made the test vacuous: the constant strategy answering a point-format answer to
\emph{every} question, the two line types included, passes every subtest, so the game has a
value-$1$ strategy whose point measurements are an arbitrary function of the point, and
Theorem~\ref{thm:lidt-cl-soundness} is false for it. That defect and its repair are recorded in
\texttt{reports/clgame-format-check-missing.md}; the statement of this definition was correct
throughout, which is why the repair changes only the Lean and this paragraph.

Where \texttt{fig:ld-decider} says ``$t \in \F_q$ is such that $x = u_0 + tw$'', the point may
fail to lie on the line and the condition is then vacuous. The formalization requires
membership, as the canonical-line game already does: the two readings agree on the support of
the question distribution, and following the convention of the test whose soundness is proved
is what keeps the two comparable. The distribution itself is uniform over the nine ordered type
pairs and the ambient $(u, s, v)$, so normalization is a single division rather than the
per-subtest bookkeeping Definition~\ref{def:lidt} needs.

Each $\chi$-fibre has exactly $q/m$ seeds (\texttt{card\_chi\_fiber}), so a uniform seed gives
a uniform direction index and the induced axis-parallel line distribution is uniform over
lines. This is the hypothesis $m \mid q$ earning its place, and it is the input
Lemma~\ref{lem:lidt-test-transfer} consumes.

\begin{theorem}[Quantum soundness of the seeded CL test]
\label{thm:lidt-cl-soundness}
\ledgernode{1.2.1}
\uses{def:lidt-cl,lem:lidt-ldc,def:inconsistency}
There are universal constants $a\geq 1$ and $0<b<1$ such that every projective
strategy passing the seeded CL test with probability at least $1-\eps$, for
$\eps>0$, admits projective measurements $G^\alice,G^\bob$ on the respective
players' spaces with outcomes tuples $(g_1,\ldots,g_{\mathrm{ldc}})$ of polynomials
of individual degree at most $d$. Their evaluations are consistent with the other
player's point measurement on average over uniform $u\in\F_q^m$, and the two
full tuple measurements are consistent with each other, all with error at most
\[
  \delta_{\mathrm{CL}}(\eps,q,m,d,\mathrm{ldc})
  = a(dm\,\mathrm{ldc})^a\bigl(\eps^b+q^{-b}+2^{-bmd}\bigr).
\]
The full tuple consistency has no question average. The general simultaneous theorem
remains open. Its single-codeword case is proved in
Theorem~\ref{thm:lidt-cl-soundness-one}; the remaining extension is
Lemma~\ref{lem:lidt-ldc}. The tensor-code intermediates below remain separate targets.
\end{theorem}

\begin{remark}[The CL-to-canonical adapter and parameter corollary]
\label{rem:lidt-cl-adapter}
\uses{def:lidt-cl,def:lidt,thm:lidt-soundness,thm:lidt-cl-soundness}
Reusing Theorem~\ref{thm:lidt-soundness} to prove the single-codeword case of
Theorem~\ref{thm:lidt-cl-soundness} requires an explicit strategy adapter. It must
control the question distributions and subtest weights, reverse coordinate order,
remove the seed labels with a proved choice or averaging argument, and transport
line answers under canonicalization and rebasing. It must also preserve the degree
bounds and projectivity, or account for their repair, and recover the two original
point-consistency conclusions and full-codeword consistency. A seeded strategy need
not assign the same measurement to different descriptions of the same line.

A separate quantitative obligation is to choose an integer $k\geq400md$, $k>0$,
and bound the canonical theorem's error, together with every adapter loss, by the
single-codeword $\delta_{\mathrm{CL}}$ above. The simultaneous extension and the
padded Pauli use must then propagate the changed dimension and any ancillas.
Formalizing the paper's tensor-code route below is an alternative way to establish
the seeded CL theorem; a proof by either route would not certify the intermediate
claims of the other.

\emph{Both obligations are now met for $\mathrm{ldc}=1$}, which suffices for the padded QLD application: Lemmas~\ref{lem:lidt-cl-adapter-maps}--\ref{lem:lidt-cl-adapter-params}
and Theorem~\ref{thm:lidt-cl-soundness-one}. What the adapter does \emph{not} supply is the
simultaneous extension of Lemma~\ref{lem:lidt-ldc}, and it certifies none of
Lemmas~\ref{lem:lidt-reduction-setup}--\ref{lem:lidt-derandomize}.
\end{remark}

\paragraph{The canonical-line route, and what it certifies.} The adapter and the parameter
corollary the remark calls for are built, and the single-codeword case of
Theorem~\ref{thm:lidt-cl-soundness} follows from Theorem~\ref{thm:lidt-soundness} --- which
this project has --- rather than from the tensor code test, which it does not.
\texttt{planning/lidt-cl-adapter.md} records the design, including two corrections found while
proving it.

One structural remark before the statements. The canonical-line test's only same-type subtest is
point self-consistency, so a pair of two line questions carries no weight on that side and
\emph{no line-synchronicity transfer is needed}: no Lemma~\ref{lem:lidt-sync-transfer}, no
Schwartz--Zippel, and no $d/q$ term from that source. The paper's route needs them because the
tensor code test has a line synchronicity condition. A $d/q$ term still appears in
$\delta_{\mathrm{CL}}$, from the canonical-line theorem's own error.

\begin{lemma}[The adapter's two maps, and acceptance]
\label{lem:lidt-cl-adapter-maps}
\uses{def:lidt-cl,def:lidt,def:tensor-strategy,def:cl-canonical}
\lean{MIPRE.LIDT.Adapter.qmap, MIPRE.LIDT.Adapter.amap, MIPRE.LIDT.Adapter.hD_qmap}
\leanok
Assume $m\mid q$ and fix a family member: a seed position $\sigma(i)$ in each $\chi$-fibre, and a
scale $c\neq0$. Let $\rho$ be coordinate reversal on $\F_q^m$. Map a canonical-line question to a
seeded one by
\[
  \mathtt{point}\ u \mapsto \mathtt{Point}\ \rho u,\qquad
  \ell \mapsto (L^{\mathrm{Ln}}_{\rho w}(\rho u_0),\ s_{\sigma},\ c\cdot\rho w),
\]
$w$ the direction of $\ell$ and $u_0$ its base point, with $s_\sigma$ the seed $\sigma$ selects in
the fibre of the $\chi$-index of $\rho w$, and the third component omitted for an axis-parallel
line. Map a seeded answer back by reading its single component, reparametrizing a diagonal-line
polynomial affinely by $t\mapsto c^{-1}t+c^{-1}\mathrm{sh}(\ell)$, where $\mathrm{sh}(\ell)$ is the
shift between the two tests' base points for $\ell$, and sending an answer whose format does not
match its question to a fixed answer.

Then on every question pair the canonical-line test gives positive weight, every answer pair the
seeded test accepts at the images is accepted by the canonical-line test at the originals.
\end{lemma}

\begin{proof}
\leanok
Five cases, one per shape of a canonical-line sample: point self-consistency, an axis-parallel
line against a point, a diagonal line against a point, and the last two with the roles swapped.
The format branches are unreachable, because the seeded test \emph{rejects} an answer whose
format does not match its question.

On axis-parallel lines the two tests' base points agree exactly, so the reparametrization is the
identity there. On diagonal lines they do not: the seeded base point zeroes the coordinate at the
first nonzero entry of $\rho w$, which is the reversal of the \emph{last} nonzero entry of $w$,
while the canonical presentation zeroes it at the first. The two parameters of a point of the line
therefore differ by the shift $\mathrm{sh}(\ell)$, and by the scale $c$; the affine
reparametrization undoes both, and preserves the degree bound because composing with a degree-one
polynomial does.

The restriction to the support is not cosmetic: off it the claim is false, a pair of two line
questions being a real question pair of the seeded test that the canonical-line test has no
subtest for.
\end{proof}

\begin{lemma}[The adapter's cost in the question distributions]
\label{lem:lidt-cl-adapter-weights}
\uses{lem:lidt-cl-adapter-maps,def:lidt-cl,def:lidt}
\lean{MIPRE.LIDT.Adapter.pushforward_le}
\leanok
Average the family of Lemma~\ref{lem:lidt-cl-adapter-maps} over the seed positions and the scale,
uniformly and independently. Then for every seeded question pair $(x,y)$ the average
canonical-line mass of its preimage is at most $3m$ times the seeded mass of $(x,y)$.
\end{lemma}

\begin{proof}
\leanok
Both distributions are sums over a sample type against an indicator, so the bound is a count of
samples. Reordering the average puts it as a weight times a count of \emph{family members}, and
that is what the shapes differ in: the image of a point question does not depend on the family
member at all, while a line question's pins the seed position at one index and, for a diagonal
line, the scale as well.

Four of the nine shapes carry no weight, by the structural remark above. For a pair of points the
averaging is neutral and the requirement is $1/(3q^m)\leq C/(9q^m)$. For an axis-parallel line
against a point the single contributing sample has weight $1/(6mq^m)$, the family is pinned at one
of $m$ indices, and the seeded test puts at least $1/(9q^{m+1})$ on the pair, so $C\geq3/2$
suffices --- the factor $q/m$ a fixed seed choice would have cost, paid back exactly by the
averaging. For a diagonal line the crude bound ``number of contributing samples times the largest
of their weights'' replaces the geometric series over the admissible $j$, which costs a factor
$m$ and nothing in $q$; with the scale pinned as well the constant is $3q/(2(q-1))$, which $3m$
covers for every $q\geq2$.

Two points where the design is forced rather than free. Carrying the direction unrescaled would
concentrate the push-forward on a $1/(q-1)$ fraction of the seeded questions describing the same
line, and force $C$ to grow with $q$; averaging over one global scale fixes it. And a
\emph{singleton} line --- direction zero --- must be sent to a seed with $\chi=m-1$, not $\chi=0$:
the seeded test has $q^{\chi}$ samples for such a question, so only the largest $\chi$ dominates
the canonical-line mass, and the wrong choice costs a factor $q^{m-1}$.
\end{proof}

\begin{lemma}[The parameter choice]
\label{lem:lidt-cl-adapter-params}
\uses{thm:lidt-soundness,def:lidt-cl}
\lean{MIPRE.LIDT.clK, MIPRE.LIDT.deltaCL, MIPRE.LIDT.lidtError_le_deltaCL}
\leanok
Take $a=2\cdot10^{11}$, $b=1/40000$ and $k=400m^3d+400m$. Then for all $m,d\geq1$, all $q$ and all
$\eps\geq0$,
\[
  100000\,k^2m^4\Bigl((3m\eps)^{b}+(d/q)^{b}+e^{-k/(2560000m^2)}\Bigr)
  \ \leq\ a(dm)^{a}\bigl(\eps^{b}+q^{-b}+2^{-bmd}\bigr),
\]
the left side being the error of Theorem~\ref{thm:lidt-soundness} at $k$ on failure $3m\eps$ and
the right side $\delta_{\mathrm{CL}}$ at $\mathrm{ldc}=1$. Moreover $k\geq400md$ and $k>0$.
\end{lemma}

\begin{proof}
\leanok
Term by term. $(3m\eps)^b=(3m)^b\eps^b\leq 3m\,\eps^b$ and $(d/q)^b=d^b q^{-b}\leq d\,q^{-b}$,
since $x^b\leq x$ for $x\geq1$ and $b\leq1$. The exponential is where $k$ is chosen:
$e^{-k/(2560000m^2)}\leq2^{-bmd}$ asks for $k\geq 2560000\,b\,m^3d\ln2=64(\ln2)m^3d$, about
$44.4\,m^3d$, and $400m^3d$ clears it; the $400m$ summand is there only so that $k>0$ and
$k\geq400md$ hold for every $m,d\geq1$. The prefactor is then
$100000k^2m^4\cdot3md\leq1.92\cdot10^{11}(dm)^{11}\leq a(dm)^a$, using $k\leq800m^3d$.

$d\geq1$ is needed and is not cosmetic: at $d=0$ the right side is identically zero, so no
positive error could be bounded by it.
\end{proof}

\begin{theorem}[Quantum soundness of the seeded CL test, one codeword]
\label{thm:lidt-cl-soundness-one}
\uses{def:lidt-cl,def:inconsistency,thm:lidt-soundness,lem:lidt-cl-adapter-maps,
  lem:lidt-cl-adapter-weights,lem:lidt-cl-adapter-params}
\lean{MIPRE.LIDT.Adapter.clSoundness_ldc_one, MIPRE.LIDT.Adapter.clSoundness_ldc_one_deltaCL}
\leanok
Let $m,d\geq1$, $m\mid q$ and $\eps\geq0$. Every projective strategy passing the seeded CL test at
$(q,m,d,1)$ with probability at least $1-\eps$ admits projective measurements
$G^\alice,G^\bob$ on the respective players' spaces with outcomes polynomials of individual degree
at most $d$, such that A's point measurement is consistent with $G^\bob$ evaluated at the point on
average over uniform $u\in\F_q^m$, $G^\alice$ evaluated at the point is consistent with B's point
measurement, and $G^\alice$ and $G^\bob$ are consistent with each other, all with error at most
$\delta_{\mathrm{CL}}(\eps,q,m,d,1)=a(dm)^a(\eps^b+q^{-b}+2^{-bmd})$ for the constants of
Lemma~\ref{lem:lidt-cl-adapter-params}.
\end{theorem}

\begin{proof}
\leanok
Lemma~\ref{lem:lidt-cl-adapter-maps} gives, for each family member, a strategy for the
canonical-line test built from the seeded one by reindexing the questions and coarse-graining the
answers --- same spaces, same state, so projectivity and the dimensions are untouched.
Lemma~\ref{lem:lidt-cl-adapter-weights} bounds the failure of the adapted strategy by $3m$ times
the failure of the original, on average over the family, so some member achieves it; that is the
choice of seed the seeding makes necessary. Theorem~\ref{thm:lidt-soundness} applies to it at the

$\rho u$. Consistency is invariant under both.
\end{proof}

\begin{theorem}[Synchronous tensor code test]
\label{thm:tensor-codes}
\ledgernode{1.2.1.7, 1.2.1.7.3}
\uses{def:sync-value,def:inconsistency,thm:orthonormalization}

contract that the seeded CL reduction consumes.
\end{proof}

\begin{corollary}[Rigidity transfer to symmetric strategies, with the marginal condition]
\label{cor:sync-transfer}
\ledgernode{1.2.1.7.2.3}
\uses{thm:almost-sync,def:sync-game,def:distance}
Let $\game$ be a symmetric game with common question marginal $\nu$, let $\mY$ and $\mB$ be
finite non-empty auxiliary sets, let $p$ be a distribution on $\mX \times \mY$, and let
$R \geq 1$ satisfy
\begin{equation*}
  p_{\mX}(x) \;\leq\; R\, \nu(x) \qquad \text{for all } x \in \mX\;.
\end{equation*}
For each $(x,y)$ let the sets $g_{xy}(a) \subseteq \mB$, $a \in \mA$, be pairwise disjoint
(not necessarily exhaustive), and for a POVM $N^y$ write
$N^y_{g_{xy}(a)} = \sum_{b \in g_{xy}(a)} N^y_b$. Let $\kappa : [0,1] \to \R_{\geq 0}$ be
convex and non-decreasing, and suppose that every symmetric projective maximally
entangled (PME) strategy $T = (\ket{\Phi}, Q)$ for $\game$ (so $\ket{\Phi}$ is
maximally entangled), of value $\omega(T)$, admits complete auxiliary POVMs $N^y$ with
\begin{equation*}
  F_p(T, N) \;:=\; \E_{(x,y) \sim p} \sum_{a \in \mA}
    \bra{\Phi} Q^x_a \ot N^y_{g_{xy}(a)} \ket{\Phi} \;\geq\; \kappa\bigl( \omega(T) \bigr)\;.
\end{equation*}
Then there are universal constants $C_0, c_0 > 0$ --- supplied by the squared-distance
decomposition behind Theorem~\ref{thm:almost-sync} --- such that, writing
$e(\delta) = C_0 \delta^{c_0}$ and $s(\delta) = 2\sqrt{2\delta} + 8\sqrt{e(\delta)}$, every
symmetric projective strategy $\strategy = (\ket{\psi}, A)$ of value $\omega$ and
synchronicity defect
$\delta = 1 - \E_{x \sim \nu} \sum_a \bra{\psi} A^x_a \ot (A^x_a)^{\mathsf{T}} \ket{\psi}$
admits complete auxiliary POVMs $M^y$ on its own Bob space with
\begin{equation*}
  F_p(\strategy, M) \;\geq\; \max\Bigl\{ 0,\;
    \kappa\bigl( \max\{0, \omega - s(\delta)\} \bigr) - \sqrt{R\, e(\delta)} \Bigr\}\;.
\end{equation*}
Two parts of the hypothesis are load-bearing and easy to lose. \emph{Symmetric} means the
Schmidt form $\ket{\psi} = \sum_j \sqrt{r_j} \ket{j}\ket{j}$ with $r_j \geq 0$ and Bob's game
measurements the transposes of Alice's; swap symmetry alone is not a sufficient input. And the
conclusion delivers a POVM: it asserts no projectivity, so a consumer needing projective
operators must project separately, at its own cost.
\end{corollary}

\begin{remark}[The printed transfer corollary is false without the marginal condition]
\label{rem:sync-transfer-marginal}
\uses{cor:sync-transfer,thm:almost-sync}
The marginal condition $p_{\mX} \leq R\nu$ in Corollary~\ref{cor:sync-transfer} is not
book-keeping. Corollary 4.1 of~\cite{Vid22} is stated for an arbitrary auxiliary distribution
$p$, and in that unrestricted form it is false; the source applies an estimate averaged over
$\nu$ to $p_{\mX}$ without justification. The counterexample is two questions wide. Let
$\game$ ask only question $0$ and check equality, let $p$ be supported on question $1$ alone,
let $g(a) = \{a\}$, and let $\kappa = 1$. Every symmetric PME strategy meets the
auxiliary hypothesis, using the transpose of its question-$1$ measurement: on a
maximally entangled state, agreement is the normalized trace of
$\sum_a (Q^1_a)^2=\Id$. But the
symmetric projective strategy $\ket{00}$ measuring $\sigma^Z$ at question $0$ and $\sigma^X$
at question $1$ has game value $1$ and synchronicity defect $0$, while every Bob POVM achieves
auxiliary agreement exactly $\tfrac12$ --- so the conclusion would force
$\tfrac12 \geq \kappa(1) = 1$. Here $p_{\mX}$ is supported where $\nu$ vanishes, so no finite
$R$ exists and Corollary~\ref{cor:sync-transfer} does not apply.

What rescues the use made of it here is that the tensor code test's own sampling is
comfortable: $p$ is uniform on point questions while $\nu(u) = \tfrac49 n^{-m}$, giving
$R = \tfrac94$, so Lemma~\ref{lem:tensor-codes-bipartite} pays only the universal
synchronization losses. This is worth recording because a formalization must carry the
condition explicitly --- it is exactly the hypothesis a paraphrase drops --- and because it is
the one defect the 2026-09-14 adversarial round of the companion campaign turned up (a major
challenge against ledger node $1.2.1.7.2.3$, resolved by amending the node to the statement
given above rather than by repairing the source).
\end{remark}

\begin{lemma}[From the synchronous to the two-prover form]
\label{lem:tensor-codes-bipartite}
\ledgernode{1.2.1.7.8}
\uses{thm:tensor-codes,cor:sync-transfer,rem:sync-transfer-marginal,thm:orthonormalization}
With the same code hypotheses and $t=n-\Delta+1$, let a finite-dimensional two-prover
projective strategy pass the two-prover tensor code test with probability at least
$1-\eps'$, and let $k\geq12mt$. There are projective measurements
$G^\alice,G^\bob$, with outcomes in $C^{\ot m}$, on the players' respective original
spaces. With error
$\eta'=\poly(m,t,k)\cdot\poly(\eps',n^{-1},e^{-\Omega(k/m^2)})$, Alice's point
measurement is consistent with $G^\bob$ evaluated at a uniform point; the restriction
of $G^\alice$ to a uniform axis-parallel line is consistent with Bob's line
measurement; and $G^\alice,G^\bob$ are consistent as full codeword measurements.
The last conclusion has no question average. The reverse point consistency follows
from the first and third conclusions and the test's point self-consistency, with a
change of universal constants.
\end{lemma}

\begin{proof}
For symmetric PME strategies, the normalized trace realizes their correlations, so
Theorem~\ref{thm:tensor-codes} supplies the PME premise of
Corollary~\ref{cor:sync-transfer}. For a general strategy, symmetrize and apply the
corrected transfer with a concavified error function, at $R = \tfrac94$ as
Remark~\ref{rem:sync-transfer-marginal} records. Symmetrization is where care is needed: the
swap construction preserves the passing probability, but what the corollary consumes is the
Schmidt-basis form with Bob's operators the transposes of Alice's, which swap symmetry alone
does not provide --- the approximate Takagi bridge of Remark~\ref{rem:lidt-formalized} is what
supplies it. The auxiliary measurements are compressed back to the original spaces
and projectivized there, using Theorem~\ref{thm:orthonormalization}; projectivity is
not a conclusion of the transfer corollary itself. The consistency estimates then
give the point, line and full-codeword conclusions. These are steps of the paper's
reviewed proof, not consequences already checked by the canonical-line Lean theorem.
\end{proof}

\begin{lemma}[The padded multi-codeword game adapter]
\label{lem:lidt-ldc-adapter}
\ledgernode{1.2.1.9.2}

direction index $j\leq z_0+1$ has weights proportional to $q^{j-1}$, with
uniform seed within each fibre. Independent choices for both players and both
line types are fixed by averaging the complete game failure, including equality
subtests. Choosing an arbitrary representative seed is not justified.

\begin{lemma}[Extracting exactly linear combined codewords]
\label{lem:lidt-ldc-extraction}

polynomial measurements to outcomes of the form

legal tuple. This gives projective measurements of the complete tuple

linear in $x$, the chance over $y$ that its restriction is exactly linear is
at most $dm/q$. It does not select independent single-codeword measurements.

$r_X, r_Z \in \F_q$ uniformly. The edges of $G^\pauli$ are: a self-loop at every type;
$(\mathtt{ALine}, W)$ and $(\mathtt{DLine}, W)$ each to $(\mathtt{Point}, W)$;
$(\mathtt{Point}, W)$ to $(\pauli, W)$; $(\mathtt{Point}, W)$ to $(\mathtt{Pair}, W)$ to
$\mathtt{Pair}$; $(\mathtt{Point}, X)$ to $\mathtt{Variable}_1$ and $(\mathtt{Point}, Z)$ to
$\mathtt{Variable}_5$; and the eighteen incidences of the Magic Square, $\mathtt{Constraint}_i$
to $\mathtt{Variable}_j$ for $j$ in the support of equation $i$. So there are thirty non-loop
edges and eighty-six ordered pairs, each of weight $1/86$.

\emph{Question content.} A type $(\mathtt{Point}, W)$ receives $u_W$; $(\mathtt{ALine}, W)$
receives $(u_0, s)$ with $u_0 = L^{\mathrm{Ln}}_{e_i}(u_W)$ and $i = \chi(s)$;
$(\mathtt{DLine}, W)$ receives $(u_0, s, v')$ with $v' = \pi_{i-1}(v)$ and
$u_0 = L^{\mathrm{Ln}}_{v'}(u_W)$ --- all three exactly as in Definition~\ref{def:lidt-cl};
$(\pauli, W)$ receives nothing; and each of $\mathtt{Pair}$, $(\mathtt{Pair}, W)$,
$\mathtt{Constraint}_i$ and $\mathtt{Variable}_j$ receives $\omega$.

\emph{Answer format.} An element of $\F_q$ for $(\mathtt{Point}, W)$; a polynomial of degree at
most $d$ for $(\mathtt{ALine}, W)$ and at most $md$ for $(\mathtt{DLine}, W)$; an element of
$\F_q^M$ for $(\pauli, W)$; a pair $(\beta_X, \beta_Z) \in \F_2^2$ for $\mathtt{Pair}$; a
single bit for $(\mathtt{Pair}, W)$ and for $\mathtt{Variable}_j$; and a triple
$(\alpha_{v_1}, \alpha_{v_2}, \alpha_{v_3}) \in \F_2^3$, indexed by the three variables of the
equation in order, for $\mathtt{Constraint}_i$.

\emph{Decision.} The verifier first checks that both answers have the format their question's
type prescribes, and rejects if not. It then applies, with $\gamma(\omega)$ as in
Lemma~\ref{fact:omega-anticomm-prob} and $g_h$ the low-degree encoding of
Definition~\ref{def:ld-encoding}:
\begin{enumerate}
\item equal types: accept iff the two answers agree;
\item $(\mathtt{Point}, W)$ against $(\mathtt{ALine}, W)$ or $(\mathtt{DLine}, W)$: accept iff
  the seeded CL decider of Definition~\ref{def:lidt-cl} at $(q, m, d, 1)$ accepts;
\item $(\mathtt{Point}, W)$ against $(\pauli, W)$: accept iff $g_{a_{\bar w}}(x_w) = a_w$;
\item $(\mathtt{Pair}, W)$ against $\mathtt{Pair}$: accept iff $a_w = \beta_W$ or
  $\gamma(\omega) \neq 0$;
\item $(\mathtt{Point}, W)$ against $(\mathtt{Pair}, W)$: accept iff
  $\fieldtr(a_w r_W) = a_{\bar w}$ or $\gamma(\omega) \neq 0$;
\item $\mathtt{Constraint}_i$ against $\mathtt{Variable}_j$: accept iff $\gamma(\omega) = 0$,
  or $j$ occurs in equation $i$, the triple $a_w$ satisfies that equation, and its entry at
  $j$ is $a_{\bar w}$;
\item $(\mathtt{Point}, W)$ against $\mathtt{Variable}_j$: accept iff $\gamma(\omega) = 0$, or
  $j = 1$, $W = X$ and $\fieldtr(a_w r_X) = a_{\bar w}$, or $j = 5$, $W = Z$ and
  $\fieldtr(a_w r_Z) = a_{\bar w}$;
\end{enumerate}
each in both orientations, and accepts on any type pair no rule names.
\end{definition}

\paragraph{Source.} The pinned paper~\cite{AuditSync26}, \texttt{ldt.tex}: the type set is
\texttt{eq:pauli-type}, the sampling procedure and type graph \texttt{sec:qld-game} and
\texttt{fig:type-graph-pauli}, and the decision procedure \texttt{fig:decider\_pauli}. No
definition of this game appeared in this blueprint before; Theorem~\ref{thm:qld} and
Lemma~\ref{lem:qld-win} were statements about an undefined object, which is why this is the
campaign's first definitional obligation.

\paragraph{Comments.} The formalization is \texttt{MIPRE.QLD.qldGame}
(\texttt{MIPRE/Background/QLD/Game.lean}). Five points, four of them the consequences of

The Magic Square indices are the layout's own, $0$-indexed

\begin{theorem}[Pauli basis test / quantum low individual degree test]
\label{thm:qld}
\ledgernode{1.2.2, 1.2.2.17}
\uses{def:qld-game,def:q-value,def:distance,lem:ms-direct-anticomm,thm:lidt-cl-soundness}
There exists a function
\begin{equation*}
  \delta_{qld}(\eps, m, d, q) = a (md)^a \bigl( \eps^b + q^{-b} + 2^{-bmd} \bigr)
\end{equation*}
for universal constants $a \geq 1$ and $0 < b < 1$ such that the following holds. For
all admissible parameter tuples $\qldparams = (q, m, d)$ and all strategies
$\strategy = (\ket{\psi}, A, B)$ for the game $\game^{\pauli}_{\qldparams}$ that
succeed with probability at least $1 - \eps$, there exist local isometries
$\phi_\alice, \phi_\bob$ mapping into $\mH' \ot (\C^q)^{\ot M}$ with $M = 2^m$, and a
state $\ket{\aux}$, such that
$\norm{\phi_\alice \ot \phi_\bob \ket{\psi} - \ket{\aux} \ot \ket{\EPR_q}^{\ot M}}
\leq \delta_{qld}$, and for $W \in \{X, Z\}$ the measurements for the questions
$(\pauli, W)$ are $\delta_{qld}$-close to the Pauli basis measurements $\tau^W_u$, $u
\in \F_q^M$, acting on $\ket{\EPR_q}^{\ot M}$.
\end{theorem}

\paragraph{Source.} The pinned paper~\cite{AuditSync26}, \texttt{qld-appendix.tex}
and its five proof files, states this Pauli theorem and its reduction to the seeded
CL test. The reduction follows the architecture of~\cite[Section 7]{JNVWY20};
the canonical-line soundness theorem of~\cite{JNVWY21qld} is related background.
Lemma~\ref{lem:ms-direct-anticomm} supplies the base anticommutation directly.

\paragraph{Comments.} This is a substantial remaining formalization target and the
workhorse of question reduction: introspection certifies the players' sampling of
their own questions against Pauli basis measurements. The reduction uses direct
Magic Square anticommutation, linearity and orthonormalization, together with the
seeded CL theorem. The available canonical-line theorem can replace the paper's
classical route through Remark~\ref{rem:lidt-cl-adapter}'s adapter, which supplies the
$\mathrm{ldc}=1$ case this reduction uses (Theorem~\ref{thm:lidt-cl-soundness-one}).

The theorem's bipartite statement matches the pipeline's $\valstar$ soundness
interface, so its uses require no additional synchronous-value transport. The verification
campaign~\cite{Audit26} further reports three corrections in the tensor-code material the
proof passes through --- $\kappa$-convexity requires $c_1 \leq 1$, the synchronicity
defect obeys $\delta_{\mathrm{sync}} \leq 4\eps$ rather than $3\eps$, and one corner block
pairs with a transpose --- none checked here, none affecting the statement above, all
relevant to whoever writes its proof.


\subsubsection{The shape of the soundness proof}

\begin{remark}[How the soundness proof of Theorem~\ref{thm:qld} is organized]
\label{rem:qld-architecture}
\uses{thm:qld,thm:lidt-cl-soundness,rem:lidt-cl-adapter,thm:orthonormalization,lem:ms-direct-anticomm}
The soundness half of Theorem~\ref{thm:qld} is developed in the five-file appendix of
the pinned paper~\cite{AuditSync26}. Its
architecture is worth stating before its pieces, because the pieces are unreadable without it.
There are five stages.
\begin{enumerate}
\item \textbf{Preliminaries}: three utility lemmas about closeness, one of which
  (Corollary~\ref{cor:ortho-from-consistency}) is used on almost every page.
\item \textbf{Win implications}: unpack ``succeeds with probability $1 - \eps$'' into one
  consistency or commutation statement per subtest (Lemma~\ref{lem:qld-win}).
\item \textbf{Expansion}: adjoin EPR ancillas and replace the strategy's measurements by ones
  on the enlarged space that approximately commute across the two bases
  (Lemmas~\ref{lem:qld-expanded-points} and~\ref{lem:qld-expanded-lines}).
\item \textbf{Combining}: build a \emph{single} measurement returning both the $X$- and the
  $Z$-polynomial, by padding the ambient space and applying seeded CL soundness to
  the padded strategy (Lemma~\ref{lem:qld-simultaneous}).
\item \textbf{Separation and the swap isometry}: extract exact Pauli operators from that
  single measurement and rotate them onto honest EPR pairs
  (Lemmas~\ref{lem:qld-exact-paulis} and~\ref{lem:qld-swap}).
\end{enumerate}
Stage 4 consumes Theorem~\ref{thm:lidt-cl-soundness} at $(q,4m,d,1)$, after
constructing a strategy for the seeded CL game and projectivizing its line POVMs.
Theorem~\ref{thm:lidt-soundness} justifies this application through
Theorem~\ref{thm:lidt-cl-soundness-one}, the $\mathrm{ldc}=1$ case being exactly what stage 4
consumes; what stage 4 must still supply is the padding itself. Stage 5 recovers the Pauli group
relations exactly, which permits the standard swap isometry; the expanded
measurements in stage 3 satisfy only approximate commutation.
\end{remark}

\begin{lemma}[Anticommuting question tuples]
\label{fact:omega-anticomm-prob}
\uses{def:ld-encoding,lem:schwartz-zippel}
\lean{MIPRE.LowDegree.indVec, MIPRE.LowDegree.indPair, MIPRE.LowDegree.indPair_eq_eval,
  MIPRE.LowDegree.acGamma, MIPRE.LowDegree.acTuples, MIPRE.LowDegree.cTuples,
  MIPRE.LowDegree.card_acTuples_add_card_cTuples,
  MIPRE.LowDegree.prob_anticommuting_ge, MIPRE.LowDegree.prob_commuting_ge}
\leanok
Let $\ind_m(x) \in \F_q^M$, $M = 2^m$, be the vector $(\ind_{m,y}(x))_y$ of the indicator
polynomials of Definition~\ref{def:ld-encoding} evaluated at a point $x \in \F_q^m$, and call
a tuple $\omega = (u_X, u_Z, r_X, r_Z) \in (\F_q^m)^2 \times \F_q^2$ \emph{anticommuting} if
\begin{equation*}
  \gamma(\omega) \;=\; \fieldtr_{q \to 2}\bigl( (r_X \cdot \ind_m(u_X)) \cdot
    (r_Z \cdot \ind_m(u_Z)) \bigr) \;\neq\; 0\;,
\end{equation*}
the inner product being the one of $\F_q^M$, and \emph{commuting} otherwise. The two cases
partition the sample space. If $m \geq 1$ and $d \geq 1$ then a uniformly random $\omega$ is
anticommuting with probability at least $(1 - 3md/q)/2$, and commuting with probability at
least $(1 - 3md/q)/2$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:schwartz-zippel}
Because $\ind_m(x) \cdot \ind_m(z) = g_{\ind_m(x)}(z)$ --- one low-degree encoding evaluated at
the other point --- $\gamma(\omega) = \fieldtr_{q \to 2}(r_X r_Z \cdot g_{\ind_m(u_X)}(u_Z))$.
The vector $\ind_m(x)$ has coordinate sum $\prod_i (x_i + (1 - x_i)) = 1$, so it is never zero
and $g_{\ind_m(u_X)}$ is a nonzero multilinear polynomial; by
Lemma~\ref{lem:schwartz-zippel} it vanishes at no more than an $m/q$ fraction of the $u_Z$.
Fix $u_X, u_Z$ with $g_{\ind_m(u_X)}(u_Z) \neq 0$ and $r_X \neq 0$, and put
$c = r_X\, g_{\ind_m(u_X)}(u_Z) \neq 0$. Then $r \mapsto \fieldtr_{q \to 2}(c r)$ is a nonzero
$\F_2$-linear functional on $\F_q$, so its kernel is a hyperplane and exactly $q/2$ of the
$r_Z$ give $\gamma \neq 0$, the other $q/2$ giving $\gamma = 0$. Multiplying the three
independent estimates gives $(1 - 1/q)(1 - m/q)/2 \geq (1 - (m+1)/q)/2$ for the anticommuting
case; the commuting case gains the whole $r_X = 0$ slice and the whole $g_{\ind_m(u_X)}(u_Z) =
0$ slice, so $(1 - 1/q)/2$ holds for it with no $m$ at all. Finally $m + 1 \leq 3md$ and
$1 \leq 3md$ when $m, d \geq 1$.
\end{proof}

\paragraph{Comments.} Two departures from the paper's statement and proof, both recorded in
the formalization's own docstring. First, the paper states the bound as
$(1 - q^{-m})(1 - q^{-1})(1 - md/q)/2$; its own correction note removes the case distinction on
$u_X = 0$ that produced the first factor, observing that $\ind_m(u_X)$ is never the zero
vector, and keeps the factor only because it remains a valid lower bound. It is dropped here.
Second, the low-degree encoding is \emph{multilinear}, so Schwartz--Zippel costs $m/q$ and not
$md/q$; the formalized bounds are $(1 - (m+1)/q)/2$ and $(1 - 1/q)/2$, and the paper's
$(1 - 3md/q)/2$ is derived from them for $m, d \geq 1$. The `$d$' in this fact is therefore
inert: it is the individual-degree parameter of the surrounding test, and the bound does not
depend on it.

The complementarity statement is not decoration. Each of the two bounds is a lower bound on a
probability, so either alone would be implied by the wrong definition of its event; that they
hold for complementary events, and add to one, is what pins them down.

\begin{lemma}[Averaging preserves closeness]
\label{lem:qld-averaging}
\ledgernode{1.2.2.1}
\uses{def:state-distance}
\lean{MIPRE.stateSqNorm_avg_le}
\leanok
Let $A^x \approx_\delta B^x$ on average over $x \sim \mu$, and let $\alpha_x$ be coefficients
with $\abs{\alpha_x} \leq 1$. Then $\E_x \alpha_x A^x \approx_\delta \E_x \alpha_x B^x$.
\end{lemma}

\begin{proof}
\leanok
Triangle inequality and Jensen. Note the hypothesis is $\abs{\alpha_x} \leq 1$, not
$\abs{\alpha_x} = 1$: the lemma is used with genuinely contractive coefficients. The
$\delta$ is preserved exactly, with no constant: the triangle inequality gives
$\norm{\E_x \alpha_x (A^x - B^x)\ket{\psi}} \leq \E_x \norm{(A^x - B^x)\ket{\psi}}$ and Jensen
for the square root gives $\E_x \sqrt{f(x)} \leq \sqrt{\E_x f(x)}$, which is the
$\mu$-probability case of \texttt{MIPRE.sum\_mul\_sqrt\_le}.
\end{proof}

\begin{lemma}[From POVM elements to generalized observables]
\label{lem:qld-povm-to-obs}
\ledgernode{1.2.2.2}
\uses{def:state-distance,lem:qld-averaging}
\lean{MIPRE.obsOf, MIPRE.stateDist_obsOf_le}
\leanok
Let $\{A^x_a\}$ and $\{B^x_a\}$ be POVMs with a common outcome set $\mA$, and let $\alpha_a$
be coefficients of unit modulus. If $A^x_a \approx_\delta B^x_a$ then the generalized
observables $A^x = \sum_a \alpha_a A^x_a$ and $B^x = \sum_a \alpha_a B^x_a$ satisfy
$A^x \approx_{\abs{\mA} \delta} B^x$.
\end{lemma}

\begin{proof}
\leanok
Triangle inequality over the outcomes, then Cauchy--Schwarz against the constant $1$:
$(\sum_a t_a)^2 \leq \abs{\mA} \sum_a t_a^2$. Unit modulus is more than the proof uses; the
formalized hypothesis is $\abs{\alpha_a} \leq 1$, and the constant $\abs{\mA}$ is exact rather
than an $O(\cdot)$.
\end{proof}

\begin{corollary}[Orthonormalization from consistency]
\label{cor:ortho-from-consistency}
\ledgernode{1.2.2.3, 1.2.2.3.1}
\uses{thm:orthonormalization,def:inconsistency,def:state-distance}
\lean{MIPRE.QLD.exists_projective_of_consistent}
\leanok
Let $\delta > 0$, let $\{Q_a\}$ be a POVM on $\mH_\alice$ and $\{R_a\}$ a POVM on $\mH_\bob$
whose inconsistency on the unit vector $\ket{\psi}$ is at most $\delta$. Then there is a
\emph{projective} measurement $\{P_a\}$ on $\mH_\alice$ with
\begin{equation*}
  \sum_a \norm{\bigl((Q_a - P_a) \ot \Id_\bob\bigr)\ket{\psi}}^2 \;<\; 18\,\delta\;,
\end{equation*}
that is, $P_a \ot \Id \approx_{\eta_{ortho}(\delta)} Q_a \ot \Id$ on $\ket{\psi}$ with
$\eta_{ortho}(\delta) = 18\delta$, linear in $\delta$ as
Theorem~\ref{thm:orthonormalization} permits.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:orthonormalization}
This is not Theorem~\ref{thm:orthonormalization} restated: that theorem requires strict
near-projectivity in a normal state, while what the appendix has is
bipartite consistency. The bridge is a Cauchy--Schwarz step. Write
$\gamma = \sum_{a \neq b} \bra{\psi} Q_a \ot R_b \ket{\psi}$, so that
$\sum_a \bra{\psi} Q_a \ot R_a \ket{\psi} = 1 - \gamma$ because
$\sum_{a,b} Q_a \ot R_b = \Id$. Since $\sum_a R_a^2 \leq \sum_a R_a = \Id$, Cauchy--Schwarz
over $a$ gives
\begin{equation*}
  1 - \gamma \;=\; \sum_a \bigl\langle (Q_a \ot \Id)\ket{\psi},\,
    (\Id \ot R_a)\ket{\psi} \bigr\rangle
  \;\leq\; \Bigl(\sum_a \bra{\psi} Q_a^2 \ot \Id \ket{\psi}\Bigr)^{1/2}\;,
\end{equation*}
hence $\sum_a \bra{\psi} Q_a^2 \ot \Id \ket{\psi} \geq (1 - \gamma)^2 = 1 - 2\gamma + \gamma^2$.
Now $2\gamma - \gamma^2 < 2\delta$ for every $\gamma \leq \delta$ with $\delta > 0$ --- at
$\gamma = \delta$ because $\gamma^2 > 0$, and at $\gamma < \delta$ because $\gamma^2 \geq 0$ ---
and if $\gamma \geq 1$ the hypothesis holds vacuously, so near-projectivity at $\eps = 2\delta$
holds \emph{strictly}. Apply Theorem~\ref{thm:orthonormalization} to
$\cal{L}(\mH_\alice)\ot\Id$ with the vector state of $\ket{\psi}$: it gives a projective family
$P_a \ot \Id$ with squared-distance sum less than $9\eps = 18\delta$.
\end{proof}

\paragraph{Comments.} The paper closes this proof differently, and the difference is a
\textbf{repair}. It derives near-projectivity at $2\gamma$ for the \emph{actual} inconsistency
$\gamma$, which is non-strict, so Theorem~\ref{thm:orthonormalization} applies only at each
$\eps > 2\gamma$; it then takes $\eps \downarrow 2\gamma$ inside the (compact) set of projective
measurements with a fixed finite outcome set and extracts a convergent subsequence to reach
$18\gamma$ itself. That limit is not needed. The slack $\gamma^2$ in
$(1-\gamma)^2 = 1 - 2\gamma + \gamma^2$ already makes the hypothesis strict at $\eps = 2\delta$
for any \emph{bound} $\delta > 0$ on the inconsistency, which is the form in which every
application supplies it and the form stated above; the compactness argument, and with it the
continuity of $\{P_a\} \mapsto \sum_a \norm{((Q_a - P_a)\ot\Id)\ket{\psi}}^2$ and the
sequential extraction, disappears. What the paper's version additionally gives is the exact
case $\delta = 0$, where the limit produces a projective measurement agreeing with $\{Q_a\}$ on
$\ket{\psi}$ exactly. Nothing downstream uses it: every invocation carries a positive error
bound. The statement above is therefore weaker only at $\delta = 0$ and stronger elsewhere,
both in being strict and in naming the constant.

\label{lem:qld-win}

  \end{equation*}
\item (\textbf{Magic square consistency}) on average over anticommuting $\omega$,
  $M^{(\mathtt{Point}, X), u_X}_{[\fieldtr(\cdot\, r_X) = b]} \ot \Id \simeq_{172\eps}
    \Id \ot M^{\mathtt{Variable}_1, \omega}_b$, and the same with $Z$, $r_Z$ and
  $\mathtt{Variable}_5$.
\end{enumerate}
\end{lemma}

\begin{proof}
\leanok
\uses{lem:ms-direct-anticomm,def:cross-distance}
Items 1--5 and 7 are one lemma applied seven times. The question distribution factors as an
ordered type edge times the content, both uniform, so a question pair arising from a fixed edge
of the type graph carries at least $1/86$ of the mass --- $86$ being the ordered pairs counted in
Definition~\ref{def:qld-game} --- and the conditional failure of any subtest is therefore at
most $86\eps$. Each rule of $\decider^\pauli$ accepts only when two named readings of the two
answers agree, so~\eqref{eq:cross-from-agreement} converts that into $172\eps$ of cross-party
distance for the measurements coarse-grained along those readings. The four rules gated on
$\gamma$ check nothing on the other value of $\gamma$, which is why their averages are
restricted.

Item 6 is not of that shape. Relabelling the $\mathtt{Constraint}_i$ and $\mathtt{Variable}_j$
answers at a fixed $\omega$ into the Magic Square game's alphabet --- the three bits placed at
the constraint's cells, the bit passed through --- gives a strategy for $\game^\MS$, and
relabelling outcomes is data processing, so that strategy wins whenever the Pauli test's own rule
would have accepted. Its failure is therefore at most the Magic Square game's own
question-weighted sum of the Pauli test's conditional failures at the corresponding question
pairs, which is $86\eps$ on average over anticommuting $\omega$ --- one factor of $86$ and not
thirty-six, because $\game^\MS$'s distribution is itself normalized. Lemma~\ref{lem:ms-direct-anticomm}
applied to each such conditional strategy and averaged gives $186624 \cdot 86 = 16049664$. This
uses the strategy's own observables and invokes no Magic Square rigidity.
\end{proof}

\paragraph{Comments.} Three departures from the paper's statement
(\cite{AuditSync26}, \texttt{qld-commutation.tex}, \texttt{lem:qld-win-implications}), and one
from this blueprint's own earlier text.

\textbf{The analysis is run on Bob's observables and carried back.}
Lemma~\ref{lem:commutation-analysis} wants the two commuting families on one side and the joint

remark following it gives a counterexample for $\approx_\delta$. The paper's

corollary needs a \emph{strictly positive} bound $\delta$ on the inconsistency. Here it is applied

averaging as an additive constant and is sent to zero at the end
(\texttt{le\_of\_forall\_pos\_le\_add}). No compactness is needed, which is the same observation the
corollary's own file records.

$(a+b+c)^2 \leq 3(a^2+b^2+c^2)$: the constant is

strategy}'s observables, which are the same operators as the $\mathtt{Variable}$ observables here:
the conditional strategy at a $\mathtt{Variable}$ question is the Pauli test's measurement at the

\paragraph{Comments.} Formalized in \texttt{MIPRE/Background/QLD/Commutation.lean}.

\textbf{This half costs $\eps$, not $\sqrt{\eps}$.} Every input above is a squared-norm bound
linear in $\eps$, and the only inequalities used are the triangle inequality and
$(a+b+c)^2 \leq 3(a^2+b^2+c^2)$, so nothing introduces a square root. \emph{Correction to an
earlier reading of this:} the square root in~\eqref{eq:qld-obs-commutation-bp} as the paper writes
it does \emph{not} belong to the commuting case either --- that case is $O(\eps)$ too
(Lemma~\ref{lem:qld-obs-commutation-comm}), since \eqref{eq:commutation-analysis-bp} is linear in
its hypothesis and Corollary~\ref{cor:ortho-from-consistency} is linear in the inconsistency. The
paper's $\sqrt{\eps}$ is a uniform \emph{weakening}, and its own text says so where the
anticommuting case is proved: ``direct averaging proves the anticommutation relation with squared
error $O(\eps)$. In particular it implies the following weaker bound, retained in the form used
below''. Nothing downstream needs the sharper form --- \texttt{lem:qld-comm-cons} asks only for
$\poly(\eps)$ --- so this is a note about where a square root is \emph{not} spent, not a repair.

\textbf{The sign is the double reversal.} Both order reversals run in the same direction --- a
product transferred across the tensor factors comes back reversed --- and what sits between them
is Bob's \emph{anti}commutator, a sum. So the two reversals do not cancel: Alice's two observables
anticommute because Bob's do, and that is the $(-1)^{\gamma}$ the expansion stage of
Lemma~\ref{lem:qld-expanded-points} cancels against the ancillas.

\begin{lemma}[Expanded point measurements]
\label{lem:qld-expanded-points}

\leanok
\uses{lem:qld-win,lem:qld-expanded-commutation,lem:qld-expanded-consistency}

\begin{theorem}[Exact linearity from approximate linearity]
\label{thm:linearity}
\ledgernode{1.2.2.7, 1.2.2.7.1, 1.2.2.7.2, 1.2.2.7.3}
\uses{def:state-distance,def:generalized-pauli,lem:naimark-dilation}
\lean{MIPRE.fourierOf, MIPRE.fourierOf_conjTranspose, MIPRE.fourierOf_posSemidef,
  MIPRE.sum_sgn_smul_fourierOf_sq, MIPRE.sum_fourierOf_sq, MIPRE.exists_exactly_linear,
  MIPRE.extVecA, MIPRE.ancillaEmbed_kron_isometry, MIPRE.norm_evec_extVecA,
  MIPRE.kron_one_mul_ancillaEmbed, MIPRE.ancillaEmbed_conjTranspose_mul_kron_one,
  MIPRE.compress_mul_kron_one, MIPRE.compress_kron_one_mul, MIPRE.qform_extVecA,
  MIPRE.stateSqNorm_sub_of_isometry, MIPRE.conjTranspose_mul_self_of_involution,
  MIPRE.exists_exactly_linear_close}
\leanok
Let $\{O^u\}_{u \in V}$, $V = \F_q^n$, be observables on the first party's space --- self-adjoint
involutions --- and let $\ket{\psi}$ be a state. Then, on the space extended by an ancilla register
$\C^V$ on the \emph{first party only}, with the extended state
$\ket{\psi'} = \ket{\psi} \ot \ket{0}$, there are observables $\{L^u\}$ that are \emph{exactly}
linear,
\begin{equation*}
  L^u L^{u'} \;=\; L^{u + u'}\;, \qquad L^0 \;=\; \Id\;,
\end{equation*}
such that the average squared distance to the original family is \emph{equal} to the average
linearity defect:
\begin{equation*}
  \E_{u \in V} \norm{\bigl(L^u - O^u \ot \Id\bigr) \ket{\psi'}}^2
    \;=\; \E_{a, b \in V} \norm{\bigl(O^a O^b - O^{a + b}\bigr) \ket{\psi}}^2\;.
\end{equation*}
In particular, if the family is $\delta$-approximately linear on average then
$L^u \approx_\delta O^u$ on average over $u$: the error is linear in $\delta$, with no loss in the
exponent and no change of constant.
\end{theorem}

\begin{proof}
\leanok
\uses{def:generalized-pauli,lem:naimark-dilation}
The Fourier argument, in three exact steps and one identity.

\textbf{Parseval.} Let $A_e = \abs{V}^{-1} \sum_a (-1)^{\fieldtr(a \cdot e)} O^a$ be the Fourier
transform of the family against the characters of the trace form
(Definition~\ref{def:generalized-pauli}). Merging the characters and summing over $e$ collapses the
double sum, because $\sum_e (-1)^{\fieldtr(c \cdot e)}$ is $\abs{V}$ at $c = 0$ and $0$ otherwise:
\begin{equation}
  \label{eq:linearity-character-sum}
  \sum_e (-1)^{\fieldtr(u \cdot e)} A_e^2 \;=\; \abs{V}^{-1} \sum_a O^a O^{a + u}\;.
\end{equation}
At $u = 0$ the right-hand side is $\abs{V}^{-1} \sum_a (O^a)^2 = \Id$, so $\{A_e^2\}$ is a POVM ---
\emph{exactly}, with no appeal to the approximate linearity. That is the only place the involution
property is used.

\textbf{Naimark.} Dilate that POVM to a projective measurement $\{P_e\}$ on
$\C^{d} \ot \C^{V}$, compressed back by the isometry $\ket{v} \mapsto \ket{v} \ot \ket{0}$
(Lemma~\ref{lem:naimark-dilation}). One isometry serves the whole family of families, which is what
lets the estimate be taken against the single state $\ket{\psi'}$.

\textbf{Inversion.} Put $L^u = \sum_e (-1)^{\fieldtr(u \cdot e)} P_e$. Mutual orthogonality of the
$P_e$ gives $L^u L^{u'} = L^{u + u'}$ and $L^0 = \sum_e P_e = \Id$, and the compression of $L^u$ is
the left-hand side of~\eqref{eq:linearity-character-sum}, hence $\abs{V}^{-1} \sum_a O^a O^{a+u}$.

\textbf{The identity.} Expanding both squared distances --- each is $2$ minus the two orders of an
overlap, the operators involved being unitary --- and compressing, the $u$-th term on the left is
the average over $a$ of the $(a, a+u)$-th term on the right, after reindexing that average by
$a \mapsto a + u$. Summing over $u$ and reindexing once more, by $u \mapsto a + u$ for each fixed
$a$, gives the stated equality. No inequality is used anywhere, which is why the constant is
exactly $1$.
\end{proof}

\end{equation*}

(\widehat{M}^{X, x}_a \widehat{M}^{Z, z}_b)$, hence positive. No approximate commutation is used
here, and no orthonormalization is needed afterwards: Lemma~\ref{lem:naimark-dilation} dilates

Lemma~\ref{lem:qld-expanded-points} bounds is the commutator of the \emph{observables}
$\widehat{W}^{r}(u)$, one pair per probe; what the sandwich needs is the commutator of the
measurement \emph{elements}, one pair per outcome. The elements are the two-variable Fourier
transform of the observables against the characters of the trace form
(Definition~\ref{def:generalized-pauli}), so by Parseval the sum over outcomes of the one is the
average over probes of the other --- and the content already averages over the probes uniformly
and independently of everything else it carries, so that average is free.

\begin{proof}
\leanok

This is the pasting lemma (Lemma~\ref{lem:pasting-updated}) at
$G_1 = \widehat{M}^{(\mathtt{Line}, Z), \ell_Z}$, $G_2 = \widehat{M}^{(\mathtt{Line}, X),
\ell_X}$ and $A = \widehat{Q}^{x,z}$, and what has to be supplied is its four hypotheses.

\textbf{The two marginal consistencies.} Lemma~\ref{lem:qld-combined-points} gives the joint
measurement against the \emph{ordered product} of the two point measurements, in either order; the
marginal step of Lemma~\ref{lem:cool-closeness-fact} turns each of those into the corresponding
marginal against a single point measurement, at ten times the input. Replacing that point
measurement by the line measurement evaluated at the point is one triangle inequality through
Bob's own point measurement --- whose two legs are item 1 of Lemma~\ref{lem:qld-expanded-points}
(Alice's copy against Bob's) and item 2 of Lemma~\ref{lem:qld-expanded-lines} (Alice's point
against Bob's line).

\textbf{The fine-grained self-consistency of the $X$-line measurement} is item 1 of
Lemma~\ref{lem:qld-expanded-lines} verbatim. The joint measurement lives on each party's space
enlarged by one $\F_q \times \F_q$ register and the line measurements do not, so both inputs are
transported by the fact that an operator with an inert ancilla has the same cross-party deviation
on the extended state as the operator itself.

\textbf{The collision term} needs the point to be uniform on its line once the line is fixed. The
content determines both, so the factorization is produced by \emph{shifting}: for a fixed
$t \in \F_q$ the map that replaces $u_W$ by $u_W + t \cdot w_W$ is a bijection of contents, so
averaging over contents and over (content, shift) pairs give the same thing; the line's question,
base point and direction are invariant under it; and the point's parameter moves by exactly $t$.
Two distinct degree-$md$ polynomials then agree at at most $md$ of the $q$ parameters, which is the
univariate case of Lemma~\ref{lem:schwartz-zippel}.
\end{proof}

\paragraph{Comments.} Formalized in \texttt{MIPRE/Background/QLD/Lines.lean}, on top of
Lemma~\ref{lem:pasting-updated} in \texttt{MIPRE/Foundations/Pasting.lean}. The headline
declaration \texttt{qld\_pairs\_of\_lines} takes only game-level hypotheses --- a strategy, its

\texttt{aPres\_items}, \texttt{dPres\_items} and \texttt{sum\_content\_collProb\_dPres} are what
the other three pairs of line types need.

\textbf{The question distribution is a parameter.} \texttt{pairs\_of\_lines\_gen} proves the lemma
over any finite nonempty sample space carrying a content per side --- $k_X$ presenting the $X$ line,
$k_Z$ the $Z$ line --- together with a shift of the $X$ point that both contents see as such; the
two contents need share nothing, because no hypothesis of Lemma~\ref{lem:pasting-updated} involves
both lines. \texttt{pairs\_of\_lines} is its instance at one content per sample, the same one on
both sides, where the two lines are therefore dependent;
\texttt{pairs\_of\_lines\_prod} (Lemma~\ref{lem:qld-padded-lines}) is its instance at a pair of
line--point data, where they are independent. \texttt{pasteLine} and
\texttt{pasteJ\_eq\_pasteLine} accordingly take the $X$ line and the $Z$ line from two contents
rather than one.

\textbf{Per pair of line types, not for the mixture.} The paper states the lemma on average over
the line--point distribution, which is an equal mixture of the axis-parallel and diagonal
branches. Here the line type is a parameter and the statement is proved for each of the four
pairs, which is \emph{stronger}: the mixture is a convex combination of the four.

\textbf{A degenerate diagonal line makes the collision probability one, and the paper's
$\eps = md/q$ does not account for it.} The seeded test's diagonal direction is
$\mathtt{zeroBelow}(\chi(s), v)$, which vanishes for a $1/q$ fraction of the contents at worst; on
such a content the ``line'' is a single point, every polynomial is evaluated at the same parameter,
and two distinct ones can agree with probability $1$. The bound above therefore carries
$md/q + 1/q$ on the diagonal side, and $\delta_P$ is still $\poly(\eps, md/q)$. See
\texttt{reports/pasting-degenerate-diagonal-line.md}.

\textbf{The shift replaces the paper's appeal to \texttt{lem:alnf} and \texttt{lem:dlnf}.} Those
say that conditioned on the line the point is uniform on it, which is what the collision bound
uses. Rather than form the conditional distribution, the formalization shifts the point along the
direction and uses that the shift is a bijection of contents for each fixed $t$
(\texttt{sum\_content\_shift\_gen}); the same device, applied to the raw direction $v$ instead of
the point, is what bounds the probability of a degenerate diagonal line.

\textbf{What the refined contract still asks for.} Three items of the contract recorded for this
lemma are not part of what is proved above, and are wanted by the lemmas that consume it: that the
outcome polynomials have degree at most $d$ on an axis-parallel component, which is
Lemma~\ref{lem:qld-axis-degree} and an additional target there; and the restriction to
specified line types and direction indices, which costs the reciprocal probability of those
choices, at most $4m^2$. That the two lines be drawn \emph{independently} is no longer one of them:
\texttt{pairs\_of\_lines\_gen} leaves the question distribution free, so the padded line--point
distribution of Lemma~\ref{lem:qld-padded-lines} is an instance rather than a gap. The version above fixes the line type on each side as a parameter and is
proved for each of the four pairs, so the restriction factor is not incurred; the reciprocal cost
appears only when a caller wants the statement for the mixture.

\begin{lemma}[Padded point measurements]
\label{lem:qld-padded-points}
\ledgernode{1.2.2.10}
\uses{lem:qld-combined-points,def:generalized-pauli,def:distance}

Write a point of the padded space $\F_q^{4m}$ as $u = (x, z, \alpha, \beta, w)$ with
$x, z \in \F_q^m$, $\alpha, \beta \in \F_q$ and $w \in \F_q^{2m - 2}$ the dummy coordinates.
With $\delta_Q$ as in Lemma~\ref{lem:qld-combined-points}, put
\begin{equation*}
  \widehat{Q}^u_c \;=\; \widehat{Q}^{x, z, \alpha, \beta}_c
    \;=\; \sum_{\substack{a, b \,:\\ \alpha a + \beta b = c}} \widehat{Q}^{x, z}_{a, b}\;,
\end{equation*}
which returns the combined value $\alpha g_X(x) + \beta g_Z(z)$ and depends on $u$ only through
$(x, z, \alpha, \beta)$ --- that is what ``independent of the dummy coordinates'' means. Then
$\{\widehat{Q}^{u}_c\}_{c \in \F_q}$ is \emph{projective}, and
\begin{enumerate}
\item (\textbf{Self-consistency}) for \emph{every} $(\alpha, \beta)$, on average over the
  content, $\widehat{Q}^{u}_c \ot \Id \simeq_{2\delta_Q} \Id \ot \widehat{Q}^{u}_c$;
\item (\textbf{Consistency with $\widehat{M}$}) on average over the content \emph{and} over
  uniform $(\alpha, \beta)$,
  \begin{gather*}
    \widehat{Q}^{u}_c \ot \Id \;\simeq_{4\delta_Q + 115352832\eps}\;
      \Id \ot \sum_{\alpha a + \beta b = c} \widehat{M}^{(\mathtt{Point}, Z), z}_b
        \widehat{M}^{(\mathtt{Point}, X), x}_a\;, \\
    \widehat{Q}^{u}_c \ot \Id \;\simeq_{4\delta_Q + 461411328\eps}\;
      \Id \ot \sum_{\alpha a + \beta b = c} \widehat{M}^{(\mathtt{Point}, X), x}_a
        \widehat{M}^{(\mathtt{Point}, Z), z}_b\;.
  \end{gather*}
\end{enumerate}
\end{lemma}

\begin{proof}
\leanok
\uses{lem:qld-combined-points,def:generalized-pauli,def:distance}
Both items are about the same coarse-graining and they are free for opposite reasons.

\textbf{Item 1 is free because both families are projective.} The fibres
$\{(a, b) : \alpha a + \beta b = c\}$ partition $\F_q^2$, so the coarse-grained families are
projective too --- a sum of mutually orthogonal projections is a projection. For projective
families the summed cross-party deviation and the agreement probability determine each other
exactly, $1 - \sum_c \bra{\psi} A_c \ot B_c \ket{\psi} = \tfrac12 \sum_c \norm{(A_c \ot \Id -
\Id \ot B_c)\ket{\psi}}^2$, and agreement can only increase under a coarse-graining applied to
both sides. So the error is unchanged, for each $(\alpha, \beta)$ separately. Note that the
state-dependent distance has \emph{no} data-processing inequality of its own; projectivity is
what buys the two conversions.

\textbf{Item 2 is not free, and Parseval is what pays for it.} The right-hand families are
\emph{not} POVMs --- they are products of two measurements --- so the argument above does not
apply, and a per-fibre triangle inequality would cost the fibre size $q$, which the conclusion
cannot afford. Instead, write
\begin{equation*}
  U_{a, b} \;=\; \bigl(\widehat{Q}^{x,z}_{a,b}\bigr)_{\alice\alice'}
    - \bigl(\widehat{M}^{Z, z}_b \widehat{M}^{X, x}_a\bigr)_{\bob\alice''}\;,
  \qquad
  D_c \;=\; \sum_{\alpha a + \beta b = c} U_{a, b}\;,
\end{equation*}
so that item 2's deviation is $\norm{D_c \ket{\hat\psi}}$. Parseval over $\F_q$ turns the sum
over fibres into an average over one probe,
$\sum_c \norm{D_c\ket{\hat\psi}}^2 = \E_r \norm{\sum_{a,b} (-1)^{\fieldtr(r(\alpha a + \beta
b))} U_{a,b} \ket{\hat\psi}}^2$, and two things then happen. The $r = 0$ term vanishes
\emph{identically}, because $\sum_{a,b} U_{a,b} = \Id - (\sum_a \widehat{M}^X_a)(\sum_b
\widehat{M}^Z_b) = 0$ --- completeness of all three families, and the only place the zero-sum
property is used. For $r \neq 0$ the pair $(r\alpha, r\beta)$ is uniform on $\F_q^2$ when
$(\alpha, \beta)$ is, so averaging over $(\alpha, \beta)$ and applying Parseval over $\F_q^2$
(Definition~\ref{def:generalized-pauli}) returns $\sum_{a,b}
\norm{U_{a,b}\ket{\hat\psi}}^2$ exactly. Summing the $q - 1$ nonzero probes gives
$\tfrac{q-1}{q} \sum_{a,b} \norm{U_{a,b}\ket{\hat\psi}}^2$, which is item 2 of
Lemma~\ref{lem:qld-combined-points} with a factor below one.

The second relation is the same argument with the two orders exchanged, using
$\sum_{a,b} \widehat{M}^{X, x}_a \widehat{M}^{Z, z}_b = \Id$ in place of the other order.
\end{proof}

\paragraph{Comments.} Formalized in \texttt{MIPRE/Background/QLD/Combined.lean}, with Parseval
over one and two copies of the field and the coarse-graining identity in
\texttt{MIPRE/Foundations/Parseval.lean} and \texttt{MIPRE/Foundations/Sandwich.lean}.

The error is \textbf{improved, not merely preserved}: item 2 comes out at $\tfrac{q-1}{q}$ times
the input, so the factor is below one and there is no $md/q$ term. That is the paper's
round-12 repair of this step, and the ledger node records it: an earlier
orthogonality-and-Cauchy--Schwarz route claimed $O(\delta_Q^{1/2})$ and was retracted, every

rather than absorbing it.

Item 1 holds for \textbf{every} $(\alpha, \beta)$, not only on average, because nothing in its
proof averages over the probes --- the coarse-graining is free pointwise. Only item 2 needs the
$(\alpha,\beta)$-average, and it needs it for the bijection $(\alpha, \beta) \mapsto (r\alpha,
r\beta)$.

has to satisfy.

\textbf{What the containment forces.} Moving along the padded line moves the $X$ block of a point
along the $X$ block of the padded direction. So where that block is nonzero the $X$ subline has no
choice: its direction must be exactly that block, and the only freedom left is \emph{which seed}
presents it --- a seed of the prescribed axis index, drawn uniformly from that index's block of
$q/m$ elements. Where the block vanishes the $X$ block of a point is constant along the padded
line, the containment is free, and the subline is sampled fresh. Same for $Z$.

\textbf{The four cases, and why the two sides differ.} A coordinate position of
$\F_q^{4m}$ lies in the $X$ block, in the $Z$ block, at $\alpha$ or $\beta$, or among the dummy
coordinates, and which blocks of the padded direction vanish depends on the case \emph{and} on the
line type. For an axis-parallel padded line the direction is a single coordinate, so at most one of
the two blocks is nonzero. For a diagonal one it is $v$ truncated below the axis index, so an index
in the $X$ block leaves the whole $Z$ block untouched: both blocks are nonzero and the $Z$ subline
is forced as well, at axis index $0$ since nothing of its direction is truncated. That is the only
place the construction looks at the line type.

\textbf{The law.} The $X$ and $Z$ blocks of a uniform padded point are uniform and independent,
because $\F_q^{4m}$ is the disjoint union of the two blocks and the $2m$ positions above them, and
precomposition with an injection of index types carries the uniform measure to the uniform measure.
The two sides' data are therefore functions of disjoint parts of the randomness: the point and the
raw direction contribute one block each, and each side's fresh seed is its own. Averaging over the
sampling space is then exactly averaging over a product of two line-point laws, restricted on a side
precisely when that side is forced, with no constant left over --- the multiplicities are the
assignments the construction does not read, and they divide out. A side that is not forced gets the
unrestricted law, which is the uniform mixture of the $m$ restricted ones; that is the mixture in
the second property.

\textbf{The third property is definitional here}, the line type being a parameter handed to both
sides rather than something the construction chooses. What the paper draws from its Property~3 ---
that the directions agree --- is the four-case statement above.
\end{proof}

\paragraph{Comments.} Formalized in \texttt{MIPRE/Background/QLD/Padded.lean}, on the block

\textbf{The construction.} \texttt{LPData} is the triple $(u, s, v)$ a seeded line--point pair is

$\delta_{\mathrm{combine}}$. Everything the statement asserts \emph{about the measurement} is formalized ---
the combining map, its two degree bounds, the POVM $\widehat Q^\ell$ itself and the
coarse-graining that turns it into an $\F_q$-valued outcome --- in
\texttt{MIPRE/Background/QLD/Combine.lean}, on top of the transfer in
\texttt{MIPRE/Background/QLD/Product.lean}. The transfer is worth separating from the rest because it
is where an apparent obstruction turns out not to be one.

\textbf{The obstruction, and why it is not one.} Lemma~\ref{lem:qld-pairs-of-lines} is proved on the
Pauli basis test's own question distribution: one content, carrying both sides' points but a
\emph{single} seed and a \emph{single} raw direction, so the $X$ line and the $Z$ line it presents
are not independent. This lemma needs the same conclusion on a \emph{product} of two independent
line--point laws, which is what Lemma~\ref{lem:qld-sublines} delivers. The two distributions are
genuinely different --- but \textbf{no hypothesis of the pasting lemma involves both lines}. Each
marginal consistency involves one line and the two points; the fine self-consistency and the
collision term involve one line only. So the only laws that have to agree are one side's line--point
data on its own, one side's data together with the \emph{other side's point}, and the two points; and
each of those has the same law under a content as under a product, because a content's two points and
its seed and raw direction are independent and uniform. \texttt{avg\_content\_eq\_pair\_X},
\texttt{avg\_content\_eq\_pair\_Z} and \texttt{avg\_content\_eq\_pair\_pts} are those three
identities, each one line on top of \texttt{avg\_comp\_equiv\_fst}: exhibit the splitting of each
sample space that isolates the part the quantity depends on, and the marginals match.

\textbf{The change of variables generalizes for free.} The device that makes the point uniform on its
line without forming a conditional distribution needs a finite nonempty sample space, a shift action,
and a direction the shift does not move --- nothing about contents. \texttt{bijective\_shift} and
\texttt{sum\_shift\_gen} are that statement, and \texttt{sum\_pair\_shift} is its instance at a pair
with the shift acting on the $X$ side only.

\textbf{The combining map, and where its whole geometry sits.} \texttt{combine} is built through
Mathlib's \texttt{Polynomial} and truncated back to a coefficient vector: it is
$\alpha(t) f_X(x(t)) + \beta(t) f_Z(z(t))$ with all four substitutions \emph{affine} in the padded
parameter $t$, so the degree bound $md+1$ is a product of degrees and nothing more.
\texttt{eval\_padCombine} is the paper's parametrization-free defining property: at the padded
parameter $t$ the combined polynomial takes the value $\alpha \cdot f_X(x) + \beta \cdot f_Z(z)$
at the corresponding point, each side's polynomial read at the parameter of its \emph{own} subline.

The only geometric input is \texttt{xBlk\_dir\_sub} and its $Z$ twin
(Lemma~\ref{lem:qld-sublines}): the $X$ block of the padded direction is \emph{either} the $X$
subline's own direction \emph{or} zero. In the first case the subline parameter moves by exactly the
padded shift, in the second it does not move at all --- so in both it is $p_0 + c\,t$ with $c$ one or
zero, and \texttt{subAff} is that pair. The paper's case analysis over the branches of the subline
construction is not needed: the disjunction is proved once and used once.

\textbf{The axis bound is the same case distinction \texttt{padCase} already makes.} A padded
direction in the $X$ or $Z$ block leaves $\alpha$ and $\beta$ constant along the line, so the
combined polynomial inherits the degree of $f_X$ and $f_Z$ --- degree $d$ exactly when
Lemma~\ref{lem:qld-axis-degree} supplies it, which is why
\texttt{degLE\_padCombine\_aline} takes that as a hypothesis. A direction at $\alpha$, $\beta$ or
a dummy coordinate leaves \emph{both} blocks constant, so the combined polynomial is affine, and
there $d \geq 1$ is all that is used.

\textbf{The measurement.} \texttt{padLineMats} is $\widehat Q^\ell$: the uniform average, over
the fresh randomness the subline construction draws, of the pasted line measurement

\textbf{Lemma~\ref{lem:qld-pairs-of-lines}, on the product.}
\texttt{pairs\_of\_lines\_prod} is that lemma at $k_X = \mathtt{pairCX}$,

\textbf{And from the game.} \texttt{pairs\_of\_lines\_prod\_of\_items} takes only game-level

\begin{lemma}[Admissible padded setup]
\label{lem:qld-global-setup}

correct phase --- and which are $\delta$-close to the strategy's
$(\mathtt{Point}, W)$ measurements on the expanded state.
\end{lemma}

\begin{proof}
\uses{lem:schwartz-zippel}
Exactness of the linearity comes from projectivity of the simultaneous measurement together
with the group law of the Pauli basis; Theorem~\ref{thm:linearity} is \emph{not} used here.
The remaining ingredient bounds the mass the simultaneous measurement places on
non-multilinear outcomes, via Lemma~\ref{lem:schwartz-zippel} applied in the correct
direction --- distinct polynomials of degree $md$ \emph{agree} on at most an $md/q$ fraction,
so it is the agreement and not the disagreement that the bound controls
(Remark~\ref{rem:qld-admitted}).
\end{proof}

\begin{definition}[Succinctly described 3SAT and decoupled 5SAT]
\label{def:succinct-3sat}
\lean{MIPRE.SAT.Circuit.formula3, MIPRE.SAT.Circuit.formula5, MIPRE.SAT.Circuit.DescribesDecider}
\leanok
A Boolean circuit $C$ with three inputs of length $m$ and three single-bit inputs
\emph{succinctly describes} the 3SAT formula on the $2^m$ variables $x_0, \ldots,
x_{2^m - 1}$ whose clauses are the $x_{i_1}^{o_1} \vee x_{i_2}^{o_2} \vee x_{i_3}^{o_3}$
with $C(i_1, i_2, i_3, o_1, o_2, o_3) = 1$, integers presented to the circuit in binary
(\cite[Definition of succinct descriptions of 3SAT formulas, Section 10]{JNVWY20}). A
\emph{decoupled 5SAT} formula has five blocks of variables and clauses with one literal from
each block, evaluated on five separate assignments; a circuit with two inputs of length
$\ell$, three of length $r$ and five single-bit inputs succinctly describes one in the same
way. Such a circuit \emph{describes the decider $\decider$ on $(n, x, y)$ at time $T$} if,
for all $a, b \in \{0,1\}^{2^\ell}$, some three assignments complete $(a, b)$ to a
satisfying tuple exactly when $a$ and $b$ are the two-bit tape encodings ($0 \mapsto 00$,
$1 \mapsto 01$, blank $\mapsto 10$) of strings of length at most $T$ on which $\decider$
accepts $(n, x, y, \cdot, \cdot)$ within $T$ steps. Circuits have fan-in and fan-out at most
two and a terminal output gate, and their size is the number of gates.
\end{definition}

\begin{theorem}[Explicit succinct Cook--Levin]
\label{thm:succinct-sat}
\ledgernode{1.3.1}
\uses{def:succinct-3sat,def:decider,lem:correct-tableau}
\lean{MIPRE.SAT.SuccinctCookLevin, MIPRE.SAT.succinctCookLevin}
\leanok
There is a polynomial-time algorithm that, given a decider $\decider$, an index $n$, a
time bound $T$ (in binary), a question length $Q$, a bound $\sigma$ and strings $x, y$, with
$\max\{Q, 2\lceil \log n \rceil\} \leq T$, $\abs{\decider} \leq \sigma$ and $\abs{x},
\abs{y} \leq Q$, outputs a circuit $C$ on $3m + 3$ inputs which succinctly describes
(Definition~\ref{def:succinct-3sat}) a 3SAT formula $\varphi$ on $2^m$ variables such that
\begin{enumerate}
\item for all $a, b \in \{0,1\}^{2T}$, the formula $\varphi$ has a satisfying assignment
  whose first $4T$ variables are $(a, b)$ if and only if $a$ and $b$ are the two-bit tape
  encodings of strings $a', b'$ of length at most $T$ such that $\decider$ accepts $(n, x,
  y, a', b')$ within $T$ steps;
\item $m = O(\log T + \log \sigma)$ and $2^m \geq 4T$;
\item $C$ has at most $s(n, T, Q, \sigma) = \poly(\log n, \log T, Q, \sigma)$ gates;
\item the algorithm runs in time $\poly(\log n, \log T, Q, \sigma)$;
\item $m$ and $s$ are computable from the parameters in time polynomial in their bit
  lengths.
\end{enumerate}
\end{theorem}

\begin{proof}
\leanok
\uses{lem:correct-tableau,def:succinct-3sat}
Fix one interpreter machine $U$, a multi-input Turing machine that runs the small-step
semantics of the ambient model on a serialized configuration, and put the five parameters
$n, x, y$ and the two answers on its input tapes, the answers on the two tapes left free.
Then the decider is *data* on an input tape and never becomes a machine, so the local check
circuit of $U$ is a constant; and the answer blocks are the contents of two tapes at time
$0$, which is item 1's block structure for free. The time bound $T$ in the cost of the
ambient model becomes a tableau length $S = 2^e$ with $e = 5\lceil \log T \rceil +
5 \lceil \log \sigma \rceil + 24$, because a run of the decider of cost $T$ is a run of
$U$ of at most $400000 (T + \sigma + 1)^5$ steps.

Apply Lemma~\ref{lem:correct-tableau} to $U$ at that length. Index its variables by bit
strings of width $m = 11 + 15 W + Q_b + G_b$, where $W = e + 4$ is the width of a time, a
position or a window offset: the top bit says whether the index is structured or is one of
the first $4T$ answer indices, and the remaining fields are a kind tag, a time, a tape, a
position, a value, a state, the thirteen window offsets and a gate number. Each clause
family of the tableau is then a predicate on those decoded fields, and each such predicate
is computed by a formula over the $3m + 3$ input bits of the describer, built from equality
and comparison with a constant, equality of two fields, and a ripple-carry increment. The
describer is the flattening of the disjunction of those formulas to a circuit, together with
two clauses forcing the cell at position $T + 3$ of each free tape to hold the blank, which
is what bounds the answers read off a satisfying assignment by $T$ rather than by $2S$.

Item 1 is the correctness of the tableau in both directions, transported along the indexing:
an assignment of the tableau gives one of the formula by decoding, and an assignment of the
formula gives one of the tableau by the canonical index, where an answer cell of a free tape
inside the answer block is sent to its answer index, so that the extension hypothesis on the
first $4T$ variables is exactly the time-$0$ row of that tape. Item 2 is arithmetic on $m$
and $e$. For items 3 and 4, every builder of the formula is a polynomial-time function of the
parameters, since the encoding of a formula is the encoding of its post-order list and each
builder is therefore a concatenation, a map or a fold over lists, with the widths and offsets
carried in unary; and then item 3 needs no separate argument, because the output of a
polynomial-time function is no larger than its running time and a circuit has at most as
many gates as its encoding has nodes. Item 5 follows by rounding $s$ up to a power of two
whose exponent is a multiple of the bit lengths of the parameters.
\end{proof}

\paragraph{Lean statement.} The theorem is the structure
\texttt{MIPRE.SAT.SuccinctCookLevin}: the algorithm a polynomial-time function of the
ambient model (\texttt{PolyTimeFun}) from $((\decider, n, T, Q, \sigma), x, y)$ to circuits
as data, ``within $T$ steps'' read as cost at most $T$ in the ambient model, the reading of
every time bound of the pipeline, and the five items as fields. An earlier statement asked
for a succinct description in the sense of Definition~\ref{def:succinct}, which is a
program answering bit queries and not the circuit the proof produces and the PCP consumes,
and it carried none of the five items, all of which Lemma~\ref{lem:decoupled-5sat} and the
PCP parameters depend on. The statement is validated by its consumer
(\texttt{MIPRE.SAT.DecoupledDescriber}, Lemma~\ref{lem:decoupled-5sat}) before its proof;
the route to the proof is \texttt{planning/succinct-cook-levin.md}.

\paragraph{Formalization status.} Complete: statement and proof, the structure inhabited by
\texttt{MIPRE.SAT.succinctCookLevin} (\texttt{MIPRE/TM/CookLevin/}). Three things the
formalization settled against the first design, all recorded in
\texttt{planning/succinct-cook-levin.md}. The index width has $W = e + 4$, not $e + 2$,
because a position field must hold $2S + 7$; the state field is $\lceil \log |Q| \rceil$
bits rather than a fixed four. Two clauses beyond the tableau are needed: soundness needs
answers of length at most $T$, and the tableau alone admits answers of length up to $2S$.
And the canonical index of an answer cell of a free tape is its answer index, not its
structured index --- that is what ties the extension hypothesis on the first $4T$ variables
to the time-$0$ row of the tape, and getting it wrong makes the soundness direction
unprovable rather than merely awkward.

Two notes on the Lean, both about scale. The check circuit of the interpreter machine is the
truth-table circuit of a fixed Boolean function on the window variables, so its gate count is
$7 \cdot 2^{|\mathrm{window}|}$ --- an enormous constant, exactly as the paper's
\texttt{lem:pack-check-size} carries one. Nothing unfolds it, and nothing may: the defining
terms of the state count and of the check circuit mention a cardinality and a truth table
whose evaluation does not terminate in practice, so they are kept opaque (the state width is
\texttt{irreducible} and the circuit is taken from an existential). And the program layer is
written so that each of the fifty-odd family formulas costs one \texttt{simp}: a field record
is a record of \emph{readers} of the describer's input whose evaluation is a structure literal
of those programs' own values, which makes every formula match its program definitionally.

\paragraph{Source.} Cook~\cite{Coo71} and Levin for the tableau reduction; Papadimitriou
and Yannakakis~\cite{PY86} for succinct representations and their complexity;
the explicit form needed here is~\cite[Section 10]{JNVWY20}.

\paragraph{Comments.} This is the classical core of answer reduction: the players prove
satisfiability of $\varphi$ (arithmetized over a low-degree extension) instead of
sending $(a,b)$. The statement is long but entirely combinatorial. Cook--Levin has been
formalized in other proof assistants; a Lean formalization with the explicit-uniformity

A caution on the construction, from the verification campaign~\cite{Audit26} and not

\begin{lemma}[Correctness of the Cook--Levin tableau]
\label{lem:correct-tableau}
\uses{def:multiinput-tm}
\lean{MIPRE.TM.CookLevin.tableau_sat_iff}
\leanok
Let $M$ be a multi-input Turing machine (Definition~\ref{def:multiinput-tm}) with $i$ input
tapes and $w$ work tapes over a finite alphabet, $\mathsf{acc}$, $s_0$, $s_1$ symbols and
$S$ a step bound, and fix the strings on some of the input tapes. The \emph{tableau formula}
of $S$ steps of $M$ is the 3SAT formula on the variables ``cell $p$ of tape $d$ at time $t$
holds $v$'' (one for each of $2S + 7$ cells and each value: a symbol, the blank or the
boundary marker), ``the head of tape $d$ is on cell $p$ at time $t$'', ``the state at time
$t$ is $q$'', the emission flags of every step, and, for every step and every choice of a
five-cell window on each tape, the gate variables of a circuit deciding local consistency
of the window; its clauses are the start row, the boundary rows, the free-tape row (a
string over $\{s_0, s_1\}$ followed by blanks), the emission bookkeeping, the acceptance
row and the circuit-to-3SAT clauses of the local check at every window. The formula is
satisfiable if and only if $M$, started on the fixed strings and on some strings over
$\{s_0, s_1\}$ on the free input tapes, halts within $S$ steps having output exactly
$\mathsf{acc}$.
\end{lemma}

\begin{proof}
\leanok
Completeness: the encoding of the run satisfies every clause, the window clauses because
the windows of a run pass the local check (\texttt{locallyConsistent\_localCfgOf}) and the
circuit computes it, which is its specification (\texttt{IsCheckCircuit}). Soundness: the
time-$0$ rows of a satisfying assignment determine the free strings; by induction on time
the rows encode the run, because a locally consistent window whose time-$t$ part is that
of a configuration has the time-$(t + 1)$ part of its successor (\texttt{post\_of\_pre});
the acceptance row and the emission bookkeeping then give the output. Every head stays on
an interior cell: a head moves at most one cell per step and a tape has $2S + 7$ cells.
\end{proof}

\paragraph{Source.} The tableau of~\cite[Section 10]{JNVWY20} and the standard argument of
Cook~\cite{Coo71}; the companion paper's \texttt{answer\_reduction.tex} (its
\texttt{lem:correct-tableau}).

\paragraph{Comments.} The lemma is Cook--Levin for one fixed machine; the theorem applies
it to a fixed interpreter machine for the decider's program, and the succinct description
is a circuit for the clauses of this formula --- pieces S2 and S3 of
\texttt{planning/succinct-cook-levin.md}. The interpreter machine (S2) is formalized:
\texttt{MIPRE.TM.Interp.U}, with \texttt{accepts\_of\_acceptsWithin} and
\texttt{acceptsWithin\_of\_accepts} (\texttt{MIPRE/TM/Interp/Run.lean}) --- on the inputs
$(\decider, n, T, x, y, a, b)$ with $\abs{a}, \abs{b} \leq T$, it halts having output
exactly $1$, within an explicit polynomial number of steps, if and only if $\decider$
accepts $(n, x, y, a, b)$ within cost $T$. The local check circuit is a parameter of the
formula, with its specification a hypothesis of the lemma, so that the lemma is proved once
and the circuit supplied with the describer. The windows are five cells wide: a head at one
of the middle three cells then sees, in the window, both the cell it leaves and the cell it
reaches, and the clamped moves of an input head at the ends of the input are decided from
the window alone.

\subsection{Self-dual normal bases}
\label{sec:rr-self-dual-basis}

\begin{definition}[Self-dual and normal bases]
\label{def:self-dual-basis}
\lean{MIPRE.LowDegree.IsSelfDualBasis, MIPRE.LowDegree.IsNormalBasis}
\leanok
Let $K / \F$ be a finite extension of finite fields, $q = \abs{\F}$. A basis $(e_i)$ of $K$
over $\F$ is \emph{self-dual} when its Gram matrix for the trace form is the identity,
$\mathrm{tr}_{K / \F}(e_i e_j) = \delta_{ij}$, and \emph{normal} when its vectors are the
Frobenius orbit $\alpha, \alpha^q, \alpha^{q^2}, \ldots$ of a single element $\alpha \in K$.
\end{definition}

\begin{lemma}[Self-dual coordinates are trace pairings]
\label{lem:downsize-field}
\uses{def:self-dual-basis}
\lean{MIPRE.LowDegree.coord_eq_trace, MIPRE.LowDegree.trace_mul_eq_dot}
\leanok
Let $(e_i)$ be a self-dual basis of $K / \F$ and $\downsize(x)$ the coordinate vector of
$x \in K$. Then
\begin{enumerate}
\item $\downsize(x)_i = \mathrm{tr}_{K/\F}(x e_i)$, and
\item $\mathrm{tr}_{K/\F}(xy) = \downsize(x) \cdot \downsize(y)$.
\end{enumerate}
\end{lemma}

\begin{proof}
\leanok
\uses{def:self-dual-basis}
For the first, expand $x = \sum_j \downsize(x)_j e_j$, multiply by $e_i$ and take the trace:
self-duality kills every term but $j = i$. The second follows from the first by the same
expansion, $\mathrm{tr}(xy) = \sum_j \downsize(x)_j \mathrm{tr}(e_j y) = \sum_j \downsize(x)_j
\downsize(y)_j$.
\end{proof}

\paragraph{Source.} Items 1 and 2 of the paper's \texttt{lem:downsize\_field}
(\texttt{preliminaries.tex}); ledger node \texttt{1.1.6.1.11}, which also records that neither
item needs normality or odd $k$. The paper's third item, compatibility of the downsized matrix
map with $\downsize$ ($\downsize_{\mathrm{m}}(M)\downsize(v) = \downsize(Mv)$), is not
formalized here: it is wanted by the conditionally linear downsizing of
Lemma~\ref{lem:cl-downsize}, which is stated over $\F_2$ and so has not needed it.

\paragraph{Comments.} These two identities are why the pipeline insists on self-duality rather
than on an arbitrary basis: they make $\downsize$ computable by the trace form, with no matrix
inversion, and turn the trace form into the standard inner product on coordinates. They are
also what collapses the Pauli phase index $G_E \downsize(b)$ to $\downsize(b)$ at $G_E = I$
(ledger node \texttt{1.1.6.1.13}).

\begin{lemma}[Deterministic self-dual normal basis]
\label{lem:self-dual-basis}
\uses{def:self-dual-basis}
\ledgernode{1.1.6, 1.1.6.1, 1.1.6.1.1}
\lean{MIPRE.SAT.shoupSelfDualNormalBasis_selfDual,
  MIPRE.SAT.shoupSelfDualNormalBasis_normal,
  MIPRE.SAT.shoupSelfDualNormalBasisProg,
  MIPRE.SAT.shoupSelfDualNormalBasisProg_correct,
  MIPRE.SAT.shoupSelfDualNormalDataProg,
  MIPRE.SAT.shoupSelfDualNormalDataProg_correct,
  MIPRE.SAT.shoupSelfDualNormalDataProg_time_le,
  MIPRE.SAT.effective_selfDualNormalBasis}
\leanok
There is a deterministic algorithm that, given an odd integer $k > 0$, outputs in time
$\poly(k)$ a self-dual normal basis of $\F_{2^k}$ over $\F_2$ together with its
multiplication tables.
\end{lemma}
\begin{proof}
\leanok
Construct the normal basis by Lemma~\ref{lem:effective-normal-element}, apply
Lemma~\ref{lem:effective-normal-selfdualization}, and transport the multiplication
table by Lemma~\ref{lem:effective-field-matrices}. The composed ambient program
has unary degree as its only input and prints both the basis vectors and all
table coefficients. Its source runtime is bounded by a fixed polynomial in $k$.
\end{proof}

in~\cite[Section 3]{JNVWY20}; the campaign has since \emph{proved} both steps --- ledger node

\paragraph{Comments.} Needed so that verifiers can do field arithmetic over
$\F_{2^k}$ using only $\F_2$-data (the ``downsize'' maps), uniformly and
deterministically. This is why field sizes are restricted to $q = 2^k$, $k$ odd

quantum. Two consumers now name it: the admissible field with a bit representation that

This lemma is the whole of what this blueprint assumes beyond Mathlib, and it is worth being
precise about which part. The companion campaign has proved the normal-basis construction and
the self-dualization --- circulant Gram self-dualization, the cyclic Frobenius vector, table
transport and the trace identities, over sixteen sub-nodes. One ingredient remains admitted,
node $1.1.6.1.1$: a fixed uniform deterministic algorithm that on input $k \geq 1$ outputs a
monic irreducible polynomial of degree $k$ over $\F_2$ in time $\poly(k)$, which is Shoup's
theorem specialized to $p = 2$. The ledger records it with an explicit contract naming what is
\emph{not} claimed --- no $\poly(\log k)$ output time, no variable-characteristic

That axiom now exists, in that shape: \texttt{MIPRE.LowDegree.exists\_shoup\_irreducible}
(\texttt{MIPRE/Foundations/LowDegree/Shoup.lean}), whose module docstring carries the contract
including the five disclaimers, and which is recorded in \texttt{MIPRE/Axioms.lean}. Its input

contract claims: had the input been $k$ in binary the same structure would have asserted the

formalize, and because an axiom appears by name in \texttt{\#print axioms} of everything that
rests on it, making the dependency machine-checkable rather than a matter of reading prose.

separation is performed directly in the cyclic algebra, as detailed in

Shoup axiom remains the only nonstandard assumption.

\begin{lemma}[Self-dualization in the group algebra]
\label{lem:group-algebra-selfdualization}
\ledgernode{1.1.6.1.7, 1.1.6.1.8}
\uses{def:self-dual-basis}
\lean{MIPRE.LowDegree.exists_mul_involute_eq, MIPRE.LowDegree.mul_self_bijective,
  MIPRE.LowDegree.exists_mul_involute_eq_of_charTwo,
  MIPRE.LowDegree.binarySquareRoot, MIPRE.LowDegree.coeff_binarySquareRoot,
  MIPRE.LowDegree.binarySquareRoot_mul_self, MIPRE.LowDegree.binarySquareRoot_mul_involute}
\leanok
Let $R$ be a commutative ring on which $x \mapsto x^2$ is a bijection and let $\tau$ be a ring
automorphism of $R$. Then every $u \in R$ with $\tau(u) = u$ is of the form $a \cdot \tau(a)$.

Squaring \emph{is} a bijection of the group algebra $\F[G]$ when $\F$ is a finite field of
characteristic two and $G$ is a finite abelian group of odd order.
For $\F=\F_2$, its inverse has the explicit coefficient formula $b(g)=u(g+g)$.
This $b$ satisfies $b^2=u$, and $b\tau(b)=u$ whenever $\tau(u)=u$.
\end{lemma}

\begin{proof}
\leanok
For the first statement, let $b$ be the square root of $u$. Then $\tau(b)^2 = \tau(b^2) =
\tau(u) = u = b^2$, so $\tau(b) = b$ by injectivity of squaring, and $b \cdot \tau(b) = b^2 = u$.

For the second, squaring is additive in characteristic two, and on a single generator
$\delta_g r$ it is $\delta_{g + g} r^2$. Doubling is injective on $G$: if $(a - b) + (a - b) =
0$ then the additive order of $a - b$ divides both $2$ and $\abs{G}$, which are coprime because
$\abs{G}$ is odd. So the coefficient of $x^2$ at $g + g$ is the square of the coefficient of $x$
at $g$, and $x^2 = 0$ forces $x = 0$. Squaring is then injective, because $(x - y)^2 = x^2 -
y^2$ in characteristic two, and injective implies bijective on a finite set.
Over $\F_2$ the coefficients are fixed by squaring, so the displayed coefficient
permutation gives the square root directly. The following lemma implements that
permutation in the ambient cost model.
\end{proof}

\paragraph{Comments.} This is the self-dualization step of Lemma~\ref{lem:self-dual-basis}, and
it is where oddness of $k$ is used. If $\alpha$ is a normal element of $\F_{2^k}/\F_2$ with
Galois group $G$, the trace-form Gram matrix of its orbit is the circulant determined by
$c \in \F_2[G]$, $c(g) = \mathrm{tr}(\alpha \cdot g\alpha)$: it is a unit, because the Gram
matrix of a basis under the nondegenerate trace form is invertible, and it is fixed by the
involution $\tau$ induced by $g \mapsto g^{-1}$, because that Gram matrix is symmetric.
Replacing $\alpha$ by $a \cdot \alpha$ replaces $c$ by $a \cdot \tau(a) \cdot c$, so a self-dual
normal basis is exactly a solution of $a \cdot \tau(a) = c^{-1}$, which is what the lemma
supplies. Nodes $1.1.6.1.7$ and $1.1.6.1.8$ carry an explicit circulant Gram manipulation
instead; the argument above replaces it, and needs no factorization of $X^k - 1$ and no norm
computation in the factors.

What the lemma does \emph{not} do is assemble a basis: the group algebra must be made to act on
$\F_{2^k}$, the Gram identity must be proved, and $c$ must be shown to be a unit. Those three
are bookkeeping, and Lemma~\ref{lem:self-dual-basis-exists} now carries them out; the algorithm
is a separate matter again.

\begin{lemma}[Existence of a self-dual normal basis]
\label{lem:self-dual-basis-exists}
\uses{def:self-dual-basis,lem:group-algebra-selfdualization}
\lean{MIPRE.LowDegree.gramPair, MIPRE.LowDegree.gramPair_ofAlg,
  MIPRE.LowDegree.isUnit_gram, MIPRE.LowDegree.exists_gramPair_eq_one,
  MIPRE.LowDegree.exists_isSelfDualBasis_isNormalBasis,
  MIPRE.LowDegree.exists_selfDualNormalBasis_two}
\leanok
Let $K / \F$ be a Galois extension of finite fields of characteristic two whose degree is odd.
Then $K$ has a basis over $\F$ that is at once normal and self-dual. In particular
$\F_{2^k}$ has a self-dual normal basis over $\F_2$ for every odd $k \geq 1$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:group-algebra-selfdualization}
Let $G = \mathrm{Gal}(K/\F)$, which is cyclic, hence abelian, so $\F[G]$ is a commutative ring, and
$\abs{G} = [K : \F]$ is odd. Let $\alpha$ be the normal element Mathlib's normal basis theorem
supplies, write $\mathrm{ofAlg}(a)$ for the element of $K$ with normal-basis coordinates
$a \in \F[G]$, and let
\[
  c(x, y) \in \F[G], \qquad c(x, y)_g = \mathrm{tr}(x \cdot g y), \qquad c = c(\alpha, \alpha).
\]
Because $\mathrm{tr} \circ g = \mathrm{tr}$, the pairing of two members of one orbit depends
only on their ratio, and expanding both coordinate sums gives the transformation law
\[
  c(\mathrm{ofAlg}(a), \mathrm{ofAlg}(b)) = a \cdot \tau(b) \cdot c,
\]
where $\tau$ is the involution of $\F[G]$ induced by $g \mapsto g^{-1}$. The element $c$ is
fixed by $\tau$, because $\mathrm{tr}(\alpha \cdot g^{-1}\alpha) =
\mathrm{tr}(g\alpha \cdot \alpha)$, and it is a unit: if $c \cdot z = 0$ then the
transformation law with $b = \tau(z)$ makes $\mathrm{tr}(\mathrm{ofAlg}(\tau z) \cdot x)$
vanish for every $x \in K$, so $\mathrm{ofAlg}(\tau z) = 0$ by nondegeneracy of the trace form,
so $z = 0$; a non-zero-divisor in a finite commutative ring is a unit. Then $c^{-1}$ is also
$\tau$-fixed, and Lemma~\ref{lem:group-algebra-selfdualization} --- squaring is bijective on
$\F[G]$ in characteristic two with $\abs{G}$ odd --- gives $a$ with $a \cdot \tau(a) = c^{-1}$.
For $\beta = \mathrm{ofAlg}(a)$ the transformation law reads $c(\beta, \beta) = 1$, which says
exactly that the Frobenius orbit of $\beta$ is orthonormal for the trace form. It is in
particular linearly independent, so it is a basis, and it is normal by construction.
\end{proof}

\paragraph{Comments.} This is existence, not construction, so it is not
Lemma~\ref{lem:self-dual-basis} and does not discharge any ledger node: the normal element
comes from Mathlib's \texttt{IsGalois.normalBasis}, which is a theorem and not an algorithm,
and nothing here is claimed to run in $\poly(k)$. What it does settle is that the three
bookkeeping items left open by Lemma~\ref{lem:group-algebra-selfdualization} --- making the
group algebra act, the Gram identity, and unitness of $c$ --- are genuinely bookkeeping, and
that the short self-dualization argument assembles. The formalization indexes the group algebra
by $G$ itself rather than by $\mathbb{Z}/k$, which is what makes $\mathrm{ofAlg}$ the inverse of
Mathlib's \texttt{normalBasis.repr} on the nose; the \texttt{CommGroup} instance comes from
cyclicity, supplied locally.

  MIPRE.SAT.shoupNormalGram_selfDualize,
  MIPRE.SAT.shoupChangedVectors_normal,
  MIPRE.SAT.shoupSelfDualBasis_selfDual,
  MIPRE.SAT.shoupSelfDualBasis_normal,
  MIPRE.SAT.shoupSelfDualizeProg,
  MIPRE.SAT.shoupSelfDualizeProg_correct,
  MIPRE.SAT.shoupSelfDualizeProg_time_le}
\leanok
Given the polynomial-basis encodings of any normal basis in the degree-$k$ Shoup
field, for odd $k$, a uniform deterministic polynomial-time ambient program
outputs the canonical encodings of a self-dual normal basis.
\end{lemma}
\begin{proof}
\leanok
The normal Gram matrix is the circulant determined by its first column. The
regular representation embeds the binary cyclic group algebra into matrices,
and the inverse of a nonsingular circulant remains circulant. Read the computed
inverse's first column at doubled indices, then print its circulant matrix $H$.
The coefficient square-root identity gives $H^2=G^{-1}$; symmetry and commutation
give $H^{\mathsf T}GH=I$. Apply $H$ to the supplied basis columns. The identity
Gram matrix proves independence, and a circulant change of coordinates preserves
the Frobenius orbit. All matrix constructions and the final change of basis are
implemented with polynomial runtime in $k$.
\end{proof}
\paragraph{Scope.} This constructs a self-dual normal basis from a supplied normal
basis. Lemma~\ref{lem:effective-normal-element} supplies that basis from unary $k$,
and Lemma~\ref{lem:self-dual-basis} composes the complete algorithm.

\subsection{The Schwartz--Zippel lemma}
\label{sec:rr-schwartz-zippel}

\effortMathlib

\begin{lemma}[Schwartz--Zippel]
\label{lem:schwartz-zippel}
\lean{MIPRE.LowDegree.agree, MIPRE.LowDegree.prob_agree_le_totalDegree,
  MIPRE.LowDegree.prob_agree_le_individualDegree,
  MIPRE.LowDegree.card_agree_le_of_natDegree}
\leanok
Let $f \neq g : \F_q^m \to \F_q$ be polynomials of total degree at most $d$. Then
$\Pr_{x \sim \F_q^m}[ f(x) = g(x) ] \leq d/q$.
\end{lemma}

\begin{proof}
\leanok
Mathlib's \texttt{MvPolynomial.schwartz\_zippel\_totalDegree} is the statement for one
nonzero polynomial and a product of arbitrary finite subsets; apply it to $f - g$ with
every subset all of $\F_q$.
\end{proof}

\paragraph{Source.} \cite{Sch80,Zip79}. A version is available in Mathlib
(multivariate polynomials over finite fields).

\paragraph{Comments.} Used throughout the low-degree machinery (distance of the
Reed--Muller code, soundness of arithmetization). Three shapes are formalized, because
three consumers want three different ones: the probability bound $d/q$ above; the
individual-degree variant $m d / q$, derived from Mathlib's
\texttt{schwartz\_zippel\_sum\_degreeOf} rather than through the lossy
$\text{total degree} \leq m d$; and the univariate case as a count rather than a
probability --- two distinct polynomials of degree at most $d$ agree at no more than $d$
points --- which is the form Lemma~\ref{lem:lidt-sync-transfer} uses. The probabilities
are stated over $\R$, which is where the low-degree machinery's error terms live, rather
than over Mathlib's $\mathbb{Q}_{\geq 0}$.

\subsection{Approximating the value from below ($\MIPstar \subseteq \RE$)}
\label{sec:rr-value-approx}

\effortMedium

\begin{lemma}[The value is approximable from below]
\label{lem:value-lower-approx}
\lean{MIPRE.ValueApprox.rePred_lt_quantumValue, MIPRE.ValueApprox.RawStrategy,
  MIPRE.ValueApprox.RawStrategy.interp, MIPRE.ValueApprox.Check,
  MIPRE.ValueApprox.primrecPred_check, MIPRE.ValueApprox.check_iff_interp,
  MIPRE.ValueApprox.exists_check_iff, MIPRE.ValueApprox.REPred.of_primrecRel_exists,
  MIPRE.rePred_lt_quantumValue_comp, MIPRE.exists_semidecider_lt_quantumValue,
  MIPRE.Cost.exists_semidecider}
\leanok
\ledgernode{1.1.7.2, 1.1.7.2.1, 1.1.7.2.2, 1.1.7.2.5, 1.1.7.2.6, 1.8.1}
\uses{def:game-description,def:q-value,def:re,lem:rational-strategies-suffice}
Write $\game_g$ for the game described by a game description $g$
(Definition~\ref{def:game-description}). The set
\[
  \{ (g, p, q) : g \text{ a game description},\ p, q \in \N,\ \valstar(\game_g) > p/q \}
\]
is recursively enumerable.
\end{lemma}

\begin{proof}
\leanok
By Lemma~\ref{lem:rational-strategies-suffice}, $\valstar(\game_g) > p/q$ if and only if
some valid exact strategy with entries in $\mathbb{Q}(i)$ has value greater than $p/q$.
A \emph{raw candidate} is a common denominator $k$, two dimensions, the Gaussian-integer
numerators of the two players' operators as lists of list matrices, and the numerators of
the state; it denotes the exact strategy with entries divided by $k$, and every valid exact
strategy with $\mathbb{Q}(i)$ entries is denoted by one (clear a common denominator of its
finitely many entries). Raw candidates are first-order data, hence enumerable. For a raw
candidate the check ``$k > 0$, each family of operators is Hermitian, idempotent
($M^2 = kM$) and sums to $k\Id$, the state is nonzero, and $p \cdot W \cdot k^2 \cdot
\langle v, v\rangle < q \cdot \sum_{x,y} w(x,y) \sum_{a,b} D(x,y,a,b)\, \mathrm{re}\langle v,
(A^x_a \otimes B^y_b) v\rangle$'' is a finite conjunction of polynomial identities and one
comparison in Gaussian-integer arithmetic, so it is primitive recursive, and it holds if
and only if the denoted strategy is valid with value greater than $p/q$ (for $q = 0$, where
$p/q = 0$, the comparison is replaced by positivity of the numerator). Dovetailing over the
candidates and halting on the first that passes the check therefore halts exactly when
$\valstar(\game_g) > p/q$; the boundary case $\valstar(\game_g) = p/q$ is correctly
non-halting.
\end{proof}

\paragraph{Source.} Folklore; see~\cite[Section 12]{JNVWY20}, and statement (S) in the
proof of \texttt{cor:mip-re} of \texttt{recursive.tex} for a complete argument. The
statement was previously given for $\synval$ ``and likewise for $\valstar$''; it is now the
$\valstar$ statement, which is the one the pipeline consumes
(Remark~\ref{rem:compression-abstract}) and the one the paper proves. The $\synval$
analogue needs its own enumeration and density argument, since rounding the entries of a
synchronous strategy preserves neither projectivity nor synchronicity; it is not needed and
not planned. Thresholds are nonnegative rationals written as fractions $p/q$ of naturals, as
in the paper's (S), where $t \in [0,1]$; a negative threshold is trivial since $\valstar
\geq 0$. In Lean the game description is \texttt{HaltingGameValue.GameData}
(Definition~\ref{def:game-description}), read as a game of the foundations by
\texttt{GameData.game}, and recursive enumerability is Mathlib's \texttt{REPred}, the
domain of a partial computable function; integers travel as pairs of naturals, because
Mathlib's primitive recursion has no arithmetic on $\mathbb{Z}$. The form the
compressibility criterion consumes (its hypothesis that the complement of $B$ is the halting
set of a program, Lemma~\ref{lem:compressible-criterion}) is
\texttt{MIPRE.exists\_semidecider\_lt\_quantumValue}: for a computable family of game
descriptions and a threshold $p/q$, a well-scoped program of the ambient model halting
exactly on the strings whose game has value greater than $p/q$. It comes from the generic
passage \texttt{MIPRE.Cost.exists\_semidecider} from an \texttt{REPred} on bit strings to a
program (Definition~\ref{def:re}): a \texttt{ToPartrec} code for the predicate, translated
into the model by \texttt{Prog.ofCode}, run after an ambient loop has shifted every bit up
by one, since a \texttt{ToPartrec} code cannot see trailing zeros of its input and the bit
$0$ encodes as \texttt{nil}.

\paragraph{Comments.} The Lean route to the $\valstar$ half departs from the paper's proof
of (S). The paper enumerates \emph{approximate} candidates --- tuples with entries in
$\frac1k \mathbb{Z}[i]$ satisfying relaxed POVM constraints, tested by positive
semidefiniteness (Lemma~\ref{lem:norm-two-psd}) --- and repairs each into an exact POVM by
conjugating with $T_x^{-1/2}$, with the explicit slack $\Delta(d,k) = 250(\abs{\cal A} +
\abs{\cal B})^2 d/k$ (its Claims ``Stability'' and ``Density''; ledger nodes
\texttt{1.1.7.2.3} and \texttt{1.1.7.2.4}). The formalization enumerates \emph{exact}
projective strategies instead: every projective measurement is $U \Pi_a U^*$ for a unitary
$U$ and coordinate projections $\Pi_a$, and up to a phase every unitary is the Cayley
transform of a skew-Hermitian matrix, which is what gets rounded
(Lemma~\ref{lem:rational-pvm-dense}). Every candidate is then a genuine strategy, so only
the density half of the argument is needed, with no slack, no matrix square root and no
positive-semidefiniteness test, and the biconditional of
Lemma~\ref{lem:rational-strategies-suffice} is sharp. Lemma~\ref{lem:norm-two-psd} is
proved but no longer on the path (the ledger's node \texttt{1.1.7.2.1}, exact arithmetic
and the decidability of the candidate test, is accounted for by the primitive recursive
check itself); Lemma~\ref{lem:perturbation-split} is used as stated. The
choice was made for the formalization's sake --- the stability claim needs the functional
calculus for $T_x^{-1/2}$ and the candidate test needs decidability of positive
semidefiniteness over $\mathbb{Q}(i)$, both heavier in Lean than the Cayley
parametrization --- and it proves the same statement (S); it reverses the route decision
recorded in Finding 3 of \texttt{planning/ledger-informed-plan.md}, which is annotated
accordingly.

\begin{lemma}[Gaussian-rational projective measurements are dense]
\label{lem:rational-pvm-dense}
\lean{MIPRE.ValueApprox.IsPVM, MIPRE.ValueApprox.IsPVM.exists_unitary_pattern,
  MIPRE.ValueApprox.cayley, MIPRE.ValueApprox.cayley_mem_unitaryGroup,
  MIPRE.ValueApprox.exists_isSkewHermitian_cayley_eq,
  MIPRE.ValueApprox.exists_norm_eq_one_isUnit_one_add_smul,
  MIPRE.ValueApprox.norm_cayley_sub_cayley_le, MIPRE.ValueApprox.GaussianRat,
  MIPRE.ValueApprox.exists_unitary_entriesIn_norm_smul_sub_le,
  MIPRE.ValueApprox.IsPVM.exists_entriesIn_norm_sub_le}
\leanok
Let $\{M_a\}_{a \in \cal A}$ be a projective measurement on $\C^d$ and $\eta > 0$. There is
a projective measurement $\{M'_a\}_{a \in \cal A}$ on $\C^d$ with all entries in
$\mathbb{Q}(i)$ such that $\norm{M_a - M'_a} \leq \eta$ for every $a$.
\end{lemma}

\begin{proof}
\leanok
Diagonalize a Hermitian combination $\sum_a w_a M_a$ with distinct real weights $w_a$:
$M_a = U \Pi_a U^*$ for a unitary $U$ and the coordinate projections $\Pi_a$ of a pattern
$r : [d] \to \cal A$. For a unit scalar $z$, $(zU) \Pi_a (zU)^* = U \Pi_a U^*$, and
$\det(\Id + zU)$ is a polynomial in $z$ with constant term $1$, so some unit $z$ makes
$\Id + zU$ invertible; then $zU = (\Id - S)(\Id + S)^{-1}$ is the Cayley transform of the
skew-Hermitian matrix $S = 2(\Id + zU)^{-1} - \Id$. Round $S$ entrywise to $R$ with entries
in $\mathbb{Q}(i)$ and re-symmetrize, $S' = (R - R^*)/2$: $S'$ is skew-Hermitian, has
entries in $\mathbb{Q}(i)$, and $\abs{S_{ij} - S'_{ij}}$ is at most the rounding
precision. Its Cayley transform $U' = (\Id - S')(\Id + S')^{-1}$ is unitary with entries
in $\mathbb{Q}(i)$, and $\norm{zU - U'} \leq 2 \norm{S - S'}$, since $zU - U' =
2(\Id + S)^{-1}(S' - S)(\Id + S')^{-1}$ and $\norm{(\Id + S)^{-1}} \leq 1$ for skew-Hermitian
$S$ (the cross term of $\norm{x + Sx}^2$ vanishes). Finally $M'_a = U' \Pi_a U'^*$
satisfies $\norm{M_a - M'_a} \leq 2 \norm{zU - U'}$.
\end{proof}

\begin{lemma}[Exact rational strategies suffice]
\label{lem:rational-strategies-suffice}
\uses{def:tensor-strategy,def:q-value,lem:rational-pvm-dense,lem:perturbation-split}
\lean{MIPRE.ValueApprox.ExactStrategy, MIPRE.ValueApprox.ExactStrategy.IsValid,
  MIPRE.ValueApprox.ExactStrategy.EntriesIn, MIPRE.ValueApprox.ExactStrategy.value,
  MIPRE.ValueApprox.ExactStrategy.toStrategy,
  MIPRE.ValueApprox.ExactStrategy.value_le_quantumValue,
  MIPRE.ValueApprox.abs_bornValue_sub_le, MIPRE.ValueApprox.exists_exactStrategy_value_gt,
  MIPRE.ValueApprox.lt_quantumValue_iff}
\leanok
\ledgernode{1.1.7.2.3, 1.1.7.2.4}
An \emph{exact strategy} for a game $\game$ is the data of dimensions $d_\alice, d_\bob$,
matrices $A^x_a \in \C^{d_\alice \times d_\alice}$ and $B^y_b \in \C^{d_\bob \times
d_\bob}$, and a vector $u \in \C^{d_\alice} \ot \C^{d_\bob}$; it is \emph{valid} if each
$\{A^x_a\}_a$ and each $\{B^y_b\}_b$ is a projective measurement and $u \neq 0$, and its
value is
\[
  \omega \;=\; \frac{1}{\norm{u}^2} \sum_{x,y} \mu(x,y) \sum_{a,b} D(x,y,a,b)\,
  \bra{u} A^x_a \ot B^y_b \ket{u},
\]
a rational function of the entries. Suppose the answer alphabets of $\game$ are nonempty.
Then for every real $t$, $\valstar(\game) > t$ if and only if some valid exact strategy
with all entries in $\mathbb{Q}(i)$ has value $\omega > t$.
\end{lemma}

\begin{proof}
\leanok
A valid exact strategy with the state normalized to $u/\norm{u}$ is a tensor-product
strategy of value $\omega$, so $\omega \leq \valstar(\game)$. Conversely, if
$\valstar(\game) > t$ then, $\valstar$ being a supremum over a nonempty set, some
tensor-product strategy $(\psi, \{A^x_a\}, \{B^y_b\})$ has value $v > t$, and it suffices
to find, for every $\varepsilon > 0$, a valid exact strategy with $\mathbb{Q}(i)$ entries of
value $> v - \varepsilon$. Replace each $\{A^x_a\}_a$ and $\{B^y_b\}_b$ by a $\mathbb{Q}(i)$
projective measurement within $\eta$ of it (Lemma~\ref{lem:rational-pvm-dense}), and
$\psi$ by an entrywise rounding $u \neq 0$ with $\norm{u - \psi} \leq \eta/4$, so that
$\norm{u/\norm{u} - \psi} \leq \eta/2$. The value is Lipschitz in the data: by
Lemma~\ref{lem:perturbation-split}, a Born term $\bra{\psi} A \ot B \ket{\psi}$ changes by
at most $\norm{A - A'}\norm{B} + \norm{A'}\norm{B - B'} + 2\norm{A'}\norm{B'}\norm{\psi -
\psi'}$ between unit vectors $\psi, \psi'$, using $\norm{A \ot B} \leq \norm{A}\norm{B}$;
projections have norm at most one, so each term moves by at most $4\eta$ and the value, a
$\mu$-average of sums of at most $\abs{\cal A}\abs{\cal B}$ such terms, by at most
$4\abs{\cal A}\abs{\cal B}\eta$. Take $\eta < \varepsilon / (4\abs{\cal A}\abs{\cal B})$.
\end{proof}

\paragraph{Comments.} The lemma is the mathematical content of
Lemma~\ref{lem:value-lower-approx}, isolated from the computability: only the right-hand
side involves the enumeration. It accounts for the ledger's stability and density nodes
(\texttt{1.1.7.2.3}, \texttt{1.1.7.2.4}) by a different proof, as explained under
Lemma~\ref{lem:value-lower-approx}: stability is vacuous for exact candidates, and density
is Lemma~\ref{lem:rational-pvm-dense} plus the Lipschitz bound, whose constant
$4\abs{\cal A}\abs{\cal B}$ is crude and immaterial. The nonemptiness of the answer
alphabets is what makes a strategy exist at all: with it $\valstar$ is a supremum over a
nonempty set, without it the supremum is $0$ by convention and the equivalence fails for
$t < 0$. Game descriptions (Definition~\ref{def:game-description}) always satisfy it. In
Lean the data type \texttt{ExactStrategy} carries no proofs, so that the computability layer
can enumerate it and test validity.

\begin{lemma}[Decidable candidate set: a norm constraint is two psd tests]
\label{lem:norm-two-psd}
\lean{MIPRE.ValueApprox.posSemidef_realSmul_one_add_and_sub_iff}
\leanok
\ledgernode{1.1.7.2.1}
Let $M$ be a Hermitian matrix and $c$ a real number. Then $c\Id + M \succeq 0$ and
$c\Id - M \succeq 0$ both hold if and only if $-c\norm{x}^2 \leq \bra{x} M \ket{x}
\leq c \norm{x}^2$ for every vector $x$ --- which for Hermitian $M$ says
$\norm{M} \leq c$.
\end{lemma}

\begin{proof}
\leanok
Both directions are the defining property of positive semidefiniteness applied to the
quadratic form, read once forwards and once backwards; the scalar step is
$\bra{x}(c\Id \pm M)\ket{x} = c\norm{x}^2 \pm \bra{x} M \ket{x}$.
\end{proof}

\begin{lemma}[The perturbation split]
\label{lem:perturbation-split}
\uses{def:q-value}
\lean{MIPRE.ValueApprox.dotProduct_mulVec_perturb,
MIPRE.ValueApprox.kronecker_sub_kronecker,
MIPRE.ValueApprox.dotProduct_kronecker_perturb}
\leanok
\ledgernode{1.1.7.2.3}
\ledgernode{1.1.7.2.4}
For matrices $M, N$ and vectors $\psi, \phi$,
\[
\bra{\psi} M \ket{\psi} - \bra{\phi} N \ket{\phi}
  = \bra{\psi} (M - N) \ket{\psi} + \bra{\psi - \phi} N \ket{\psi}
    + \bra{\phi} N \ket{\psi - \phi},
\]
and $A \ot B - A' \ot B' = (A - A') \ot B + A' \ot (B - B')$; combining the two splits a
bipartite Born expectation in which both players' operators and the state have moved into
four terms, each moving exactly one of $A$, $B$ and $\psi$.
\end{lemma}

\begin{proof}
\leanok
Expand and cancel. Both identities are exact: no norm and no estimate occurs in either,
and the analytic bounds are obtained afterwards by bounding the terms separately.
\end{proof}

\paragraph{Comments.} This is the easy inclusion $\MIPstar \subseteq \RE$ and one half
of Theorem~\ref{thm:mipstar-eq-re}. The pipeline consumes only the $\valstar$ half, as the
semidecision procedure of Remark~\ref{rem:compression-abstract}. The commuting-operator analogue (approximating
$\valco$ from \emph{above} via the NPA hierarchy~\cite{NPA08}) is \emph{not} needed for
the main theorem, but becomes relevant for the downstream consequences in
Section~\ref{ch:downstream}.
